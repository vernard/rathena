// ==================== HELPER FUNCTIONS ====================
// Format number as human readable (4200 -> "4.2k", 1500000 -> "1.5m")
function	script	F_ShortNum	{
	.@n = getarg(0);
	if (.@n >= 1000000) {
		.@whole = .@n / 1000000;
		.@decimal = (.@n % 1000000) / 100000;
		return .@whole + "." + .@decimal + "m";
	} else if (.@n >= 1000) {
		.@whole = .@n / 1000;
		.@decimal = (.@n % 1000) / 100;
		return .@whole + "." + .@decimal + "k";
	}
	return "" + .@n;
}

// Calculate cooldown/discovery bonus (returns 25-300 as percentage)
// Tracks hunts within a 3-day window per character per monster
// First time EVER = 300%, then within cooldown: 1-2 hunts = 100%, 3rd = 75%, 4th = 50%, 5th+ = 25%
function	script	F_BountyCooldown	{
	.@char_id = getarg(0);
	.@mob_id = getarg(1);
	.@cooldown_seconds = getarg(2);

	// Query cooldown data including first_clear flag
	.@rows = query_sql("SELECT `hunt_count`, `first_clear`, TIMESTAMPDIFF(SECOND, `first_hunt_time`, NOW()) as elapsed FROM `bounty_cooldowns` WHERE `char_id` = " + .@char_id + " AND `mob_id` = " + .@mob_id, .@hunt_count, .@first_clear, .@elapsed);

	// No record = first time ever hunting this monster!
	if (.@rows == 0 || (!.@hunt_count && !.@first_clear)) {
		return 300;  // Discovery bonus: triple rewards!
	}

	// Check if cooldown expired (3 days passed)
	if (.@cooldown_seconds > 0 && .@elapsed >= .@cooldown_seconds) {
		return 100;  // Cooldown expired, full rewards
	}

	// Cooldown active - return reward based on hunt count
	// hunt_count tracks completed hunts within the cooldown window
	if (.@hunt_count <= 1) return 100;  // 1st or 2nd hunt = full rewards (grace period)
	if (.@hunt_count == 2) return 75;   // 3rd hunt
	if (.@hunt_count == 3) return 50;   // 4th hunt
	return 25;                           // 5th+ hunt
}

// Get remaining cooldown time as formatted string
function	script	F_CooldownTimeLeft	{
	.@char_id = getarg(0);
	.@mob_id = getarg(1);
	.@cooldown_seconds = getarg(2);

	query_sql("SELECT TIMESTAMPDIFF(SECOND, NOW(), DATE_ADD(`first_hunt_time`, INTERVAL " + .@cooldown_seconds + " SECOND)) as remaining FROM `bounty_cooldowns` WHERE `char_id` = " + .@char_id + " AND `mob_id` = " + .@mob_id, .@remaining);

	if (.@remaining <= 0) return "";

	.@days = .@remaining / 86400;
	.@hours = (.@remaining % 86400) / 3600;
	.@mins = (.@remaining % 3600) / 60;

	if (.@days > 0)
		return .@days + "d " + .@hours + "h";
	if (.@hours > 0)
		return .@hours + "h " + .@mins + "m";
	return .@mins + "m";
}

				query_sql("INSERT INTO `bounty_cooldowns` (`char_id`, `mob_id`, `hunt_count`, `first_hunt_time`, `first_clear`) VALUES (" + getcharid(0) + ", " + Bounty_MobID + ", 1, NOW(), 1) ON DUPLICATE KEY UPDATE hunt_count = hunt_count + 1, first_hunt_time = IF(TIMESTAMPDIFF(SECOND, first_hunt_time, NOW()) >= " + .@cd_sec + ", NOW(), first_hunt_time), first_clear = 1");

				// Increment total hunts and clear bounty
				#HUNT_TOTAL += 1;
				Bounty_Map$ = "";
				Bounty_MobID = 0;
				Bounty_Target = 0;
				Bounty_Tier = 0;
				Bounty_Kills = 0;
				Bounty_StartKills = 0;
				Bounty_MOTD_Bonus = 0;

				specialeffect2 EF_ANGEL;
				specialeffect2 EF_TRUESIGHT;

				next;
				mes "[Bounty Board]";
				mes "Well done, hunter! You now have ^0055FF" + #HUNT_POINTS + "^000000 Hunt Points.";
				mes " ";
				mes "Total bounties completed: ^0055FF" + #HUNT_TOTAL + "^000000";
				mes " ";
				mes "Visit the ^0055FFMonster Hunter Guild^000000 in Prontera to spend your points!";
				close;

			case 2:
				close;
			}
		} else {
			switch(select(" ~ ^777777Close^000000: ~ Abandon Bounty")) {
			case 1:
				close;

			case 2:
				// Abandon Bounty
				mes "[Bounty Board]";
				mes "Abandon your ^0055FF" + getd(".MapName_" + Bounty_Map$) + "^000000 bounty?";
				mes " ";
				mes "Target: " + .@mob_name$;
				mes "Progress: " + Bounty_Kills + " / " + Bounty_Target;
				mes " ";
				mes "^FF8800All progress will be lost.^000000";
				next;
				if (select(" ~ ^777777No, keep it^000000: ~ Yes, abandon") == 1) {
					mes "[Bounty Board]";
					mes "Bounty kept.";
					close;
				} else {
					callsub S_ClearBounty;
					mes "[Bounty Board]";
					mes "Bounty abandoned.";
					mes " ";
					mes "You can now accept a bounty here!";
					dispbottom "[Bounty] Bounty abandoned.", 0xFF8800;
				}
				close;
			}
		}
	}
	end;

// ==================== SHARED SUBROUTINES ====================
// These reduce code duplication for reward status, calculations, and granting

// S_ClearBounty - Clears all bounty character variables
S_ClearBounty:
	Bounty_Map$ = "";
	Bounty_MobID = 0;
	Bounty_Target = 0;
	Bounty_Tier = 0;
	Bounty_StartKills = 0;
	Bounty_Kills = 0;
	Bounty_MOTD_Bonus = 0;
	return;

// S_ShowRewardStatus - Shows MOTD/Discovery/Cooldown status message
// Requires: .@cooldown_pct must be set before calling
S_ShowRewardStatus:
	// Safeguard: cooldown_pct should never be 0 or negative
	if (.@cooldown_pct < 25) .@cooldown_pct = 100;
	if (Bounty_MOTD_Bonus) {
		mes "^FF8800* Monster of the Day: 3x EXP & Points! *^000000";
	} else if (.@cooldown_pct >= 300) {
		mes "^00FF00* Discovery Bonus: 3x rewards! *^000000";
	} else if (.@cooldown_pct < 100) {
		mes "^FF8800(Cooldown: " + .@cooldown_pct + "% rewards)^000000";
	}
	return;

// S_CalculateEffectiveMult - Calculates effective multiplier (MOTD 3x or cooldown%)
// Requires: .@cooldown_pct must be set before calling
// Sets: .@effective_mult (100-300)
S_CalculateEffectiveMult:
	// Safeguard: cooldown_pct should never be 0 or negative
	if (.@cooldown_pct < 25) .@cooldown_pct = 100;
	if (Bounty_MOTD_Bonus) {
		.@effective_mult = 300;
	} else {
		.@effective_mult = .@cooldown_pct;
	}
	return;

// S_CalculateBaseRewards - Calculates base rewards before multipliers
// Requires: .@mob_lv, .@kills, .@mult must be set before calling
// Sets: .@base_exp, .@job_exp, .@zeny, .@points
S_CalculateBaseRewards:
	// EXP scaling: 1x at level 10, ~50x at level 80 (quadratic)
	.@exp_scale = (.@mob_lv > 10) ? ((.@mob_lv - 10) * (.@mob_lv - 10) / 98) : 1;
	if (.@exp_scale < 1) .@exp_scale = 1;

	.@base_exp = ((100 + .@mob_lv * 5) * .@kills * .@mult * .@exp_scale) / 100;
	.@job_exp = ((60 + .@mob_lv * 3) * .@kills * .@mult * .@exp_scale) / 100;
	.@zeny = ((20 + .@mob_lv) * .@kills * .@mult) / 5;
	.@points = ((.@mob_lv / 10 + 1) * .@mult * .@kills / 25) / 100;
	if (.@points < 1) .@points = 1;
	return;

// S_ApplyMultipliers - Applies effective multiplier and server EXP rate
// Requires: .@base_exp, .@job_exp, .@zeny, .@points, .@effective_mult must be set
// Modifies: .@base_exp, .@job_exp, .@zeny, .@points
S_ApplyMultipliers:
	.@base_exp = .@base_exp * .@effective_mult / 100;
	.@job_exp = .@job_exp * .@effective_mult / 100;
	.@zeny = .@zeny * .@effective_mult / 100;
	.@points = .@points * .@effective_mult / 100;
	if (.@points < 1) .@points = 1;

	// Apply server EXP rate
	.@exp_rate = getbattleflag("base_exp_rate");
	if (.@exp_rate < 1) .@exp_rate = 100;
	.@base_exp = .@base_exp * .@exp_rate / 100;
	.@job_exp = .@job_exp * .@exp_rate / 100;
	return;

// S_CalculateEstExp - Calculates estimated EXP percentage for display
// Requires: .@mob_lv, .@kills, .@mult, .@effective_mult must be set before calling
// Sets: .@est_pct (percentage of level)
S_CalculateEstExp:
	// Safeguards
	if (.@mult < 1) .@mult = 100;
	if (.@effective_mult < 25) .@effective_mult = 100;
	.@exp_scale = (.@mob_lv > 10) ? ((.@mob_lv - 10) * (.@mob_lv - 10) / 98) : 1;
	if (.@exp_scale < 1) .@exp_scale = 1;
	.@est_exp = ((100 + .@mob_lv * 5) * .@kills * .@mult * .@exp_scale) / 100;
	.@est_exp = .@est_exp * .@effective_mult / 100;
	.@exp_rate = getbattleflag("base_exp_rate");
	if (.@exp_rate < 1) .@exp_rate = 100;
	.@est_exp = .@est_exp * .@exp_rate / 100;
	.@next_exp = (NextBaseExp > 0) ? NextBaseExp : 1;
	.@est_pct = .@est_exp * 100 / .@next_exp;
	return;

// S_GrantBountyRewards - Grants rewards, logs to DB, updates cooldown, clears bounty
// Requires: .@base_exp, .@job_exp, .@zeny, .@points must be set before calling
S_GrantBountyRewards:
	// Grant rewards
	getexp .@base_exp, .@job_exp;
	Zeny += .@zeny;
	#HUNT_POINTS += .@points;

	// Log to database
	query_sql("INSERT INTO `bounty_log` (`char_id`, `account_id`, `map`, `mob_id`, `kill_target`, `tier`, `exp_reward`, `zeny_reward`, `points_reward`) VALUES (" +
		getcharid(0) + ", " + getcharid(3) + ", '" + escape_sql(Bounty_Map$) + "', " + Bounty_MobID + ", " + Bounty_Target + ", " + Bounty_Tier + ", " + .@base_exp + ", " + .@zeny + ", " + .@points + ")");

	// Update cooldown table
	.@cd_sec = getvariableofnpc(.CooldownSeconds, "BountyBoard");
	query_sql("INSERT INTO `bounty_cooldowns` (`char_id`, `mob_id`, `hunt_count`, `first_hunt_time`, `first_clear`) VALUES (" +
		getcharid(0) + ", " + Bounty_MobID + ", 1, NOW(), 1) ON DUPLICATE KEY UPDATE hunt_count = hunt_count + 1, first_hunt_time = IF(TIMESTAMPDIFF(SECOND, first_hunt_time, NOW()) >= " + .@cd_sec + ", NOW(), first_hunt_time), first_clear = 1");

	// Increment total hunts
	#HUNT_TOTAL += 1;

	// Clear bounty variables
	callsub S_ClearBounty;
	return;
}

// ==================== KILL EVENT HANDLER ====================
// Single invisible NPC to handle kill notifications (not duplicated)
-	script	BountyKillTracker	FAKE_NPC,{
	end;

OnNPCKillEvent:
	// Save kill info before any context switching
	.@killed_mob = killedrid;
	.@mob_name$ = getmonsterinfo(.@killed_mob, MOB_NAME);
	getmapxy(.@killer_map$, .@kx, .@ky, BL_PC);
	.@killer_aid = getcharid(3);
	.@killer_cid = getcharid(0);

	// Update killer's last_kill timestamp (for party AFK detection)
	// This ensures the "active" check has accurate data for all kills, not just rare drops
	query_sql("INSERT INTO `char_mob_kills` (`char_id`, `mob_id`, `kill_count`, `last_kill`) VALUES (" +
		.@killer_cid + ", " + .@killed_mob + ", 0, NOW()) ON DUPLICATE KEY UPDATE `last_kill` = NOW()");

	// If killer is under AFK penalty, track kills to potentially clear it
	if (Bounty_AFKPenalty > gettimetick(2)) {
		Bounty_PenaltyKills = Bounty_PenaltyKills + 1;
		if (Bounty_PenaltyKills >= 5) {
			Bounty_AFKPenalty = 0;  // Clear penalty
			Bounty_PenaltyKills = 0;
			dispbottom "[Bounty] Party sharing restored - welcome back!", 0x00FF00;
		}
	}

	// Get debug flag from config
	.@debug = getvariableofnpc(.DebugAFK, "BountyBoard");

	// Check if killer has a bounty for this monster
	.@killer_has_bounty = (Bounty_Map$ != "" && Bounty_MobID == .@killed_mob);

	// Update killer's bounty progress if they have one
	if (.@killer_has_bounty) {
		query_sql("SELECT COALESCE(kill_count, 0) FROM `char_mob_kills` WHERE `char_id` = " + .@killer_cid + " AND `mob_id` = " + Bounty_MobID, .@current_kills);
		Bounty_Kills = .@current_kills - Bounty_StartKills;
		if (Bounty_Kills < 0) Bounty_Kills = 0;

		if (.@debug) {
			if (Bounty_Kills >= Bounty_Target) {
				dispbottom "[Bounty] ^00FF00Complete!^000000 " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$ + " (self) - Return to claim reward!", 0x00FF00;
			} else {
				dispbottom "[Bounty] " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$ + " (self)", 0x00AAFF;
			}
		} else {
			if (Bounty_Kills >= Bounty_Target) {
				dispbottom "[Bounty] ^00FF00Complete!^000000 " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$ + " - Return to claim reward!", 0x00FF00;
			} else {
				dispbottom "[Bounty] " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$, 0x00AAFF;
			}
		}
	}

	// ==================== PARTY KILL SHARING ====================
	// Anti-AFK: Party members must be NEARBY (14 cells) OR have killed something recently (120s)
	.@party_id = getcharid(1);
	if (.@party_id == 0)
		end;  // Not in party, done

	// Get AFK config
	.@afk_timeout = getvariableofnpc(.AFKTimeout, "BountyBoard");
	if (.@afk_timeout < 1) .@afk_timeout = 60;
	.@stationary_timeout = getvariableofnpc(.StationaryTimeout, "BountyBoard");
	if (.@stationary_timeout < 1) .@stationary_timeout = 30;

	// Get party member lists and copy to LOCAL arrays immediately
	// (global $@ arrays can be overwritten by other events during the loop)
	getpartymember .@party_id, 1;  // Account IDs -> $@partymemberaid[]
	getpartymember .@party_id, 2;  // Char IDs -> $@partymembercid[]
	.@pm_count = $@partymembercount;
	for (.@j = 0; .@j < .@pm_count; .@j++) {
		.@pm_aid[.@j] = $@partymemberaid[.@j];
		.@pm_cid[.@j] = $@partymembercid[.@j];
	}

	// Loop through party members
	for (.@i = 0; .@i < .@pm_count; .@i++) {
		// Skip the killer (they already got their kill tracked)
		if (.@pm_aid[.@i] == .@killer_aid)
			continue;

		// Check if party member is online
		if (!isloggedin(.@pm_aid[.@i], .@pm_cid[.@i]))
			continue;

		// Check if party member is on same map
		getmapxy(.@pm_map$, .@px, .@py, BL_PC, rid2name(.@pm_aid[.@i]));
		if (.@pm_map$ != .@killer_map$)
			continue;

		// Switch context to party member early (needed for variable access)
		if (!attachrid(.@pm_aid[.@i]))
			continue;

		// ==================== AFK DETECTION ====================
		.@now = gettimetick(2);

		// Check if under 10-minute AFK penalty (from previous 3-minute stationary detection)
		if (Bounty_AFKPenalty > .@now) {
			if (.@debug && Bounty_MobID == .@killed_mob) {
				.@penalty_left = Bounty_AFKPenalty - .@now;
				dispbottom "[Bounty] ^FF0000BLOCKED^000000 - AFK penalty " + .@penalty_left + "s left (kill " + Bounty_PenaltyKills + "/5 to clear)", 0xFF0000;
			}
			attachrid(.@killer_aid);
			continue;  // Still under penalty
		}

		// Check 1: Is party member within 14 cells of the kill?
		// Bards/Dancers who stand still while buffing nearby should get credit
		.@distance = distance(.@kx, .@ky, .@px, .@py);
		.@is_nearby = (.@distance <= 14);
		.@reason$ = "";

		if (.@is_nearby) {
			// Nearby - reset movement tracking (so penalty doesn't accumulate)
			Bounty_LastX = .@px;
			Bounty_LastY = .@py;
			Bounty_LastMap$ = .@pm_map$;
			Bounty_LastMove = .@now;
			.@reason$ = "nearby " + .@distance + "c";
		} else {
			// NOT nearby - must be active OR moving

			// Track movement status
			.@is_stationary = false;
			.@stationary_time = 0;
			if (Bounty_LastX == .@px && Bounty_LastY == .@py && Bounty_LastMap$ == .@pm_map$) {
				.@stationary_time = .@now - Bounty_LastMove;

				// If stationary for 3+ minutes, apply 10-minute penalty (always blocks)
				if (.@stationary_time >= 180) {
					Bounty_AFKPenalty = .@now + 600;  // 10 minute penalty
					Bounty_PenaltyKills = 0;  // Reset redemption counter
					if (.@debug && Bounty_MobID == .@killed_mob) {
						dispbottom "[Bounty] ^FF0000BLOCKED^000000 - AFK 3min! 10-min penalty applied (kill 5 to clear)", 0xFF0000;
					}
					attachrid(.@killer_aid);
					continue;
				}

				// Stationary for too long
				.@is_stationary = (.@stationary_time >= .@stationary_timeout);
			} else {
				// Position changed - update movement tracking
				Bounty_LastX = .@px;
				Bounty_LastY = .@py;
				Bounty_LastMap$ = .@pm_map$;
				Bounty_LastMove = .@now;
			}

			// Check kill activity (killed something in last 60 seconds)
			.@idle_time = 0;
			.@has_kills = query_sql("SELECT TIMESTAMPDIFF(SECOND, MAX(last_kill), NOW()) FROM `char_mob_kills` WHERE `char_id` = " + .@pm_cid[.@i], .@idle_time);
			.@is_active = (.@has_kills > 0 && .@idle_time >= 0 && .@idle_time <= .@afk_timeout);

			// If active (killed recently), pass regardless of movement
			// If not active, must NOT be stationary for 15+ seconds
			if (!.@is_active && .@is_stationary) {
				if (.@debug && Bounty_MobID == .@killed_mob) {
					dispbottom "[Bounty] ^FF0000BLOCKED^000000 - stationary " + .@stationary_time + "s, no kills in " + .@afk_timeout + "s", 0xFF0000;
				}
				attachrid(.@killer_aid);
				continue;  // Not active + stationary = AFK
			}

			// Build reason string for debug
			if (.@is_active)
				.@reason$ = "active " + .@idle_time + "s";
			else
				.@reason$ = "moving " + .@stationary_time + "s";
		}

		// Check if party member has a bounty for this monster (same map already verified above)
		if (Bounty_MobID == .@killed_mob) {
			// Increment their kill count in database (share the kill)
			// NOTE: Do NOT update last_kill here - only the actual killer gets last_kill updated
			// This prevents AFK players from resetting their activity timer via shared kills
			query_sql("INSERT INTO `char_mob_kills` (`char_id`, `mob_id`, `kill_count`) VALUES (" +
				getcharid(0) + ", " + .@killed_mob + ", 1) ON DUPLICATE KEY UPDATE `kill_count` = `kill_count` + 1");

			// Update their Bounty_Kills variable
			query_sql("SELECT COALESCE(kill_count, 0) FROM `char_mob_kills` WHERE `char_id` = " + getcharid(0) + " AND `mob_id` = " + .@killed_mob, .@pm_kills);
			Bounty_Kills = .@pm_kills - Bounty_StartKills;
			if (Bounty_Kills < 0) Bounty_Kills = 0;

			// Notify party member (with debug info if enabled)
			if (.@debug) {
				if (Bounty_Kills >= Bounty_Target) {
					dispbottom "[Bounty] ^00FF00Complete!^000000 " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$ + " (party: " + .@reason$ + ") - Return to claim!", 0x00FF00;
				} else {
					dispbottom "[Bounty] " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$ + " (party: " + .@reason$ + ")", 0x00AAFF;
				}
			} else {
				if (Bounty_Kills >= Bounty_Target) {
					dispbottom "[Bounty] ^00FF00Complete!^000000 " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$ + " - Return to claim!", 0x00FF00;
				} else {
					dispbottom "[Bounty] " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$, 0x00AAFF;
				}
			}
		}

		// Switch back to killer

	// Check for active bounty
	if (Bounty_Map$ != "") {
		.@mob_name$ = getmonsterinfo(Bounty_MobID, MOB_NAME);

		// Update kill count from database
		query_sql("SELECT COALESCE(kill_count, 0) FROM `char_mob_kills` WHERE `char_id` = " + getcharid(0) + " AND `mob_id` = " + Bounty_MobID, .@current_kills);
		Bounty_Kills = .@current_kills - Bounty_StartKills;
		if (Bounty_Kills < 0) Bounty_Kills = 0;

		mes " ";
		mes "^777777--------------------^000000";
		if (Bounty_Kills >= Bounty_Target) {
			mes "^00FF00Active Bounty: COMPLETE!^000000";
		} else {
			mes "^FFAA00Active Bounty:^000000";
		}
		mes "  " + .@mob_name$ + " [" + Bounty_MobID + "] (" + Bounty_Kills + "/" + Bounty_Target + ")";

		// Check cooldown/discovery status and show reward status
		.@cooldown_pct = callfunc("F_BountyCooldown", getcharid(0), Bounty_MobID, getvariableofnpc(.CooldownSeconds, "BountyBoard"));
		if (.@cooldown_pct < 25) .@cooldown_pct = 100;
		if (Bounty_MOTD_Bonus) {
			mes "  ^FF8800* Monster of the Day: 3x! *^000000";
		} else if (.@cooldown_pct >= 300) {
			mes "  ^00FF00* Discovery: 3x rewards! *^000000";
		} else if (.@cooldown_pct < 100) {
			mes "  ^FF8800(Cooldown: " + .@cooldown_pct + "%)^000000";
		}

		// Calculate estimated EXP (same formula as S_CalculateEstExp)
		.@mob_lv = getmonsterinfo(Bounty_MobID, MOB_LV);
		.@mult = 100;  // Single tier
		.@effective_mult = Bounty_MOTD_Bonus ? 300 : .@cooldown_pct;
		.@exp_scale = (.@mob_lv > 10) ? ((.@mob_lv - 10) * (.@mob_lv - 10) / 98) : 1;
		if (.@exp_scale < 1) .@exp_scale = 1;
		.@est_exp = ((100 + .@mob_lv * 5) * Bounty_Target * .@mult * .@exp_scale) / 100;
		.@est_exp = .@est_exp * .@effective_mult / 100;
		.@exp_rate = getbattleflag("base_exp_rate");
		if (.@exp_rate < 1) .@exp_rate = 100;
		.@est_exp = .@est_exp * .@exp_rate / 100;
		.@next_exp = (NextBaseExp > 0) ? NextBaseExp : 1;
		.@est_pct = .@est_exp * 100 / .@next_exp;
		if (NextBaseExp > 0)
			mes "  Est. EXP: ~" + .@est_pct + "% of level";
	}
	next;

	// Build menu based on bounty status
	if (Bounty_Map$ != "" && Bounty_Kills >= Bounty_Target) {
		// Bounty complete - show claim option first
		switch(select(" ~ ^00FF00Claim Bounty Rewards^000000: ~ Suggest Monsters: ~ Browse Shop: ~ View Cooldowns: ~ View Top Hunters: ~ About Hunt Points: ~ ^777777Leave^000000")) {
		case 1:
			goto L_ClaimBounty;
		case 2:
			goto L_SuggestMonsters;
		case 3:
			goto L_Shop;
		case 4:
			goto L_Cooldowns;
		case 5:
			goto L_TopHunters;
		case 6:
			goto L_About;
		case 7:
			goto L_Leave;
		}
	} else if (Bounty_Map$ != "") {
		// Bounty in progress - show progress option
		switch(select(" ~ View Bounty Progress: ~ Suggest Monsters: ~ Browse Shop: ~ View Cooldowns: ~ View Top Hunters: ~ About Hunt Points: ~ ^777777Leave^000000")) {
		case 1:
			goto L_BountyProgress;
		case 2:
			goto L_SuggestMonsters;
		case 3:
			goto L_Shop;
		case 4:
			goto L_Cooldowns;
		case 5:
			goto L_TopHunters;
		case 6:
			goto L_About;
		case 7:
			goto L_Leave;
		}
	} else {
		// No active bounty
		switch(select(" ~ Suggest Monsters: ~ Browse Shop: ~ View Cooldowns: ~ View Top Hunters: ~ About Hunt Points: ~ ^777777Leave^000000")) {
		case 1:
			goto L_SuggestMonsters;
		case 2:
			goto L_Shop;
		case 3:
			goto L_Cooldowns;
		case 4:
			goto L_TopHunters;
		case 5:
			goto L_About;
		case 6:
			goto L_Leave;
		}
	}
	end;

L_ClaimBounty:
	.@mob_name$ = getmonsterinfo(Bounty_MobID, MOB_NAME);
	.@mob_lv = getmonsterinfo(Bounty_MobID, MOB_LV);

	// Level check - if too low, no EXP (prevents trans exploit)
	.@min_player_lv = max(1, .@mob_lv - 10);
	.@level_ok = (BaseLevel >= .@min_player_lv);

	.@mult = 100;  // Single tier
	.@kills = Bounty_Target;

	// Calculate base rewards (same formula as BountyBoard::S_CalculateBaseRewards)
	.@exp_scale = (.@mob_lv > 10) ? ((.@mob_lv - 10) * (.@mob_lv - 10) / 98) : 1;
	if (.@exp_scale < 1) .@exp_scale = 1;
	.@base_exp = ((100 + .@mob_lv * 5) * .@kills * .@mult * .@exp_scale) / 100;
	.@job_exp = ((60 + .@mob_lv * 3) * .@kills * .@mult * .@exp_scale) / 100;
	.@zeny = ((20 + .@mob_lv) * .@kills * .@mult) / 5;
	.@points = ((.@mob_lv / 10 + 1) * .@mult * .@kills / 25) / 100;
	if (.@points < 1) .@points = 1;

	// Get cooldown and calculate effective multiplier (MOTD 3x or cooldown%)
	.@cooldown_pct = callfunc("F_BountyCooldown", getcharid(0), Bounty_MobID, getvariableofnpc(.CooldownSeconds, "BountyBoard"));
	if (.@cooldown_pct < 25) .@cooldown_pct = 100;
	.@effective_mult = Bounty_MOTD_Bonus ? 300 : .@cooldown_pct;

	// Apply multiplier and server EXP rate (same as BountyBoard::S_ApplyMultipliers)
	.@base_exp = .@base_exp * .@effective_mult / 100;
	.@job_exp = .@job_exp * .@effective_mult / 100;
	.@zeny = .@zeny * .@effective_mult / 100;
	.@points = .@points * .@effective_mult / 100;
	if (.@points < 1) .@points = 1;
	.@exp_rate = getbattleflag("base_exp_rate");
	if (.@exp_rate < 1) .@exp_rate = 100;
	.@base_exp = .@base_exp * .@exp_rate / 100;
	.@job_exp = .@job_exp * .@exp_rate / 100;

	// Calculate % of player's level for display
	.@next_exp = (NextBaseExp > 0) ? NextBaseExp : 1;
	.@pct = .@base_exp * 100 / .@next_exp;

	mes "[Monster Hunter Guild]";
	mes "^00FF00Bounty Complete!^000000";
	mes "^0055FF" + Bounty_Target + "x " + .@mob_name$ + "^000000 defeated!";
	// Show reward status
	if (Bounty_MOTD_Bonus) {
		mes "^FF8800* Monster of the Day: 3x EXP & Points! *^000000";
	} else if (.@cooldown_pct >= 300) {
		mes "^00FF00* Discovery Bonus: 3x rewards! *^000000";
	} else if (.@cooldown_pct < 100) {
		mes "^FF8800(Cooldown: " + .@cooldown_pct + "% rewards)^000000";
	}
	mes " ";
	mes "Rewards:";
	if (.@level_ok) {
		mes "  EXP: ^0055FF~" + .@pct + "% of level^000000";
	} else {
		mes "  EXP: ^FF0000None (Lv " + .@min_player_lv + "+ required)^000000";
	}
	mes "  Zeny: ^0055FF" + callfunc("F_ShortNum", .@zeny) + "^000000";
	mes "  Hunt Points: ^0055FF" + .@points + "^000000";

	// Grant rewards (no EXP if under-leveled - prevents trans exploit)
	if (.@level_ok) {
		getexp .@base_exp, .@job_exp;
	}
	Zeny += .@zeny;
	#HUNT_POINTS += .@points;

	// Log to database (log 0 EXP if skipped)
	.@logged_exp = .@level_ok ? .@base_exp : 0;
	query_sql("INSERT INTO `bounty_log` (`char_id`, `account_id`, `map`, `mob_id`, `kill_target`, `tier`, `exp_reward`, `zeny_reward`, `points_reward`) VALUES (" +
		getcharid(0) + ", " + getcharid(3) + ", '" + escape_sql(Bounty_Map$) + "', " + Bounty_MobID + ", " + Bounty_Target + ", " + Bounty_Tier + ", " + .@logged_exp + ", " + .@zeny + ", " + .@points + ")");

	// Update bounty cooldown (single query with ON DUPLICATE KEY, same as S_GrantBountyRewards)
	.@cd_sec = getvariableofnpc(.CooldownSeconds, "BountyBoard");
	query_sql("INSERT INTO `bounty_cooldowns` (`char_id`, `mob_id`, `hunt_count`, `first_hunt_time`, `first_clear`) VALUES (" +
		getcharid(0) + ", " + Bounty_MobID + ", 1, NOW(), 1) ON DUPLICATE KEY UPDATE hunt_count = hunt_count + 1, first_hunt_time = IF(TIMESTAMPDIFF(SECOND, first_hunt_time, NOW()) >= " + .@cd_sec + ", NOW(), first_hunt_time), first_clear = 1");

	#HUNT_TOTAL += 1;

	// Store for display before clearing
	.@completed_mob = Bounty_MobID;

	// Clear bounty (same as S_ClearBounty)
	Bounty_Map$ = "";
	Bounty_MobID = 0;
	Bounty_Target = 0;
	Bounty_Tier = 0;
	Bounty_StartKills = 0;
	Bounty_Kills = 0;
	Bounty_MOTD_Bonus = 0;

	specialeffect2 EF_ANGEL;
	specialeffect2 EF_TRUESIGHT;

	next;
	mes "[Monster Hunter Guild]";
	mes "Well done, hunter!";
	mes " ";
	mes "You now have ^0055FF" + #HUNT_POINTS + "^000000 Hunt Points.";
	mes "Total bounties completed: ^0055FF" + #HUNT_TOTAL + "^000000";
	// Show next cooldown status
	.@next_cooldown = callfunc("F_BountyCooldown", getcharid(0), .@completed_mob, getvariableofnpc(.CooldownSeconds, "BountyBoard"));
	if (.@next_cooldown < 100) {
		.@time_left$ = callfunc("F_CooldownTimeLeft", getcharid(0), .@completed_mob, getvariableofnpc(.CooldownSeconds, "BountyBoard"));
		mes " ";
		mes "^FF8800Note: " + .@mob_name$ + " now at " + .@next_cooldown + "% rewards^000000";
		mes "^777777Full rewards in " + .@time_left$ + "^000000";
	}
	close;

L_BountyProgress:
	.@mob_name$ = getmonsterinfo(Bounty_MobID, MOB_NAME);
	.@remaining = Bounty_Target - Bounty_Kills;

	mes "[Monster Hunter Guild]";
	mes "Your active bounty:";
	mes " ";
	mes "^777777Location:^000000 ^0055FF" + Bounty_Map$ + "^000000";
	mes "^777777Target:^000000 ^0055FF" + .@mob_name$ + "^000000";
	mes "^777777Progress:^000000 ^0055FF" + Bounty_Kills + " / " + Bounty_Target + "^000000";

	// Check cooldown/discovery status
	.@cooldown_pct = callfunc("F_BountyCooldown", getcharid(0), Bounty_MobID, getvariableofnpc(.CooldownSeconds, "BountyBoard"));
	if (.@cooldown_pct < 25) .@cooldown_pct = 100;

	// Show reward status
	if (Bounty_MOTD_Bonus) {
		mes "^FF8800* Monster of the Day: 3x EXP & Points! *^000000";
	} else if (.@cooldown_pct >= 300) {
		mes "^00FF00* Discovery Bonus: 3x rewards! *^000000";
	} else if (.@cooldown_pct < 100) {
		mes "^FF8800(Cooldown: " + .@cooldown_pct + "% rewards)^000000";
	}

	// Calculate estimated EXP (same formula as S_CalculateEstExp)
	.@mob_lv = getmonsterinfo(Bounty_MobID, MOB_LV);
	.@mult = 100;  // Single tier
	.@effective_mult = Bounty_MOTD_Bonus ? 300 : .@cooldown_pct;
	.@exp_scale = (.@mob_lv > 10) ? ((.@mob_lv - 10) * (.@mob_lv - 10) / 98) : 1;
	if (.@exp_scale < 1) .@exp_scale = 1;
	.@est_exp = ((100 + .@mob_lv * 5) * Bounty_Target * .@mult * .@exp_scale) / 100;
	.@est_exp = .@est_exp * .@effective_mult / 100;
	.@exp_rate = getbattleflag("base_exp_rate");
	if (.@exp_rate < 1) .@exp_rate = 100;
	.@est_exp = .@est_exp * .@exp_rate / 100;
	.@next_exp = (NextBaseExp > 0) ? NextBaseExp : 1;
	.@est_pct = .@est_exp * 100 / .@next_exp;
	if (NextBaseExp > 0)
		mes "^777777Est. EXP:^000000 ^0055FF~" + .@est_pct + "% of level^000000";
	mes " ";
	mes "^FF8800" + .@remaining + " more to go!^000000";
	mes " ";
	mes "Keep hunting and return when done!";
	close;

L_TopHunters:
	mes "[Monster Hunter Guild]";
	mes "The top hunters are:";
	mes " ";
	query_sql("SELECT c.`name`, SUM(b.`points_reward`) as total_points, COUNT(*) as bounties FROM `bounty_log` b JOIN `char` c ON b.char_id = c.char_id GROUP BY b.char_id ORDER BY total_points DESC LIMIT 10", .@name$, .@points, .@count);
	for (.@i = 0; .@i < getarraysize(.@name$); .@i++) {
		mes "^0055FF#" + (.@i+1) + "^000000 " + .@name$[.@i] + " - " + .@points[.@i] + " pts (" + .@count[.@i] + " bounties)";
	}
	if (!getarraysize(.@name$)) {
		mes "^777777No hunters yet!^000000";
		mes "Be the first to complete a bounty!";
	}
	close;

L_Cooldowns:
	mes "[Monster Hunter Guild]";
	mes "Your active bounty cooldowns:";
	mes " ";

	// Query all active cooldowns for this character (within 3 days)
	.@cooldown_seconds = getvariableofnpc(.CooldownSeconds, "BountyBoard");
	.@count = query_sql("SELECT bc.mob_id, bc.hunt_count, TIMESTAMPDIFF(SECOND, NOW(), DATE_ADD(bc.first_hunt_time, INTERVAL " + .@cooldown_seconds + " SECOND)) as remaining FROM `bounty_cooldowns` bc WHERE bc.char_id = " + getcharid(0) + " AND TIMESTAMPDIFF(SECOND, bc.first_hunt_time, NOW()) < " + .@cooldown_seconds + " ORDER BY remaining ASC LIMIT 15", .@mob_ids, .@hunt_counts, .@remaining);

	if (.@count == 0) {
		mes "^00AA00No active cooldowns!^000000";
		mes "All bounties offer full rewards.";
	} else {
		for (.@i = 0; .@i < .@count; .@i++) {
			.@mob_name$ = getmonsterinfo(.@mob_ids[.@i], MOB_NAME);

			// Calculate time remaining
			.@days = .@remaining[.@i] / 86400;
			.@hours = (.@remaining[.@i] % 86400) / 3600;
			.@mins = (.@remaining[.@i] % 3600) / 60;

			if (.@days > 0)
				.@time$ = .@days + "d " + .@hours + "h";
			else if (.@hours > 0)
				.@time$ = .@hours + "h " + .@mins + "m";
			else
				.@time$ = .@mins + "m";

			// Determine reward percentage based on hunt count
			if (.@hunt_counts[.@i] == 1) .@pct = 75;
			else if (.@hunt_counts[.@i] == 2) .@pct = 50;
			else .@pct = 25;

			// Color code by severity
			if (.@pct >= 75)
				mes "^D29922" + .@mob_name$ + "^000000 - " + .@pct + "% (" + .@time$ + ")";
			else if (.@pct >= 50)
				mes "^FF8800" + .@mob_name$ + "^000000 - " + .@pct + "% (" + .@time$ + ")";
			else
				mes "^FF0000" + .@mob_name$ + "^000000 - " + .@pct + "% (" + .@time$ + ")";
		}
		mes " ";
		mes "^777777Cooldowns reset 3 days after^000000";
		mes "^777777first hunt of that monster.^000000";
	}
	close;

L_SuggestMonsters:
	// Build the suggestion list synchronously
	// Results stored in @suggest_* player variables
	callfunc "F_BuildSuggestions";

	if (@suggest_count == 0) {
		mes "[Monster Hunter Guild]";
		mes "^FF8800No suggestions available.^000000";
		mes "Try visiting some dungeons first!";
		goto L_SuggestCleanup;
	}

	// Display suggestions grouped by location
	@suggest_shown = 0;

L_SuggestDisplay:
	mes "[Monster Hunter Guild]";
	if (@suggest_shown == 0) {
		mes "Best bounties for Lv" + BaseLevel + ":";
	} else {
		mes "More suggestions:";
	}
	mes " ";

	.@show_count = 0;
	.@last_loc$ = "";
	.@start = @suggest_shown;
	.@limit = (@suggest_count > @suggest_shown + 10) ? @suggest_shown + 10 : @suggest_count;

	for (.@i = .@start; .@i < .@limit; .@i++) {
		// Show location header if changed
		if (@suggest_loc$[.@i] != .@last_loc$) {
			if (.@last_loc$ != "") mes " ";
			mes "^0055FF" + @suggest_loc$[.@i] + "^000000";
			.@last_loc$ = @suggest_loc$[.@i];
		}

		.@name$ = getmonsterinfo(@suggest_mob[.@i], MOB_NAME);
		.@zeny_k = @suggest_zeny[.@i] / 1000;

		// Build reward string
		if (@suggest_cd[.@i] >= 300) {
			// Discovery bonus - highlight it
			.@reward$ = "^00FF00[3x] " + @suggest_exp[.@i] + "% " + .@zeny_k + "k^000000";
		} else if (@suggest_cd[.@i] < 100) {
			// On cooldown - show reduced and dim
			.@reward$ = "^FF8800" + @suggest_exp[.@i] + "% " + .@zeny_k + "k (" + @suggest_cd[.@i] + "%)^000000";
		} else {
			// Normal
			.@reward$ = @suggest_exp[.@i] + "% " + .@zeny_k + "k";
		}

		mes "  ^777777Lv" + @suggest_lv[.@i] + "^000000 " + .@name$ + " " + .@reward$;
		.@show_count++;
		@suggest_shown++;
	}

	// Show more option if there are more
	if (@suggest_shown < @suggest_count) {
		next;
		if (select(" ~ Show " + (@suggest_count - @suggest_shown) + " more: ~ Done") == 1) {
			goto L_SuggestDisplay;
		}
	}

L_SuggestCleanup:
	// Clean up player variables
	deletearray @suggest_mob;
	deletearray @suggest_lv;
	deletearray @suggest_map$;
	deletearray @suggest_loc$;
	deletearray @suggest_exp;
	deletearray @suggest_zeny;
	deletearray @suggest_cd;
	deletearray @suggest_score;
	@suggest_count = 0;
	@suggest_shown = 0;
	close;

L_Leave:

	close;

L_Shop:
	callshop "hunt_point_shop", 1;
	npcshopattach "hunt_point_shop";
	end;

OnBuyItem:
	// Calculate total cost
	.@size = getarraysize(@bought_nameid);
	.@total_cost = 0;

	for (.@i = 0; .@i < .@size; .@i++) {
		for (.@j = 0; .@j < getarraysize(.ShopItems); .@j++) {
			if (@bought_nameid[.@i] == .ShopItems[.@j]) {
				.@total_cost += (.ShopCosts[.@j] * @bought_quantity[.@i]);
				break;
			}
		}
	}

	mes "[Monster Hunter Guild]";

	if (#HUNT_POINTS < .@total_cost) {
		mes "You don't have enough Hunt Points!";
		mes " ";
		mes "Cost: ^FF0000" + .@total_cost + "^000000 points";
		mes "Have: ^0055FF" + #HUNT_POINTS + "^000000 points";
		close;
	}

	// Check inventory space
	for (.@i = 0; .@i < .@size; .@i++) {
		if (checkweight(@bought_nameid[.@i], @bought_quantity[.@i]) == 0) {
			mes "Your inventory is too full!";
			mes "Please make some room.";
			close;
		}
	}

	// Deduct Hunt Points
	#HUNT_POINTS -= .@total_cost;

	// Give purchased items
	for (.@i = 0; .@i < .@size; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
	}

	mes "Purchase complete!";
	mes " ";
	mes "^0055FFPoints spent: " + .@total_cost + "^000000";
	mes "^0055FFRemaining: " + #HUNT_POINTS + " Hunt Points^000000";

	specialeffect2 EF_STEAL;

	// Cleanup
	deletearray @bought_nameid[0], .@size;
	deletearray @bought_quantity[0], .@size;
	close;

OnInit:
	// Shop items and costs (parallel arrays)
	setarray .ShopItems[0],
		608,    // Yggdrasil Seed
		607,    // Yggdrasil Berry
		12075,  // Str Dish10
		12090,  // Agi Dish10
		12085,  // Vit Dish10
		12080,  // Int Dish10
		12095,  // Dex Dish10
		12100,  // Luk Dish10
		4103,   // Horong Card (Sight)
		4040,   // Creamy Card (Teleport)
		7776;   // Gym Pass (+200 weight)

	setarray .ShopCosts[0],
		3,      // Yggdrasil Seed
		5,      // Yggdrasil Berry
		30,     // Str Dish10
		30,     // Agi Dish10
		30,     // Vit Dish10
		30,     // Int Dish10
		30,     // Dex Dish10
		30,     // Luk Dish10
		300,    // Horong Card
		400,    // Creamy Card
		200;    // Gym Pass

	// Build visual shop (prices shown are point costs)
	// Clear items first to prevent duplicates on reload
	npcshopdelitem "hunt_point_shop", 512;
	for (.@i = 0; .@i < getarraysize(.ShopItems); .@i++) {
		npcshopdelitem "hunt_point_shop", .ShopItems[.@i];
	}
	// Add items fresh
	for (.@i = 0; .@i < getarraysize(.ShopItems); .@i++) {
		npcshopadditem "hunt_point_shop", .ShopItems[.@i], .ShopCosts[.@i];
	}

	// Warmup: access BountyBoard to ensure cross-NPC calls work on first player interaction
	.@warmup = getvariableofnpc(.Initialized, "BountyBoard");
	end;
}
