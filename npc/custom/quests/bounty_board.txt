// ==================== HELPER FUNCTION ====================
// Calculate overleveled penalty (returns 20-100 as percentage)
// 30 level grace period, then -5% per level, minimum 50%, severe cap 20% at 55+ levels
function	script	F_OverlevelPenalty	{
	.@player_lv = getarg(0);
	.@mob_lv = getarg(1);
	.@level_diff = .@player_lv - .@mob_lv;

	// No penalty for hunting same level or higher monsters
	if (.@level_diff <= 30)
		return 100;

	// Severe penalty for huge gaps (55+ levels over)
	if (.@level_diff > 55)
		return 20;

	// Moderate penalty: -5% per level over 30, minimum 50%
	.@penalty = 100 - ((.@level_diff - 30) * 5);
	return max(50, .@penalty);
}

// Format number as human readable (4200 → "4.2k", 1500000 → "1.5m")
function	script	F_ShortNum	{
	.@n = getarg(0);
	if (.@n >= 1000000) {
		.@whole = .@n / 1000000;
		.@decimal = (.@n % 1000000) / 100000;
		return .@whole + "." + .@decimal + "m";
	} else if (.@n >= 1000) {
		.@whole = .@n / 1000;
		.@decimal = (.@n % 1000) / 100;
		return .@whole + "." + .@decimal + "k";
	}
	return "" + .@n;
}

// ==================== MASTER NPC SCRIPT ====================
-	script	BountyBoard	-1,{
	// Get map-specific configuration
	// NPC unique name suffix format: "mapname" or "mapname_2", "mapname_3", etc.
	.@npc_suffix$ = strnpcinfo(4);

	// Strip any "_N" suffix to get the base map name for config lookup
	// Check if the name ends with _2, _3, etc.
	.@underscore_pos = getstrlen(.@npc_suffix$) - 2;
	if (.@underscore_pos > 0 && charat(.@npc_suffix$, .@underscore_pos) == "_") {
		.@map$ = substr(.@npc_suffix$, 0, .@underscore_pos);
	} else {
		.@map$ = .@npc_suffix$;
	}

	// Find this map in our configuration
	.@map_idx = inarray(.Maps$, .@map$);
	if (.@map_idx < 0) {
		mes "[Bounty Board]";
		mes "^FF0000Error: This bounty board is not configured.^000000";
		close;
	}

	// Get monster IDs for this map
	.@mob_count = getd(".Mobs_" + .@map$ + "_count");
	if (.@mob_count < 1) {
		mes "[Bounty Board]";
		mes "^FF0000Error: No monsters configured for this location.^000000";
		close;
	}

	// Copy monster IDs to local array
	for (.@i = 0; .@i < .@mob_count; .@i++) {
		.@mobs[.@i] = getd(".Mobs_" + .@map$ + "[" + .@i + "]");
	}

	mes "[Bounty Board]";
	mes "Welcome, adventurer!";
	mes "This is the ^0055FF" + getd(".MapName_" + .@map$) + "^000000 Bounty Board.";
	mes " ";

	// Check if player has active bounty
	if (Bounty_Map$ != "") {
		if (Bounty_Map$ == .@map$) {
			mes "^00AA00You have an active bounty here.^000000";
			mes "Target: ^0055FF" + getmonsterinfo(Bounty_MobID, MOB_NAME) + "^000000";
			mes "Progress: ^0055FF" + Bounty_Kills + " / " + Bounty_Target + "^000000";
		} else {
			mes "^FF8800You have an active bounty at another location.^000000";
			mes "Complete or abandon it first.";
		}
	} else {
		mes "Select a bounty to begin hunting!";
	}
	next;

	// Menu options
	if (Bounty_Map$ == "") {
		// No active bounty - show monster selection
		mes "[Bounty Board]";
		mes "Available bounties:";
		mes " ";

		// Build menu of monsters
		.@menu$ = "";
		for (.@i = 0; .@i < .@mob_count; .@i++) {
			.@mob_id = .@mobs[.@i];
			.@mob_name$ = getmonsterinfo(.@mob_id, MOB_NAME);
			.@mob_lv = getmonsterinfo(.@mob_id, MOB_LV);
			mes "^0055FF[" + (.@i+1) + "]^000000 " + .@mob_name$ + " (Lv " + .@mob_lv + ")";
			.@menu$ = .@menu$ + " ~ " + .@mob_name$ + ":";
		}
		.@menu$ = .@menu$ + " ~ ^777777Cancel^000000";
		next;

		.@choice = select(.@menu$) - 1;
		if (.@choice >= .@mob_count) {
			mes "[Bounty Board]";
			mes "Come back when you're ready to hunt!";
			close;
		}

		// Selected monster
		.@selected_mob = .@mobs[.@choice];
		.@mob_name$ = getmonsterinfo(.@selected_mob, MOB_NAME);
		.@mob_lv = getmonsterinfo(.@selected_mob, MOB_LV);

		mes "[Bounty Board]";
		mes "Target: ^0055FF" + .@mob_name$ + "^000000 (Lv " + .@mob_lv + ")";
		mes " ";
		mes "Select how many to hunt:";
		mes "^777777Higher tiers give bonus rewards!^000000";
		next;

		// Calculate rewards for each tier
		// Get player's EXP to next level for percentage calculation
		.@next_exp = (NextBaseExp > 0) ? NextBaseExp : 1;

		// Calculate overleveled penalty (applies to all rewards)
		.@penalty_pct = callfunc("F_OverlevelPenalty", BaseLevel, .@mob_lv);

		// EXP scaling: 1x at level 10, ~50x at level 80 (quadratic)
		// Formula: max(1, (mob_lv - 10)^2 / 98) gives 1 at lv10, 16 at lv50, 50 at lv80
		.@exp_scale = (.@mob_lv > 10) ? ((.@mob_lv - 10) * (.@mob_lv - 10) / 98) : 1;
		if (.@exp_scale < 1) .@exp_scale = 1;

		for (.@t = 0; .@t < getarraysize(.Tiers); .@t++) {
			.@kills = .Tiers[.@t];
			.@mult = .TierMult[.@t];

			// EXP formula with quadratic level scaling for meaningful high-level rewards
			.@base_exp[.@t] = ((100 + .@mob_lv * 5) * .@kills * .@mult * .@exp_scale) / 100;
			.@job_exp[.@t] = ((60 + .@mob_lv * 3) * .@kills * .@mult * .@exp_scale) / 100;
			.@zeny[.@t] = ((20 + .@mob_lv) * .@kills * .@mult) / 5;
			.@points[.@t] = ((.@mob_lv / 10 + 1) * .@mult * .@kills / 25) / 100;
			if (.@points[.@t] < 1) .@points[.@t] = 1;

			// Apply overleveled penalty to all rewards
			.@base_exp[.@t] = .@base_exp[.@t] * .@penalty_pct / 100;
			.@job_exp[.@t] = .@job_exp[.@t] * .@penalty_pct / 100;
			.@zeny[.@t] = .@zeny[.@t] * .@penalty_pct / 100;
			.@points[.@t] = .@points[.@t] * .@penalty_pct / 100;

			// Minimum rewards after penalty
			if (.@points[.@t] < 1) .@points[.@t] = 1;

			// Calculate % of player's level
			.@pct[.@t] = .@base_exp[.@t] * 100 / .@next_exp;
		}

		mes "[Bounty Board]";
		// Show overleveled penalty warning
		if (.@penalty_pct < 100) {
			mes "^FF8800Warning: Overleveled penalty!^000000";
			mes "^777777You are Lv " + BaseLevel + " hunting Lv " + .@mob_lv + " monsters.^000000";
			mes "^FF0000Rewards reduced to " + .@penalty_pct + "%^000000";
			mes " ";
		}
		// Build tier display dynamically
		.@tier_count = getarraysize(.Tiers);
		for (.@t = 0; .@t < .@tier_count; .@t++) {
			.@bonus = .TierMult[.@t] - 100;
			if (.@bonus > 0)
				mes "^0055FF[" + (.@t+1) + "]^000000 Hunt " + .Tiers[.@t] + " - ~" + .@pct[.@t] + "% lvl, " + callfunc("F_ShortNum", .@zeny[.@t]) + " zeny, " + .@points[.@t] + " pts ^00AA00(+" + .@bonus + "%)^000000";
			else
				mes "^0055FF[" + (.@t+1) + "]^000000 Hunt " + .Tiers[.@t] + " - ~" + .@pct[.@t] + "% lvl, " + callfunc("F_ShortNum", .@zeny[.@t]) + " zeny, " + .@points[.@t] + " pts";
		}
		next;

		// Build menu dynamically
		.@menu$ = "";
		for (.@t = 0; .@t < .@tier_count; .@t++) {
			.@bonus = .TierMult[.@t] - 100;
			if (.@bonus > 0)
				.@menu$ = .@menu$ + " ~ Hunt " + .Tiers[.@t] + " (+" + .@bonus + "%):";
			else
				.@menu$ = .@menu$ + " ~ Hunt " + .Tiers[.@t] + ":";
		}
		.@menu$ = .@menu$ + " ~ ^777777Cancel^000000";
		.@tier_choice = select(.@menu$) - 1;

		if (.@tier_choice >= .@tier_count) {
			mes "[Bounty Board]";
			mes "Come back when you're ready!";
			close;
		}

		// Level requirement check - player must be within 20 levels of monster
		.@min_player_lv = max(1, .@mob_lv - 20);
		if (BaseLevel < .@min_player_lv) {
			mes "[Bounty Board]";
			mes "^FF0000You are too low level for this bounty!^000000";
			mes " ";
			mes "Target: ^0055FF" + .@mob_name$ + "^000000 (Lv " + .@mob_lv + ")";
			mes "Required level: ^0055FF" + .@min_player_lv + "+^000000";
			mes "Your level: ^FF0000" + BaseLevel + "^000000";
			close;
		}

		// Accept bounty
		.@target = .Tiers[.@tier_choice];

		// Get current kill count from char_mob_kills
		query_sql("SELECT COALESCE(kill_count, 0) FROM `char_mob_kills` WHERE `char_id` = " + getcharid(0) + " AND `mob_id` = " + .@selected_mob, .@start_kills);

		// Store bounty info
		Bounty_Map$ = .@map$;
		Bounty_MobID = .@selected_mob;
		Bounty_Target = .@target;
		Bounty_Tier = .@tier_choice + 1;
		Bounty_StartKills = .@start_kills;
		Bounty_Kills = 0;

		mes "[Bounty Board]";
		mes "Bounty accepted!";
		mes " ";
		mes "Target: ^0055FF" + .@mob_name$ + "^000000";
		mes "Amount: ^0055FF" + .@target + "^000000";
		mes " ";
		mes "Hunt them and return here to claim your reward.";
		mes "^777777Your kills are tracked automatically.^000000";
		specialeffect2 EF_STEAL;

		// Show in chat log for quick reference
		dispbottom "[Bounty] Accepted: Hunt " + .@target + "x " + .@mob_name$ + " on " + .@map$, 0x00AAFF;
		close;

	} else if (Bounty_Map$ == .@map$) {
		// Active bounty on THIS map
		.@mob_name$ = getmonsterinfo(Bounty_MobID, MOB_NAME);

		// Update kill count from database immediately
		query_sql("SELECT COALESCE(kill_count, 0) FROM `char_mob_kills` WHERE `char_id` = " + getcharid(0) + " AND `mob_id` = " + Bounty_MobID, .@current_kills);
		Bounty_Kills = .@current_kills - Bounty_StartKills;
		if (Bounty_Kills < 0) Bounty_Kills = 0;

		// Auto-claim if bounty is complete
		if (Bounty_Kills >= Bounty_Target) {
			// Calculate final rewards (same formula as tier selection)
			.@mob_lv = getmonsterinfo(Bounty_MobID, MOB_LV);
			.@mult = .TierMult[Bounty_Tier - 1];
			.@kills = Bounty_Target;

			// EXP scaling: 1x at level 10, ~50x at level 80 (quadratic)
			.@exp_scale = (.@mob_lv > 10) ? ((.@mob_lv - 10) * (.@mob_lv - 10) / 98) : 1;
			if (.@exp_scale < 1) .@exp_scale = 1;

			.@base_exp = ((100 + .@mob_lv * 5) * .@kills * .@mult * .@exp_scale) / 100;
			.@job_exp = ((60 + .@mob_lv * 3) * .@kills * .@mult * .@exp_scale) / 100;
			.@zeny = ((20 + .@mob_lv) * .@kills * .@mult) / 5;
			.@points = ((.@mob_lv / 10 + 1) * .@mult * .@kills / 25) / 100;
			if (.@points < 1) .@points = 1;

			// Apply overleveled penalty to all rewards
			.@penalty_pct = callfunc("F_OverlevelPenalty", BaseLevel, .@mob_lv);
			.@base_exp = .@base_exp * .@penalty_pct / 100;
			.@job_exp = .@job_exp * .@penalty_pct / 100;
			.@zeny = .@zeny * .@penalty_pct / 100;
			.@points = .@points * .@penalty_pct / 100;

			// Minimum rewards after penalty
			if (.@points < 1) .@points = 1;

			// Calculate % of player's level for display
			.@next_exp = (NextBaseExp > 0) ? NextBaseExp : 1;
			.@pct = .@base_exp * 100 / .@next_exp;

			mes "[Bounty Board]";
			mes "^00FF00Bounty Complete!^000000";
			mes "^0055FF" + Bounty_Target + "x " + .@mob_name$ + "^000000 defeated!";
			if (.@penalty_pct < 100) {
				mes "^FF8800(Overleveled: " + .@penalty_pct + "% rewards)^000000";
			}
			mes " ";
			mes "Rewards:";
			mes "  EXP: ^0055FF~" + .@pct + "% of level^000000";
			mes "  Zeny: ^0055FF" + callfunc("F_ShortNum", .@zeny) + "^000000";
			mes "  Hunt Points: ^0055FF" + .@points + "^000000";

			// Grant rewards (getexp handles multi-level ups properly)
			getexp .@base_exp, .@job_exp;
			Zeny += .@zeny;
			#HUNT_POINTS += .@points;

			// Log to database
			query_sql("INSERT INTO `bounty_log` (`char_id`, `account_id`, `map`, `mob_id`, `kill_target`, `tier`, `exp_reward`, `zeny_reward`, `points_reward`) VALUES (" +
				getcharid(0) + ", " + getcharid(3) + ", '" + escape_sql(Bounty_Map$) + "', " + Bounty_MobID + ", " + Bounty_Target + ", " + Bounty_Tier + ", " + .@base_exp + ", " + .@zeny + ", " + .@points + ")");

			// Increment completion counter
			#HUNT_TOTAL += 1;

			// Clear bounty
			Bounty_Map$ = "";
			Bounty_MobID = 0;
			Bounty_Target = 0;
			Bounty_Tier = 0;
			Bounty_StartKills = 0;
			Bounty_Kills = 0;

			specialeffect2 EF_ANGEL;
			specialeffect2 EF_TRUESIGHT;

			next;
			mes "[Bounty Board]";
			mes "Well done, hunter! You now have ^0055FF" + #HUNT_POINTS + "^000000 Hunt Points.";
			mes " ";
			mes "Total bounties completed: ^0055FF" + #HUNT_TOTAL + "^000000";
			mes " ";
			mes "Visit the ^0055FFMonster Hunter Guild^000000 in Prontera to spend your points!";
			close;
		} else {
			// Bounty in progress - show status and menu
			mes "[Bounty Board]";
			mes "Target: ^0055FF" + .@mob_name$ + "^000000";
			mes "Progress: ^0055FF" + Bounty_Kills + " / " + Bounty_Target + "^000000";
			.@remaining = Bounty_Target - Bounty_Kills;
			mes "^FF8800" + .@remaining + " more to go!^000000";
			next;

			switch(select(" ~ Abandon Bounty: ~ ^777777Cancel^000000")) {
			case 1:
				// Abandon Bounty
				mes "[Bounty Board]";
				mes "Are you sure you want to abandon this bounty?";
				mes "All progress will be lost.";
				next;
				if (select(" ~ Yes, abandon: ~ No, keep it") == 1) {
					Bounty_Map$ = "";
					Bounty_MobID = 0;
					Bounty_Target = 0;
					Bounty_Tier = 0;
					Bounty_StartKills = 0;
					Bounty_Kills = 0;
					mes "[Bounty Board]";
					mes "Bounty abandoned.";
					mes "You can accept a new one anytime.";
				} else {
					mes "[Bounty Board]";
					mes "Good luck with your hunt!";
				}
				close;

			case 2:
				mes "[Bounty Board]";
				mes "Good luck hunting!";
				close;
			}
		}

	} else {
		// Active bounty on DIFFERENT map
		mes "[Bounty Board]";
		mes "You have an active bounty at:";
		mes "^0055FF" + getd(".MapName_" + Bounty_Map$) + "^000000";
		mes " ";
		mes "Complete or abandon it before";
		mes "accepting a new bounty.";
		close;
	}
	end;

// ==================== KILL EVENT HANDLER ====================
// Single invisible NPC to handle kill notifications (not duplicated)
-	script	BountyKillTracker	FAKE_NPC,{
	end;

OnNPCKillEvent:
	// Save kill info before any context switching
	.@killed_mob = killedrid;
	.@mob_name$ = getmonsterinfo(.@killed_mob, MOB_NAME);
	getmapxy(.@killer_map$, .@kx, .@ky, BL_PC);
	.@killer_aid = getcharid(3);
	.@killer_cid = getcharid(0);

	// Check if killer has a bounty for this monster
	.@killer_has_bounty = (Bounty_Map$ != "" && Bounty_MobID == .@killed_mob);

	// Update killer's bounty progress if they have one
	if (.@killer_has_bounty) {
		query_sql("SELECT COALESCE(kill_count, 0) FROM `char_mob_kills` WHERE `char_id` = " + .@killer_cid + " AND `mob_id` = " + Bounty_MobID, .@current_kills);
		Bounty_Kills = .@current_kills - Bounty_StartKills;
		if (Bounty_Kills < 0) Bounty_Kills = 0;

		if (Bounty_Kills >= Bounty_Target) {
			dispbottom "[Bounty] ^00FF00Complete!^000000 " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$ + " - Return to claim reward!", 0x00FF00;
		} else {
			dispbottom "[Bounty] " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$, 0x00AAFF;
		}
	}

	// ==================== PARTY KILL SHARING ====================
	.@party_id = getcharid(1);
	if (.@party_id == 0)
		end;  // Not in party, done

	// Get party member lists
	getpartymember .@party_id, 1;  // Account IDs -> $@partymemberaid[]
	getpartymember .@party_id, 2;  // Char IDs -> $@partymembercid[]

	// Loop through party members
	for (.@i = 0; .@i < $@partymembercount; .@i++) {
		// Skip the killer (they already got their kill tracked)
		if ($@partymemberaid[.@i] == .@killer_aid)
			continue;

		// Check if party member is online
		if (!isloggedin($@partymemberaid[.@i], $@partymembercid[.@i]))
			continue;

		// Check if party member is on same map
		getmapxy(.@pm_map$, .@px, .@py, BL_PC, rid2name($@partymemberaid[.@i]));
		if (.@pm_map$ != .@killer_map$)
			continue;

		// Switch context to party member
		attachrid($@partymemberaid[.@i]);

		// Check if party member has a bounty for this monster on this map
		if (Bounty_Map$ == .@killer_map$ && Bounty_MobID == .@killed_mob) {
			// Increment their kill count in database (share the kill)
			query_sql("INSERT INTO `char_mob_kills` (`char_id`, `mob_id`, `kill_count`) VALUES (" +
				getcharid(0) + ", " + .@killed_mob + ", 1) ON DUPLICATE KEY UPDATE `kill_count` = `kill_count` + 1");

			// Update their Bounty_Kills variable
			query_sql("SELECT COALESCE(kill_count, 0) FROM `char_mob_kills` WHERE `char_id` = " + getcharid(0) + " AND `mob_id` = " + .@killed_mob, .@pm_kills);
			Bounty_Kills = .@pm_kills - Bounty_StartKills;
			if (Bounty_Kills < 0) Bounty_Kills = 0;

			// Notify party member
			if (Bounty_Kills >= Bounty_Target) {
				dispbottom "[Bounty] ^00FF00Complete!^000000 " + Bounty_Kills + "/" + Bounty_Target + " " + .@mob_name$ + " (party) - Return to claim!", 0x00FF00;
			} else {

	.@total_cost = 0;

	for (.@i = 0; .@i < .@size; .@i++) {
		for (.@j = 0; .@j < getarraysize(.ShopItems); .@j++) {
			if (@bought_nameid[.@i] == .ShopItems[.@j]) {
				.@total_cost += (.ShopCosts[.@j] * @bought_quantity[.@i]);
				break;
			}
		}
	}

	mes "[Monster Hunter Guild]";

	if (#HUNT_POINTS < .@total_cost) {
		mes "You don't have enough Hunt Points!";
		mes " ";
		mes "Cost: ^FF0000" + .@total_cost + "^000000 points";
		mes "Have: ^0055FF" + #HUNT_POINTS + "^000000 points";
		close;
	}

	// Check inventory space
	for (.@i = 0; .@i < .@size; .@i++) {
		if (checkweight(@bought_nameid[.@i], @bought_quantity[.@i]) == 0) {
			mes "Your inventory is too full!";
			mes "Please make some room.";
			close;
		}
	}

	// Deduct Hunt Points
	#HUNT_POINTS -= .@total_cost;

	// Give purchased items
	for (.@i = 0; .@i < .@size; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
	}

	mes "Purchase complete!";
	mes " ";
	mes "^0055FFPoints spent: " + .@total_cost + "^000000";
	mes "^0055FFRemaining: " + #HUNT_POINTS + " Hunt Points^000000";

	specialeffect2 EF_STEAL;

	// Cleanup
	deletearray @bought_nameid[0], .@size;
	deletearray @bought_quantity[0], .@size;
	close;

OnInit:
	// Shop items and costs (parallel arrays)
	setarray .ShopItems[0],
		12028,  // Box of Thunder
		12016,  // Speed Potion
		608,    // Yggdrasil Leaf
		607,    // Yggdrasil Berry
		12075,  // Str Dish10
		12090,  // Agi Dish10
		12085,  // Vit Dish10
		12080,  // Int Dish10
		12095,  // Dex Dish10
		12100,  // Luk Dish10
		4103,   // Horong Card (Sight)
		4040;   // Creamy Card (Teleport)

	setarray .ShopCosts[0],
		25,     // Box of Thunder
		25,     // Speed Potion
		50,     // Yggdrasil Leaf
		75,     // Yggdrasil Berry
		30,     // Str Dish10
		30,     // Agi Dish10
		30,     // Vit Dish10
		30,     // Int Dish10
		30,     // Dex Dish10
		30,     // Luk Dish10
		400,    // Horong Card
		500;    // Creamy Card

	// Build visual shop (prices shown are point costs)
	// Clear items first to prevent duplicates on reload
	npcshopdelitem "hunt_point_shop", 512;
	for (.@i = 0; .@i < getarraysize(.ShopItems); .@i++) {
		npcshopdelitem "hunt_point_shop", .ShopItems[.@i];
	}
	// Add items fresh
	for (.@i = 0; .@i < getarraysize(.ShopItems); .@i++) {
		npcshopadditem "hunt_point_shop", .ShopItems[.@i], .ShopCosts[.@i];
	}
	end;
}
