//===== Custom Script ==========================================
//= Continental Guard - Solo-Friendly Version
//===== Description: ============================================
//= Allows access to Dimensional Gorge (moc_fild21/22) for solo players.
//= Quest requires: 30 Live Coals + 50 Glacial Hearts (one-time)
//= Based on Episode 12.1 Continental Guard Quest
//==============================================================

// Disable original Continental Guards on startup
-	script	CGuard_Disabler	-1,{
OnInit:
    // Disable the original party-required Continental Guards
    // Main NPC has unique name "MocConGuard", duplicates use "Continental Guard#XX"
    disablenpc "MocConGuard";
    disablenpc "Continental Guard#02";
    disablenpc "Continental Guard#03";
    disablenpc "Continental Guard#04";
    disablenpc "Continental Guard#05";
    disablenpc "Continental Guard#06";
    disablenpc "Continental Guard#07";
    disablenpc "Continental Guard#08";
    end;
}

// Solo-Friendly Continental Guard - Entry to Dimensional Gorge
// Placed at same location as original #01 (which is now disabled)
moc_fild20,354,183,3	script	Continental Guard	707,{
    // Check if player has completed the quest
    if (CGUARD_QUEST >= 3) {
        mes "[Continental Guard]";
        mes "Ah, you're a certified member of the Continental Guard.";
        mes "Where would you like to go?";
        next;
        switch(select("Enter Dimensional Gorge (moc_fild21):Enter Second Field (moc_fild22):Stay here")) {
        case 1:
            mes "[Continental Guard]";
            mes "Be careful out there. Satan Morocc's power grows stronger every day.";
            close2;
            warp "moc_fild21",38,193;
            end;
        case 2:
            mes "[Continental Guard]";
            mes "The second field is even more dangerous. Stay alert.";
            close2;
            warp "moc_fild22",38,193;
            end;
        case 3:
            mes "[Continental Guard]";
            mes "Very well. Stay safe.";
            close;
        }
    }

    // Quest not started
    if (CGUARD_QUEST == 0) {
        mes "[Continental Guard]";
        mes "Halt! No civilians are allowed beyond this point.";
        mes "The Dimensional Gorge ahead is extremely dangerous.";
        next;
        mes "[Continental Guard]";
        mes "Satan Morocc has resurrected and claimed the Sograt Desert. His power has torn holes in the fabric of space-time itself.";
        next;
        switch(select("I want to help fight Satan Morocc.:I'll be going then.")) {
        case 1:
            if (BaseLevel < 70) {
                mes "[Continental Guard]";
                mes "Your courage is admirable, but you lack experience.";
                mes "Come back when you've reached at least ^FF0000Base Level 70^000000.";
                close;
            }
            mes "[Continental Guard]";
            mes "You wish to join the Continental Guard? Very well.";
            mes "But first, you must prove your strength.";
            next;
            mes "[Continental Guard]";
            mes "Our soldiers need supplies to maintain their posts in the desert.";
            next;
            mes "[Continental Guard]";
            mes "Bring me ^0000FF30 Live Coals^000000 from the creatures in Thor Volcano.";
            mes "These everlasting flames will keep our bonfires burning.";
            set CGUARD_QUEST, 1;
            close;
        case 2:
            mes "[Continental Guard]";
            mes "A wise decision. This is no place for the unprepared.";
            close;
        }
    }

    // Stage 1: Collecting Live Coals
    if (CGUARD_QUEST == 1) {
        mes "[Continental Guard]";
        mes "Have you brought the ^0000FF30 Live Coals^000000?";
        next;
        if (countitem(7098) >= 30) {
            mes "[Continental Guard]";
            mes "Excellent work! These will serve us well.";
            next;
            mes "[Continental Guard]";
            mes "But I have one more request before I can grant you access.";
            next;
            mes "[Continental Guard]";
            mes "Our allied soldiers from the Schwarzwald Republic are struggling in this desert heat.";
            next;
            mes "[Continental Guard]";
            mes "Bring me ^0000FF50 Glacial Hearts^000000 from the Ice Dungeon.";
            mes "These frozen hearts will help cool our suffering comrades.";
            delitem 7098, 30; // Live_Coal
            set CGUARD_QUEST, 2;
            close;
        } else {
            mes "[Continental Guard]";
            mes "You still need ^0000FF" + (30 - countitem(7098)) + " more Live Coals^000000.";
            mes "The creatures in Thor Volcano carry them.";
            close;
        }
    }

    // Stage 2: Collecting Glacial Hearts
    if (CGUARD_QUEST == 2) {
        mes "[Continental Guard]";
        mes "Have you brought the ^0000FF50 Glacial Hearts^000000?";
        next;
        if (countitem(7561) >= 50) {
            mes "[Continental Guard]";
            mes "Perfect. Our soldiers will be most grateful.";
            delitem 7561, 50; // Ice_Heart (Glacial Heart)
            next;
            mes "[Continental Guard]";
            mes "You have proven yourself worthy.";
            mes "Welcome to the Continental Guard!";
            next;
            mes "[Continental Guard]";
            mes "Take this certificate. It grants you access to the Dimensional Gorge whenever you need.";
            getitem 7826, 1; // Continental_Guard_Paper
            set CGUARD_QUEST, 3;
            next;
            mes "[Continental Guard]";
            mes "The fate of the continent rests on our shoulders.";
            mes "Fight well, and may fortune favor you.";
            close;
        } else {
            mes "[Continental Guard]";
            mes "You still need ^0000FF" + (50 - countitem(7561)) + " more Glacial Hearts^000000.";
            mes "The Ice Titans in Ice Dungeon drop them.";
            close;
        }
    }
    end;
}
