//===== Additional Comments: =================================
//= 1.0 Initial script.
//= 1.1 Aligned coordinates with @go.
//= 1.2 Added 5 buffs: Increase Agi, Blessing, Impositio Manus,
//=     Magnificat, and Angelus (4 minute duration, level 10)
//= 1.3 Added equipment identification feature with configurable
//=     buff duration (10 minutes default)
//============================================================

-	script	Healer	-1,{

	.@Price = 0;	// Zeny required for heal
	.@Buffs = 1;	// Also buff players? (1: yes / 0: no)
	.@Delay = 1;	// Heal delay, in seconds
	.@BuffDuration = 600000;	// Buff duration in milliseconds (600000 = 10 minutes, 240000 = 4 minutes standard)
	.@IdentifyItems = 1;	// Identify all unidentified equipment? (1: yes / 0: no)

	if (@HD > gettimetick(2))
		end;
	if (.@Price) {
		message strcharinfo(0), "Healing costs " + callfunc("F_InsertComma",.@Price) + " Zeny.";
		if (Zeny < .@Price)
			end;
		if (select("^0055FFHeal^000000:^777777Cancel^000000") == 2)
			end;
		Zeny -= .@Price;
	}
	specialeffect2 EF_HEAL2;
	percentheal 100,100;
	if (.@Buffs) {
		// Increase Agi
		specialeffect2 EF_INCAGILITY;
		sc_start SC_INCREASEAGI,.@BuffDuration,10;
		// Blessing
		specialeffect2 EF_BLESSING;
		sc_start SC_BLESSING,.@BuffDuration,10;
		// Impositio Manus
		specialeffect2 EF_IMPOSITIO;
		sc_start SC_IMPOSITIO,.@BuffDuration,10;
		// Magnificat
		specialeffect2 EF_MAGNIFICAT;
		sc_start SC_MAGNIFICAT,.@BuffDuration,10;
		// Angelus
		specialeffect2 EF_ANGELUS;
		sc_start SC_ANGELUS,.@BuffDuration,10;
	}
	if (.@IdentifyItems) {
		// Identify all unidentified equipment in inventory
		.@identified = 0;
		getinventorylist;
		.@inventorysize = getarraysize(@inventorylist_id);
		for (.@i = 0; .@i < .@inventorysize; .@i++) {
			// Check if item is not identified
			if (@inventorylist_identify[.@i] == 0) {
				// Get item type (4 = armor, 5 = weapon, 8 = pet armor)
				.@type = getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE);
				if (.@type == IT_WEAPON || .@type == IT_ARMOR || .@type == IT_PETARMOR) {
					// Remove unidentified item and add identified version
					delitem2 @inventorylist_id[.@i], 1, 0, @inventorylist_refine[.@i],
						@inventorylist_attribute[.@i], @inventorylist_card1[.@i],
						@inventorylist_card2[.@i], @inventorylist_card3[.@i],
						@inventorylist_card4[.@i];
					getitem2 @inventorylist_id[.@i], 1, 1, @inventorylist_refine[.@i],
						@inventorylist_attribute[.@i], @inventorylist_card1[.@i],
						@inventorylist_card2[.@i], @inventorylist_card3[.@i],
						@inventorylist_card4[.@i];
					.@identified++;
				}
			}
		}
		if (.@identified > 0) {
			specialeffect2 EF_MAGNIFIERITEM;
			message strcharinfo(0), "Identified " + .@identified + " item(s).";
		}
	}
	if (.@Delay)
		@HD = gettimetick(2) + .@Delay;
	end;
}
