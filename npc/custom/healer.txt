//= Healer (Two-Tier Pricing)
//===== By: ==================================================
//= Euphy (original)
//= Modified by Claude Code
//===== Current Version: =====================================
//= 3.0
//===== Compatible With: =====================================
//= rAthena Project
//===== Description: =========================================
//= Healer with level-based pricing:
//= - Level 60 and below: FREE heal + buffs
//= - Level 61+: Heal costs 3,000z, Buffs cost 10,000z
//===== Additional Comments: =================================
//= 3.0 Buffs include heal, grace period for buff purchasers
//============================================================

-	script	Healer	-1,{

	.@IdentifyItems = 1;	// Identify all unidentified equipment? (1: yes / 0: no)

	// Pricing configuration
	.@HealPrice = 3000;
	.@BuffsPrice = 10000;		// Includes heal + identify
	.@FreeLevel = 60;
	.@BuffDuration = 600000;	// 10 minutes
	.@GracePeriod = 180;		// 3 minutes grace after paying for buffs
	.@MonkGrace = 10;			// 10 seconds for Monks (prevent Asura abuse)

	// ============================================================
	// Check if player qualifies for free service
	// ============================================================
	if (.@FreeLevel > 0 && BaseLevel <= .@FreeLevel) {
		// Low-level players get everything free
		message strcharinfo(0), "Free service for adventurers level " + .@FreeLevel + " and below!";
		callsub S_Heal, .@IdentifyItems;
		callsub S_Buffs, .@BuffDuration;
		end;
	}

	// ============================================================
	// Check grace period (for buff purchasers)
	// ============================================================
	.@now = gettimetick(2);
	if (healer_paid_until > .@now) {
		// Check if they've used up their 10 free uses
		if (healer_grace_uses >= 10) {
			mes "[Nurse]";
			mes "Goodness, you've been here";
			mes "quite a few times now!";
			mes " ";
			mes "I hate to ask, but the supplies";
			mes "do cost money to restock...";
			next;
			// Fall through to payment menu
		} else {
			healer_grace_uses = healer_grace_uses + 1;
			mes "[Nurse]";
			mes "Back again already?";
			mes "Let me take care of that for you.";
			close2;
			callsub S_Heal, .@IdentifyItems;
			callsub S_Buffs, .@BuffDuration;
			end;
		}
	}

	// ============================================================
	// Level 61+ players - show menu
	// ============================================================
	mes "[Nurse]";
	mes "Oh my, you look exhausted!";
	mes "Come, let me take a look at you.";
	mes " ";
	mes "What can I do for you today?";
	next;

	switch(select(
		"Heal + Identify (" + callfunc("F_InsertComma", .@HealPrice) + "z)",
		"Heal + Identify + Buffs (" + callfunc("F_InsertComma", .@BuffsPrice) + "z)",
		"Cancel"
	)) {
		case 1:
			// Heal only
			if (Zeny < .@HealPrice) {
				mes "[Nurse]";
				mes "Oh dear... I'm afraid the";
				mes "medical supplies aren't free.";
				mes " ";
				mes "You need " + callfunc("F_InsertComma", .@HealPrice) + " Zeny.";
				close;
			}
			Zeny -= .@HealPrice;
			close2;
			callsub S_Heal, .@IdentifyItems;
			end;

		case 2:
			// Full service (heal + identify + buffs)
			if (Zeny < .@BuffsPrice) {
				mes "[Nurse]";
				mes "Oh dear... I'm afraid the";
				mes "medical supplies aren't free.";
				mes " ";
				mes "You need " + callfunc("F_InsertComma", .@BuffsPrice) + " Zeny.";
				close;
			}
			Zeny -= .@BuffsPrice;

			// Set grace period based on class
			// Monks and Champions get shorter grace to prevent Asura Strike abuse
			if ((eaclass()&EAJ_UPPERMASK) == EAJ_MONK) {
				.@grace = .@MonkGrace;
			} else {
				.@grace = .@GracePeriod;
			}

			healer_paid_until = .@now + .@grace;
			healer_grace_uses = 0;  // Reset usage counter on payment

			mes "[Nurse]";
			mes "There you go, all patched up!";
			mes " ";
			mes "These tonics should keep you";
			mes "going for a while. Be careful!";
			close2;
			callsub S_Heal, .@IdentifyItems;
			callsub S_Buffs, .@BuffDuration;
			end;

		case 3:
			// Cancel
			mes "[Nurse]";
			mes "Stay safe out there!";
			mes "Come back if you need anything.";
			close;
	}
	end;

// ============================================================
// Subroutine: Heal + Identify
// ============================================================
S_Heal:
	specialeffect2 EF_HEAL2;
	percentheal 100,100;

	if (getarg(0)) {
		.@identified = identifyall();
		if (.@identified > 0) {
			specialeffect2 EF_MAGNIFIERITEM;
			message strcharinfo(0), "Identified " + .@identified + " item(s).";
		}
	}
	return;

// ============================================================
// Subroutine: Apply all 8 buffs
// ============================================================
S_Buffs:
	.@duration = getarg(0);

	// 1. Increase Agi (+12 AGI, +25% move speed)
	specialeffect2 EF_INCAGILITY;
	sc_start SC_INCREASEAGI, .@duration, 10;

	// 2. Blessing (+10 STR/INT/DEX)
	specialeffect2 EF_BLESSING;
	sc_start SC_BLESSING, .@duration, 10;

	// 3. Impositio Manus (+25 ATK)
	specialeffect2 EF_IMPOSITIO;
	sc_start SC_IMPOSITIO, .@duration, 10;

	// 4. Magnificat (2x SP regen)
	specialeffect2 EF_MAGNIFICAT;
	sc_start SC_MAGNIFICAT, .@duration, 10;

	// 5. Angelus (+50% DEF)
	specialeffect2 EF_ANGELUS;
	sc_start SC_ANGELUS, .@duration, 10;

	// 6. Kyrie Eleison (blocks hits/damage shield)
	// Max duration is 120 seconds (skill cap), not affected by .@duration
	specialeffect2 EF_KYRIE;
	sc_start SC_KYRIE, 120000, 10;

	// 7. Gloria (+30 LUK)
	specialeffect2 EF_GLORIA;
	sc_start SC_GLORIA, .@duration, 10;

	// 8. Assumptio (halves damage in pre-renewal)
	specialeffect2 EF_ASSUMPTIO;
	sc_start SC_ASSUMPTIO, .@duration, 5;

	// Re-heal after buffs so Blessing's INT bonus affects max SP
	percentheal 100,100;
	return;
}
