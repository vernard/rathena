//===== Description: =========================================
//= Enhanced healer script with 5 buffs and item identification.
//===== Additional Comments: =================================
//= 1.0 Initial script.
//= 1.1 Aligned coordinates with @go.
//= 1.2 Added 5 buffs: Increase Agi, Blessing, Impositio Manus,
//=     Magnificat, and Angelus (4 minute duration, level 10)
//= 1.3 Added equipment identification feature with configurable
//=     buff duration (10 minutes default)
//= 1.4 Fixed heal order: buffs applied before heal so Blessing's
//=     INT bonus is reflected in max SP before percentheal
//= 1.5 Two-tier pricing: 1k for heal+identify, 5k for heal+buffs+identify
//= 1.6 Configurable pricing via env vars, free for low-level players
//============================================================

-	script	Healer	-1,{

	.@IdentifyItems = 1;	// Identify all unidentified equipment? (1: yes / 0: no)

	// Configurable settings (values baked in at build time from environment variables)
	.@BasicPrice = 1000;
	.@BuffsPrice = 5000;
	.@FreeLevel = 30;
	.@BuffDuration = 600000;

	// Free full service for low-level adventurers (skip menu)
	if (.@FreeLevel > 0 && BaseLevel <= .@FreeLevel) {
		.@Price = 0;
		.@Buffs = 1;
		message strcharinfo(0), "Free service for adventurers level " + .@FreeLevel + " and below!";
	} else {
		// Menu selection for higher level players
		switch(select("Heal + Identify (" + .@BasicPrice + "z):Heal + Buffs + Identify (" + .@BuffsPrice + "z):Cancel")) {
			case 1:
				.@Price = .@BasicPrice;
				.@Buffs = 0;
				break;
			case 2:
				.@Price = .@BuffsPrice;
				.@Buffs = 1;
				break;
			case 3:
				end;
		}
	}

	if (Zeny < .@Price) {
		message strcharinfo(0), "You need " + callfunc("F_InsertComma",.@Price) + " Zeny.";
		end;
	}
	Zeny -= .@Price;
	// Apply buffs first so stat bonuses affect max HP/SP before heal
	if (.@Buffs) {
		// Increase Agi
		specialeffect2 EF_INCAGILITY;
		sc_start SC_INCREASEAGI,.@BuffDuration,10;
		// Blessing
		specialeffect2 EF_BLESSING;
		sc_start SC_BLESSING,.@BuffDuration,10;
		// Impositio Manus
		specialeffect2 EF_IMPOSITIO;
		sc_start SC_IMPOSITIO,.@BuffDuration,10;
		// Magnificat
		specialeffect2 EF_MAGNIFICAT;
		sc_start SC_MAGNIFICAT,.@BuffDuration,10;
		// Angelus
		specialeffect2 EF_ANGELUS;
		sc_start SC_ANGELUS,.@BuffDuration,10;
	}
	// Heal after buffs so Blessing's INT bonus is reflected in max SP
	specialeffect2 EF_HEAL2;
	percentheal 100,100;
	if (.@IdentifyItems) {
		// Identify all unidentified items in inventory using built-in command
		// identifyall() returns count of items identified and identifies them atomically
		.@identified = identifyall();
		if (.@identified > 0) {
			specialeffect2 EF_MAGNIFIERITEM;
			message strcharinfo(0), "Identified " + .@identified + " item(s).";
		}
	}
	end;
}
