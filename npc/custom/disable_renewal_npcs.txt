//===== rAthena Script =======================================
//= Disable Renewal-Only NPCs
//===== Description: =========================================
//= Disables Healer and Warpra NPCs on Renewal-only maps
//= that are inaccessible in Pre-Renewal mode.
//= Maps: Brasilis, Dewata, Eclage, El Dicastes, Manuk, Mora, Splendide
//============================================================

-	script	DisableRenewalNPCs	-1,{
OnInit:
	// Disable Healers on Renewal-only maps
	// (These maps have no travel NPCs in Pre-Renewal)
	disablenpc "Healer#man";    // Manuk
	disablenpc "Healer#spl";    // Splendide
	disablenpc "Healer#bra";    // Brasilis
	disablenpc "Healer#dew";    // Dewata
	disablenpc "Healer#dic";    // El Dicastes
	disablenpc "Healer#mora";   // Mora
	// Note: Eclage has no Healer

	// Disable Warpra on Renewal-only maps
	disablenpc "Warpra#36";     // Brasilis
	disablenpc "Warpra#37";     // El Dicastes
	disablenpc "Warpra#38";     // Manuk
	disablenpc "Warpra#39";     // Mora
	disablenpc "Warpra#42";     // Splendide
	disablenpc "Warpra#44";     // Dewata
	disablenpc "Warpra#45";     // Eclage
	end;
}
