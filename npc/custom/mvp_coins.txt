//===== rAthena Script =======================================
//= MVP Coin System
//===== Description: =========================================
//= Tradeable MVP coins dropped when killing MVP monsters.
//= Coins are tiered by MVP EXP reward (better difficulty scaling):
//=   - Silver MVP Coin (30010): < 25k MvpExp - 1 point
//=   - Gold MVP Coin (30011): 25k-500k MvpExp - 3 points
//=   - Mithril MVP Coin (30012): 500k+ MvpExp - 10 points
//=
//= When in a party, a random member on the same map receives
//= the coin. Exchange coins at NPC for exclusive items.
//===== Changelog: ==========================================
//= 1.0 Initial implementation (level-based)
//= 1.1 Switched to EXP-based tiers, added Platinum tier
//= 1.2 Simplified to 3 tiers with matching client sprites
//============================================================

// Invisible NPC for MVP kill detection
-	script	MVPCoinSystem	FAKE_NPC,{
	end;

OnNPCKillEvent:
	// Check if killed mob is an MVP (has MVP experience)
	.@mvp_exp = getmonsterinfo(killedrid, MOB_MVPEXP);
	if (.@mvp_exp <= 0) end;

	.@mob_name$ = getmonsterinfo(killedrid, MOB_NAME);

	// Determine coin tier by MVP EXP (better difficulty scaling than level)
	// Mithril: 500k+ MvpExp (Fallen Bishop, Kiel, Bio Labs, Beelzebub)
	// Gold: 25k-500k MvpExp (Doppel, Dark Lord, Amon Ra, Baphomet, Dracula, LoD, Atroce, Gloom)
	// Silver: <25k MvpExp (GTB, Eddga, Drake, Maya, Moonlight)
	if (.@mvp_exp >= 500000) {
		.@coin_id = 30012;  // Mithril MVP Coin
		.@coin_name$ = "^87CEEBMithril MVP Coin^000000";
		.@coin_color = 0x87CEEB;
	} else if (.@mvp_exp >= 25000) {
		.@coin_id = 30011;  // Gold MVP Coin
		.@coin_name$ = "^FFD700Gold MVP Coin^000000";
		.@coin_color = 0xFFD700;
	} else {
		.@coin_id = 30010;  // Silver MVP Coin
		.@coin_name$ = "^C0C0C0Silver MVP Coin^000000";
		.@coin_color = 0xC0C0C0;
	}

	// Get killer's current map position
	getmapxy(.@map$, .@x, .@y, BL_PC);

	// Check if killer is in a party
	.@party_id = getcharid(1);
	if (.@party_id) {
		// Get party member account and character IDs
		getpartymember .@party_id, 1;  // Account IDs -> $@partymemberaid[]
		getpartymember .@party_id, 2;  // Character IDs -> $@partymembercid[]

		// Build list of eligible members (online + same map + alive)
		.@eligible_count = 0;
		for (.@i = 0; .@i < $@partymembercount; .@i++) {
			// Check if party member is online
			if (!isloggedin($@partymemberaid[.@i], $@partymembercid[.@i]))
				continue;

			// Check if party member is on same map
			getmapxy(.@m$, .@px, .@py, BL_PC, rid2name($@partymemberaid[.@i]));
			if (.@m$ != .@map$)
				continue;

			// Check if party member is alive
			if (readparam(HP, $@partymembercid[.@i]) <= 0)
				continue;

			// Add to eligible list
			.@eligible_aid[.@eligible_count] = $@partymemberaid[.@i];
			.@eligible_cid[.@eligible_count] = $@partymembercid[.@i];
			.@eligible_count++;
		}

		// Give to random eligible member
		if (.@eligible_count > 0) {
			.@winner_idx = rand(.@eligible_count);
			.@winner_aid = .@eligible_aid[.@winner_idx];
			.@winner_cid = .@eligible_cid[.@winner_idx];

			// Give coin to winner
			getitem .@coin_id, 1, .@winner_aid;

			// Get winner's name for announcement
			.@winner_name$ = rid2name(.@winner_aid);

			// Announce to all party members on the map
			for (.@i = 0; .@i < .@eligible_count; .@i++) {
				if (.@eligible_aid[.@i] == .@winner_aid) {
					// Message to winner
					dispbottom "[MVP Coin] You received a " + .@coin_name$ + " from " + .@mob_name$ + "!", .@coin_color, .@winner_aid;
				} else {
					// Message to other party members
					dispbottom "[MVP Coin] " + .@winner_name$ + " received a " + .@coin_name$ + " from " + .@mob_name$ + "!", 0x888888, .@eligible_aid[.@i];
				}
			}
		}
	} else {
		// Solo player - give directly to killer
		getitem .@coin_id, 1;
		dispbottom "[MVP Coin] You received a " + .@coin_name$ + " from " + .@mob_name$ + "!", .@coin_color;
	}
	end;
}

// ============================================================
// MVP Coin Exchange NPC
// ============================================================

// Master NPC in Prontera
prontera,164,172,4	script	MVP Coin Exchange	4_M_MASKMAN,{
	mes "[MVP Coin Exchange]";
	mes "Welcome! I can exchange your";
	mes "^0055FFMVP Coins^000000 for valuable items.";
	mes " ";
	mes "MVP Coins drop from boss monsters";
	mes "when you defeat them in battle.";
	next;

	// Calculate current balance
	.@silver = countitem(30010);
	.@gold = countitem(30011);
	.@mithril = countitem(30012);
	.@total_points = .@silver + (.@gold * 3) + (.@mithril * 10);

	mes "[MVP Coin Exchange]";
	mes "^0055FFYour MVP Coins:^000000";
	mes "^C0C0C0Silver^000000: " + .@silver + " (1 pt each)";
	mes "^FFD700Gold^000000: " + .@gold + " (3 pts each)";
	mes "^87CEEBMithril^000000: " + .@mithril + " (10 pts each)";
	mes " ";
	mes "^0055FFTotal Value: ^FF0000" + .@total_points + " points^000000";
	next;

	switch(select("Browse Shop:Coin Information:Cancel")) {
	case 1:
		callshop "mvp_coin_shop", 1;
		npcshopattach "mvp_coin_shop";
		end;

	case 2:
		mes "[MVP Coin Exchange]";
		mes "^0055FFMVP Coin Tiers:^000000";
		mes "(based on MVP EXP reward)";
		mes " ";
		mes "^C0C0C0Silver^000000: Easy MVPs (1 pt)";
		mes "GTB, Eddga, Drake, Maya...";
		next;
		mes "[MVP Coin Exchange]";
		mes "^FFD700Gold^000000: Medium/Hard MVPs (3 pts)";
		mes "Doppelganger, Dark Lord,";
		mes "Amon Ra, Baphomet, Dracula,";
		mes "Lord of Death, Atroce, Gloom...";
		next;
		mes "[MVP Coin Exchange]";
		mes "^87CEEBMithril^000000: Extreme MVPs (10 pts)";
		mes "Fallen Bishop, Kiel D-01,";
		mes "Bio Lab MVPs, Beelzebub...";
		next;
		mes "[MVP Coin Exchange]";
		mes "^0055FFParty Distribution:^000000";
		mes "When hunting in a party, coins";
		mes "are given to a random member";
		mes "on the same map!";
		close;

	case 3:
		mes "[MVP Coin Exchange]";
		mes "Come back anytime!";
		close;
	}
	end;

OnBuyItem:
	// Calculate total cost of purchase
	.@size = getarraysize(@bought_nameid);
	.@total_cost = 0;

	for (.@i = 0; .@i < .@size; .@i++) {
		// Find this item's price in our shop arrays
		for (.@j = 0; .@j < getarraysize(.ShopItems); .@j++) {
			if (@bought_nameid[.@i] == .ShopItems[.@j]) {
				.@total_cost += (.ShopPrices[.@j] * @bought_quantity[.@i]);
				break;
			}
		}
	}

	// Calculate player's total points from coins
	.@silver = countitem(30010);
	.@gold = countitem(30011);
	.@mithril = countitem(30012);
	.@player_points = .@silver + (.@gold * 3) + (.@mithril * 10);

	mes "[MVP Coin Exchange]";

	if (.@player_points < .@total_cost) {
		mes "You don't have enough coins!";
		mes " ";
		mes "Cost: ^FF0000" + .@total_cost + "^000000 points";
		mes "Have: ^0055FF" + .@player_points + "^000000 points";
		close;
	}

	// Check inventory space
	for (.@i = 0; .@i < .@size; .@i++) {
		if (checkweight(@bought_nameid[.@i], @bought_quantity[.@i]) == 0) {
			mes "Your inventory is too full!";
			mes "Please make some room.";
			close;
		}
	}

	// Consume coins (use highest value first: Mithril -> Gold -> Silver)
	.@remaining = .@total_cost;

	// Spend mithril coins first (10 pts each)
	.@mithril_spend = min(.@mithril, .@remaining / 10);
	if (.@mithril_spend > 0) {
		delitem 30012, .@mithril_spend;
		.@remaining -= .@mithril_spend * 10;
	}

	// Spend gold coins next (3 pts each)
	.@gold_spend = min(countitem(30011), .@remaining / 3);
	if (.@gold_spend > 0) {
		delitem 30011, .@gold_spend;
		.@remaining -= .@gold_spend * 3;
	}

	// Spend silver coins last (1 pt each)
	if (.@remaining > 0) {
		delitem 30010, .@remaining;
	}

	// Give purchased items
	for (.@i = 0; .@i < .@size; .@i++) {
		getitem @bought_nameid[.@i], @bought_quantity[.@i];
	}

	// Show confirmation
	mes "Purchase complete!";
	mes " ";
	mes "^0055FFCoins spent: " + .@total_cost + " points^000000";

	// Show new balance
	.@new_silver = countitem(30010);
	.@new_gold = countitem(30011);
	.@new_mithril = countitem(30012);
	.@new_total = .@new_silver + (.@new_gold * 3) + (.@new_mithril * 10);
	mes "^0055FFRemaining: " + .@new_total + " points^000000";

	specialeffect2 EF_STEAL;

	// Cleanup
	deletearray @bought_nameid[0], .@size;
	deletearray @bought_quantity[0], .@size;
	close;

OnInit:
	// Define shop items and their point costs
	// Format: Item ID, Point Cost
	setarray .ShopItems[0],
		678,     // Poison Bottle
		12016,   // Speed Potion
		14003,   // Elite Siege Supply Box
		7621;    // Token of Siegfried

	setarray .ShopPrices[0],
		1,       // Poison Bottle = 1 pt
		2,       // Speed Potion = 2 pts
		5,       // Elite Siege Supply = 5 pts
		20;      // Token of Siegfried = 20 pts

	// Build the visual shop (prices shown are point costs)
	// Clear all items first to prevent duplicates on reload
	npcshopdelitem "mvp_coin_shop", 512;
	for (.@i = 0; .@i < getarraysize(.ShopItems); .@i++) {
		npcshopdelitem "mvp_coin_shop", .ShopItems[.@i];
	}
	// Now add items fresh
	for (.@i = 0; .@i < getarraysize(.ShopItems); .@i++) {
		npcshopadditem "mvp_coin_shop", .ShopItems[.@i], .ShopPrices[.@i];
	}
	end;
}

// 'no' disables Discount skill affecting displayed prices
-	shop	mvp_coin_shop	-1,no,512:-1
