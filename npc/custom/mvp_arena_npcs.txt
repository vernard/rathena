// ============================================================
// Override the MVP Arena Guide to remove the free heal option
// Players should use the paid Healer NPC instead
// ============================================================
quiz_00,49,31,4	script	MVP Arena Guide	778,{
	mes "[ ^0065DFMVP Arena Guide^000000 ]";
	mes "Welcome and behold this sacred place. Here you will find out if you truly have what it takes to call yourself a warrior.";
	next;
	switch(select("Information","Return to Prontera","Cancel")) {
	case 1:
		mes "[ ^0065DFMVP Arena Guide^000000 ]";
		mes "There are four Keepers, and each can spawn four different MVPs.";
		mes "There are eight rooms per Keeper, and sixteen MVPs in total.";
		close;
	case 2:
		warp "prontera",156,179;
		close;
	case 3:
		close;
	}
}

// ============================================================
// Override the Protector function from mvp_arena.txt
// Navigation only - healing removed (use skills in lobby instead)
// ============================================================
function	script	Protector	{
	.@nextMap$ = getarg(0);

	if (.@nextMap$ != "") {
		switch(select("Harder Monsters:Exit to Lobby")) {
		case 1:
			warp .@nextMap$, 102, 102;
			end;
		case 2:
			warp "quiz_00", 50, 24;
			end;
		}
	} else {
		// Hardest room - only exit option
		switch(select("Exit to Lobby")) {
		case 1:
			warp "quiz_00", 50, 24;
			end;
		}
	}
}

// ============================================================
// MVP Arena Lobby Healer - duplicate of town Healer
// Same behavior: free for lv60-, paid for lv61+, grace period
// ============================================================
quiz_00,57,21,4	duplicate(Healer)	Healer#mvp_arena	569

// MVP Arena Enchanter - duplicate of Element Endow NPC
// Positioned to the right of the Healer - useful for matching element to MVP weakness
quiz_00,61,21,4	duplicate(Enchanter)	Enchanter#mvp_arena	4_M_SAGE_A

// MVP Arena Tool Dealer - duplicate of Prontera Premium Tool Dealer
// Same items: potions, converters, resist potions, Ygg Leaf, etc.
quiz_00,53,17,4	duplicate(Tool Dealer#prt_center)	Tool Dealer#mvp	900

// MVP Arena Kafra - provides storage and save point
// Positioned near the Healer
// Save point is set to quiz_00 so players respawn in the arena lobby after death
quiz_00,53,21,4	script	Kafra Employee#mvp	117,{
	cutin "kafra_01",2;
	callfunc "F_KafSet";
	mes "[Kafra Employee]";
	mes "Welcome to the";
	mes "MVP Arena, adventurer~";
	mes "The Kafra Services are";
	mes "here to support your";
	mes "MVP hunting! How may";
	mes "I help you today?";
	callfunc "F_Kafra",5,0,0,40,800;
	savepoint "quiz_00",50,24,1,1;
	callfunc "F_KafEnd",0,1,"in the MVP Arena lobby";
}
