//===== Custom Script =======================================
//= Card Recycler - Trade Cards for Hunt Points
//===== Description: =========================================
//= Allows players to trade unwanted monster cards for Hunt Points.
//= Excludes MVP and Mini-boss cards to protect valuable items.
//= Integrates with the Bounty Board Hunt Points economy.
//============================================================

prontera,168,180,5	script	Card Recycler	91,{
	// Configuration
	.@points_per_card = 15;

	mes "[Card Recycler]";
	mes "Hello there, hunter!";
	mes " ";
	mes "I collect unwanted ^0055FFmonster cards^000000.";
	mes "For each card, I'll give you";
	mes "^00CC00" + .@points_per_card + " Hunt Point" + ((.@points_per_card > 1) ? "s" : "") + "^000000.";
	next;

	mes "[Card Recycler]";
	mes "^777777Note: I don't accept MVP or";
	mes "Mini-boss cards - those are";
	mes "too valuable to recycle!^000000";
	next;

	// Scan inventory for tradeable cards
	getinventorylist;
	.@card_count = 0;

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		.@item_id = @inventorylist_id[.@i];

		// Check if item is a card (type 6 = IT_CARD)
		if (getiteminfo(.@item_id, ITEMINFO_TYPE) != IT_CARD)
			continue;

		// Check if card is blacklisted (MVP or Mini-boss)
		if (callfunc("F_IsBlacklistedCard", .@item_id))
			continue;

		// Valid card found - add to list
		.@card_ids[.@card_count] = .@item_id;
		.@card_amounts[.@card_count] = @inventorylist_amount[.@i];
		.@card_count++;
	}

	if (.@card_count == 0) {
		mes "[Card Recycler]";
		mes "You don't have any cards I can accept.";
		mes " ";
		mes "Come back when you have some";
		mes "regular monster cards to trade!";
		close;
	}

	// Build menu of tradeable cards
	.@menu$ = "";
	for (.@i = 0; .@i < .@card_count; .@i++) {
		.@menu$ = .@menu$ + getitemname(.@card_ids[.@i]) + " x" + .@card_amounts[.@i] + ":";
	}
	.@menu$ = .@menu$ + "^777777Cancel^000000";

	mes "[Card Recycler]";
	mes "Which card would you like to trade?";
	mes " ";
	mes "Your Hunt Points: ^00CC00" + #HUNT_POINTS + "^000000";
	next;

	.@choice = select(.@menu$) - 1;

	if (.@choice >= .@card_count) {
		mes "[Card Recycler]";
		mes "No problem! Come back anytime.";
		close;
	}

	.@selected_id = .@card_ids[.@choice];
	.@selected_amount = countitem(.@selected_id);

	if (.@selected_amount <= 0) {
		mes "[Card Recycler]";
		mes "Hmm, you don't seem to have";
		mes "that card anymore.";
		close;
	}

	mes "[Card Recycler]";
	mes "You have ^0055FF" + .@selected_amount + " " + getitemname(.@selected_id) + "^000000.";
	mes " ";
	mes "How many would you like to trade?";
	mes "^777777(Each = " + .@points_per_card + " Hunt Point)^000000";
	next;

	input .@trade_amount;

	if (.@trade_amount <= 0) {
		mes "[Card Recycler]";
		mes "Changed your mind? No worries!";
		close;
	}

	if (.@trade_amount > .@selected_amount) {
		mes "[Card Recycler]";
		mes "You don't have that many!";
		mes "You only have " + .@selected_amount + ".";
		close;
	}

	.@total_points = .@trade_amount * .@points_per_card;

	mes "[Card Recycler]";
	mes "Trade ^0055FF" + .@trade_amount + " " + getitemname(.@selected_id) + "^000000";
	mes "for ^00CC00" + .@total_points + " Hunt Point" + ((.@total_points > 1) ? "s" : "") + "^000000?";
	next;

	if (select("Yes, trade them:No, cancel") == 2) {
		mes "[Card Recycler]";
		mes "No problem. Come back anytime!";
		close;
	}

	// Verify cards still exist
	if (countitem(.@selected_id) < .@trade_amount) {
		mes "[Card Recycler]";
		mes "Wait, your cards are gone!";
		mes "Did you move them?";
		close;
	}

	// Execute trade
	delitem .@selected_id, .@trade_amount;
	#HUNT_POINTS += .@total_points;

	specialeffect2 EF_STEAL;

	mes "[Card Recycler]";
	mes "Done! Thank you for the cards.";
	mes " ";
	mes "^00CC00+" + .@total_points + " Hunt Points^000000";
	mes " ";
	mes "Your total: ^00CC00" + #HUNT_POINTS + " Hunt Points^000000";
	close;
}
