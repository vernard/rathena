//===== rAthena Script =======================================
//= Event Trigger Library
//= Shared functions for conditional trigger management
//===== Description: =========================================
//= Provides reusable UI and logic for event NPCs to manage
//= their conditional trigger groups and conditions.
//=
//= Usage in event NPCs:
//=   callfunc "F_TriggerMenu", <event_code$>;
//============================================================

// ============================================================
// Main Trigger Management Menu
// Args: event_code$ - The event type code (e.g., "invasion")
// ============================================================
function	script	F_TriggerMenu	{
	.@event_code$ = getarg(0);

	// Get event type info
	query_sql "SELECT id, name FROM event_types WHERE code = '"+escape_sql(.@event_code$)+"'", .@event_type_id, .@event_name$;
	if (!.@event_type_id) {
		mes "^FF0000Error: Unknown event type '"+.@event_code$+"'^000000";
		return;
	}

	// Count trigger groups
	query_sql "SELECT COUNT(*) FROM event_trigger_groups WHERE event_type_id = "+.@event_type_id, .@group_count;

F_TriggerMenu_Loop:
	// Re-count groups (may have changed)
	query_sql "SELECT COUNT(*) FROM event_trigger_groups WHERE event_type_id = "+.@event_type_id, .@group_count;

	clear;
	mes "[Trigger Settings]";
	mes "^0055FF"+.@event_name$+"^000000";
	mes " ";
	mes "Manage conditional triggers for this event.";
	mes "^777777("+.@group_count+" trigger group"+((.@group_count != 1) ? "s" : "")+" configured)^000000";
	next;

	switch(select(
		"View/Edit Trigger Groups",
		"Create New Trigger Group",
		"Test Triggers Now",
		"View Recent Triggers",
		"Back"
	)) {
	case 1:
		callfunc "F_TriggerGroups", .@event_type_id, .@event_name$, .@event_code$;
		goto F_TriggerMenu_Loop;
	case 2:
		callfunc "F_CreateTriggerGroup", .@event_type_id, .@event_code$;
		goto F_TriggerMenu_Loop;
	case 3:
		callfunc "F_TestTriggers", .@event_type_id, .@event_code$;
		goto F_TriggerMenu_Loop;
	case 4:
		callfunc "F_RecentTriggers", .@event_type_id, .@event_name$;
		goto F_TriggerMenu_Loop;
	case 5:
		return;
	}
	return;
}

// ============================================================
// View/Manage Trigger Groups
// ============================================================
function	script	F_TriggerGroups	{
	.@event_type_id = getarg(0);
	.@event_name$ = getarg(1);
	.@event_code$ = getarg(2);

F_TriggerGroups_Loop:
	clear;
	mes "[Trigger Groups]";
	mes "^0055FF"+.@event_name$+"^000000";
	mes " ";

	// Get all groups
	query_sql "SELECT id, name, enabled, check_interval_minutes, priority FROM event_trigger_groups WHERE event_type_id = "+.@event_type_id+" ORDER BY priority DESC", .@ids, .@names$, .@enabled, .@intervals, .@priority;

	.@count = getarraysize(.@ids);
	if (.@count == 0) {
		mes "No trigger groups configured.";
		mes " ";
		mes "^777777Trigger groups define conditions";
		mes "that can start this event.^000000";
		next;
		if (select("Create New Group:Back") == 1) {
			callfunc "F_CreateTriggerGroup", .@event_type_id, .@event_code$;
			goto F_TriggerGroups_Loop;
		}
		return;
	}

	mes "Select a group to view/edit:";
	mes " ";

	// Build menu
	.@menu$ = "";
	for (.@i = 0; .@i < .@count; .@i++) {
		.@status$ = (.@enabled[.@i]) ? "^00AA00ON^000000" : "^FF0000OFF^000000";
		.@menu$ = .@menu$ + .@status$ + " " + .@names$[.@i] + ":";
	}
	.@menu$ = .@menu$ + "Back";

	next;
	.@sel = select(.@menu$) - 1;

	if (.@sel >= .@count) return;  // Back selected

	// Show group details
	callfunc "F_GroupDetail", .@ids[.@sel], .@names$[.@sel], .@enabled[.@sel], .@intervals[.@sel], .@event_code$;
	goto F_TriggerGroups_Loop;
}

// ============================================================
// Group Detail View
// ============================================================
function	script	F_GroupDetail	{
	.@group_id = getarg(0);
	.@group_name$ = getarg(1);
	.@enabled = getarg(2);
	.@interval = getarg(3);
	.@event_code$ = getarg(4, "");

F_GroupDetail_Loop:
	clear;
	mes "["+.@group_name$+"]";
	mes " ";

	// Get description and event_params
	query_sql "SELECT description, last_checked, event_params FROM event_trigger_groups WHERE id = "+.@group_id, .@desc$, .@last_checked$, .@params$;

	if (.@desc$ != "") {
		mes "^777777"+.@desc$+"^000000";
		mes " ";
	}

	mes "Status: " + ((.@enabled) ? "^00AA00Enabled^000000" : "^FF0000Disabled^000000");
	mes "Check Interval: ^0055FF"+.@interval+" min^000000";
	if (.@last_checked$ != "" && .@last_checked$ != "NULL") {
		mes "Last Checked: ^777777"+.@last_checked$+"^000000";
	}

	// Show event params if set
	if (.@params$ != "" && .@params$ != "null" && .@params$ != "NULL") {
		mes " ";
		mes "^0055FFEvent Settings:^000000";
		callfunc "F_DisplayEventParams", .@event_code$, .@params$;
	}
	mes " ";

	// Show conditions summary
	query_sql "SELECT COUNT(*) FROM event_trigger_conditions WHERE group_id = "+.@group_id, .@cond_count;
	mes "Conditions: ^0055FF"+.@cond_count+"^000000";

	next;
	switch(select(
		((.@enabled) ? "Disable" : "Enable") + " Group",
		"Manage Conditions ("+.@cond_count+")",
		"View Check History",
		"Edit Event Settings",
		"Edit Check Interval",
		"^FF0000Delete Group^000000",
		"Back"
	)) {
	case 1:
		// Toggle enabled
		.@new_enabled = !.@enabled;
		query_sql "UPDATE event_trigger_groups SET enabled = "+.@new_enabled+" WHERE id = "+.@group_id;
		.@enabled = .@new_enabled;
		mes "Group " + ((.@enabled) ? "^00AA00enabled^000000" : "^FF0000disabled^000000") + ".";
		next;
		goto F_GroupDetail_Loop;
	case 2:
		callfunc "F_ManageConditions", .@group_id, .@group_name$;
		goto F_GroupDetail_Loop;
	case 3:
		callfunc "F_ViewCheckHistory", .@group_id, .@group_name$;
		goto F_GroupDetail_Loop;
	case 4:
		callfunc "F_EditEventParams", .@group_id, .@event_code$;
		goto F_GroupDetail_Loop;
	case 5:
		callfunc "F_EditInterval", .@group_id;
		query_sql "SELECT check_interval_minutes FROM event_trigger_groups WHERE id = "+.@group_id, .@interval;
		goto F_GroupDetail_Loop;
	case 6:
		if (callfunc("F_DeleteGroup", .@group_id, .@group_name$)) {
			return;  // Group deleted, go back to list
		}
		goto F_GroupDetail_Loop;
	case 7:
		return;
	}
	return;
}

// ============================================================
// View Conditions for a Group
// ============================================================
function	script	F_ViewConditions	{
	.@group_id = getarg(0);
	.@group_name$ = getarg(1);

	clear;
	mes "[Conditions]";
	mes "^0055FF"+.@group_name$+"^000000";
	mes " ";
	mes "All conditions must pass (AND logic)";
	mes "for this trigger group to fire.";
	mes " ";

	// Get conditions ordered by eval_order
	query_sql "SELECT condition_type, params, required, eval_order FROM event_trigger_conditions WHERE group_id = "+.@group_id+" ORDER BY eval_order", .@types$, .@params$, .@required, .@orders;

	.@count = getarraysize(.@types$);
	if (.@count == 0) {
		mes "^777777No conditions configured.^000000";
		next;
		select("Back");
		return;
	}

	for (.@i = 0; .@i < .@count; .@i++) {
		.@req$ = (.@required[.@i]) ? "" : " ^777777(optional)^000000";
		.@desc$ = callfunc("F_ConditionDesc", .@types$[.@i], .@params$[.@i]);
		mes "^0055FF"+(.@i+1)+".^000000 "+.@desc$+.@req$;
	}

	next;
	select("Back");
	return;
}

// ============================================================
// Get human-readable condition description
// NOTE: Avoid colons (:) - they break select() menus!
// ============================================================
function	script	F_ConditionDesc	{
	.@type$ = getarg(0);
	.@params$ = getarg(1);

	if (.@type$ == "random_chance") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.percent'))", .@pct$;
		return "^00AA00"+.@pct$+"%^000000 random chance";
	}
	else if (.@type$ == "outside_woe") {
		return "Outside WoE times";
	}
	else if (.@type$ == "cooldown") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.minutes'))", .@min$;
		return "^00AA00"+.@min$+"^000000 min cooldown";
	}
	else if (.@type$ == "time_window") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.start_hour')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.end_hour'))", .@start$, .@end$;
		return "^00AA00"+.@start$+"h - "+.@end$+"h^000000 time window";
	}
	else if (.@type$ == "player_count") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.min')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.max'))", .@min$, .@max$;
		if (.@max$ == "" || .@max$ == "null" || .@max$ == "0") {
			return "^00AA00"+.@min$+"+^000000 players online";
		}
		return "^00AA00"+.@min$+"-"+.@max$+"^000000 players online";
	}
	else if (.@type$ == "active_player_count") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.min')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.window_minutes'))", .@min$, .@win$;
		return "^00AA00"+.@min$+"+^000000 active in "+.@win$+" min";
	}
	else if (.@type$ == "no_active_event") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.events'))", .@events$;
		return "No conflict with "+.@events$;
	}
	else if (.@type$ == "day_of_week") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.days'))", .@days$;
		return "Days ["+.@days$+"]";
	}

	return .@type$+" ("+.@params$+")";
}

// ============================================================
// Edit Check Interval
// ============================================================
function	script	F_EditInterval	{
	.@group_id = getarg(0);

	clear;
	mes "[Check Interval]";
	mes " ";
	mes "How often should this trigger";
	mes "group be evaluated?";
	mes " ";
	mes "^777777Lower = more responsive but";
	mes "uses more server resources.^000000";
	next;

	switch(select(
		"Every 1 minute",
		"Every 5 minutes",
		"Every 10 minutes",
		"Every 15 minutes",
		"Every 30 minutes",
		"Cancel"
	)) {
	case 1: .@new_interval = 1; break;
	case 2: .@new_interval = 5; break;
	case 3: .@new_interval = 10; break;
	case 4: .@new_interval = 15; break;
	case 5: .@new_interval = 30; break;
	case 6: return;
	}

	query_sql "UPDATE event_trigger_groups SET check_interval_minutes = "+.@new_interval+" WHERE id = "+.@group_id;
	mes "Interval updated to ^00AA00"+.@new_interval+" minutes^000000.";
	next;
	return;
}

// ============================================================
// Test Triggers (evaluate now without waiting)
// ============================================================
function	script	F_TestTriggers	{
	.@event_type_id = getarg(0);
	.@event_code$ = getarg(1);

	clear;
	mes "[Test Triggers]";
	mes " ";
	mes "This will evaluate all trigger";
	mes "groups right now and show which";
	mes "conditions pass or fail.";
	mes " ";
	mes "^FF0000Note: This is a dry-run.^000000";
	mes "^FF0000The event won't actually start.^000000";
	next;

	if (select("Run Test:Cancel") == 2) return;

	clear;
	mes "[Test Results]";
	mes " ";

	// Get all enabled groups
	query_sql "SELECT id, name, last_ended_config_key FROM event_trigger_groups WHERE event_type_id = "+.@event_type_id+" AND enabled = 1 ORDER BY priority DESC", .@gids, .@gnames$, .@config_keys$;

	.@gcount = getarraysize(.@gids);
	if (.@gcount == 0) {
		mes "^FF0000No enabled trigger groups.^000000";
		next;
		select("Back");
		return;
	}

	for (.@g = 0; .@g < .@gcount; .@g++) {
		mes "^0055FF"+.@gnames$[.@g]+"^000000:";

		// Get conditions
		query_sql "SELECT condition_type, params, required FROM event_trigger_conditions WHERE group_id = "+.@gids[.@g]+" ORDER BY eval_order", .@ctypes$, .@cparams$, .@creq;

		.@all_passed = 1;
		.@ccount = getarraysize(.@ctypes$);

		for (.@c = 0; .@c < .@ccount; .@c++) {
			.@result = callfunc("F_EvalCondition", .@ctypes$[.@c], .@cparams$[.@c], .@event_type_id, .@config_keys$[.@g]);
			.@status$ = (.@result) ? "^00AA00PASS^000000" : "^FF0000FAIL^000000";
			.@req$ = (.@creq[.@c]) ? "" : " (opt)";
			mes "  "+.@status$+" "+callfunc("F_ConditionDesc", .@ctypes$[.@c], .@cparams$[.@c])+.@req$;

			if (!.@result && .@creq[.@c]) .@all_passed = 0;
		}

		if (.@all_passed) {
			mes "  ^00AA00--> GROUP WOULD TRIGGER^000000";
		} else {
			mes "  ^777777--> Group blocked^000000";
		}
		mes " ";
	}

	next;
	select("Back");
	return;
}

// ============================================================
// Evaluate a condition (for testing - returns value directly)
// This duplicates logic from scheduler but returns value
// ============================================================
function	script	F_EvalCondition	{
	.@type$ = getarg(0);
	.@params$ = getarg(1);
	.@event_type_id = getarg(2);
	.@config_key$ = getarg(3, "");

	if (.@type$ == "random_chance") {
		// For testing, always pass random (we want to see other conditions)
		return 1;
	}
	else if (.@type$ == "outside_woe") {
		if (agitcheck() || agitcheck2()) return 0;
		return 1;
	}
	else if (.@type$ == "no_active_event") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.events'))", .@events$;
		if (countstr(.@events$, "invasion") > 0) {
			query_sql "SELECT 1 FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'invasion' AND ei.status = 'active' LIMIT 1", .@active;
			if (.@active || $@InvasionActive) return 0;
		}
		if (countstr(.@events$, "exp_boost") > 0) {
			query_sql "SELECT 1 FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'exp_boost' AND ei.status = 'active' LIMIT 1", .@active;
			if (.@active || $@ExpBoostActive) return 0;
		}
		if (countstr(.@events$, "happy_hour") > 0) {
			query_sql "SELECT 1 FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'happy_hour' AND ei.status = 'active' LIMIT 1", .@active;
			if (.@active || $@HappyHourActive) return 0;
		}
		return 1;
	}
	else if (.@type$ == "cooldown") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.minutes'))", .@minutes$;
		.@minutes = atoi(.@minutes$);
		if (.@minutes <= 0) return 1;
		if (.@config_key$ == "") return 1;
		query_sql "SELECT config_value FROM event_config WHERE config_key = '"+escape_sql(.@config_key$)+"'", .@last_ended$;
		.@last_ended = atoi(.@last_ended$);
		if (.@last_ended == 0) return 1;
		.@elapsed_min = (gettimetick(2) - .@last_ended) / 60;
		return (.@elapsed_min >= .@minutes);
	}
	else if (.@type$ == "time_window") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.start_hour')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.end_hour'))", .@start$, .@end$;
		.@start = atoi(.@start$);
		.@end = atoi(.@end$);
		.@hour = gettime(3);
		return (.@hour >= .@start && .@hour < .@end);
	}
	else if (.@type$ == "player_count") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.min')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.max'))", .@min$, .@max$;
		.@min = atoi(.@min$);
		.@max = atoi(.@max$);
		.@count = getusers(1);
		if (.@min > 0 && .@count < .@min) return 0;
		if (.@max > 0 && .@count > .@max) return 0;
		return 1;
	}
	else if (.@type$ == "active_player_count") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.min')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.window_minutes'))", .@min$, .@window$;
		.@min = atoi(.@min$);
		.@window = atoi(.@window$);
		if (.@window <= 0) .@window = 10;
		query_sql "SELECT COUNT(*) FROM player_activity_tracking WHERE position_changed = 1 AND last_seen >= DATE_SUB(NOW(), INTERVAL "+.@window+" MINUTE)", .@active_count;
		return (.@active_count >= .@min);
	}
	else if (.@type$ == "day_of_week") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.days'))", .@days$;
		.@today = gettime(4);
		.@today_mon = (.@today == 0) ? 6 : (.@today - 1);
		return (countstr(.@days$, ""+.@today_mon) > 0);
	}

	return 0;
}

// ============================================================
// View Recent Triggers (history)
// ============================================================
function	script	F_RecentTriggers	{
	.@event_type_id = getarg(0);
	.@event_name$ = getarg(1);

	clear;
	mes "[Recent Triggers]";
	mes "^0055FF"+.@event_name$+"^000000";
	mes " ";

	// Get last 5 conditional triggers
	query_sql "SELECT DATE_FORMAT(actual_start, '%m/%d %H:%i'), tg.name FROM event_instances ei LEFT JOIN event_trigger_groups tg ON ei.trigger_group_id = tg.id WHERE ei.event_type_id = "+.@event_type_id+" AND ei.trigger_type = 'conditional' ORDER BY ei.actual_start DESC LIMIT 5", .@times$, .@groups$;

	.@count = getarraysize(.@times$);
	if (.@count == 0) {
		mes "No conditional triggers yet.";
	} else {
		for (.@i = 0; .@i < .@count; .@i++) {
			.@group$ = (.@groups$[.@i] != "") ? .@groups$[.@i] : "Unknown";
			mes "^0055FF"+.@times$[.@i]+"^000000 - "+.@group$;
		}
	}

	next;
	select("Back");
	return;
}

// ============================================================
// Create New Trigger Group
// ============================================================
function	script	F_CreateTriggerGroup	{
	.@event_type_id = getarg(0);
	.@event_code$ = getarg(1);

	clear;
	mes "[Create Trigger Group]";
	mes " ";
	mes "Create a new conditional trigger";
	mes "group for this event.";
	mes " ";
	mes "^777777A trigger group is a set of";
	mes "conditions that must ALL pass";
	mes "for the event to start.^000000";
	next;

	// Get group name
	mes "[Group Name]";
	mes " ";
	mes "Enter a name for this trigger group:";
	mes "^777777(e.g., 'Peak Hours', 'Weekend Rush')^000000";
	next;
	input .@name$;

	if (.@name$ == "" || getstrlen(.@name$) < 3) {
		mes "^FF0000Name must be at least 3 characters.^000000";
		next;
		return;
	}

	// Get description
	clear;
	mes "[Description]";
	mes " ";
	mes "Enter a brief description:";
	mes "^777777(optional - press Enter to skip)^000000";
	next;
	input .@desc$;

	// Get check interval
	clear;
	mes "[Check Interval]";
	mes " ";
	mes "How often should conditions be";
	mes "evaluated?";
	next;

	switch(select(
		"Every 1 minute (responsive)",
		"Every 5 minutes (balanced)",
		"Every 10 minutes (low load)",
		"Every 15 minutes",
		"Cancel"
	)) {
	case 1: .@interval = 1; break;
	case 2: .@interval = 5; break;
	case 3: .@interval = 10; break;
	case 4: .@interval = 15; break;
	case 5: return;
	}

	// Get cooldown config key
	.@config_key$ = .@event_code$ + "_last_ended";

	// Insert the group
	query_sql "INSERT INTO event_trigger_groups (event_type_id, name, description, enabled, check_interval_minutes, last_ended_config_key) VALUES ("+.@event_type_id+", '"+escape_sql(.@name$)+"', '"+escape_sql(.@desc$)+"', 0, "+.@interval+", '"+escape_sql(.@config_key$)+"')";

	// Get the new group ID
	query_sql "SELECT LAST_INSERT_ID()", .@group_id;

	clear;
	mes "[Group Created]";
	mes " ";
	mes "^00AA00"+.@name$+"^000000 created!";
	mes " ";
	mes "The group is ^FF0000disabled^000000 by default.";
	mes " ";
	mes "Next steps:";
	mes "1. Add conditions to the group";
	mes "2. Configure event settings";
	mes "3. Enable when ready";
	next;

	if (select("Add Conditions Now:Done") == 1) {
		callfunc "F_ManageConditions", .@group_id, .@name$;
	}
	return;
}

// ============================================================
// Manage Conditions (list with add/edit/delete)
// ============================================================
function	script	F_ManageConditions	{
	.@group_id = getarg(0);
	.@group_name$ = getarg(1);

F_ManageConditions_Loop:
	clear;
	mes "[Conditions]";
	mes "^0055FF"+.@group_name$+"^000000";
	mes " ";
	mes "All conditions must pass (AND logic)";
	mes "for this trigger to fire.";
	mes " ";

	// Get conditions
	query_sql "SELECT id, condition_type, params, required FROM event_trigger_conditions WHERE group_id = "+.@group_id+" ORDER BY eval_order", .@ids, .@types$, .@params$, .@required;

	.@count = getarraysize(.@ids);

	if (.@count == 0) {
		mes "^777777No conditions yet.^000000";
	} else {
		for (.@i = 0; .@i < .@count; .@i++) {
			.@req$ = (.@required[.@i]) ? "" : " ^777777(opt)^000000";
			.@desc$ = callfunc("F_ConditionDesc", .@types$[.@i], .@params$[.@i]);
			mes "^0055FF"+(.@i+1)+".^000000 "+.@desc$+.@req$;
		}
	}
	mes " ";
	next;

	// Build menu
	.@menu$ = "Add Condition:";
	if (.@count > 0) {
		.@menu$ = .@menu$ + "Edit Condition:Delete Condition:";
	}
	.@menu$ = .@menu$ + "Back";

	.@sel = select(.@menu$);

	if (.@count == 0) {
		// Only Add and Back available
		if (.@sel == 1) {
			callfunc "F_AddCondition", .@group_id;
			goto F_ManageConditions_Loop;
		}
		return;  // Back
	}

	switch(.@sel) {
	case 1:
		callfunc "F_AddCondition", .@group_id;
		goto F_ManageConditions_Loop;
	case 2:
		// Edit - select which condition
		clear;
		mes "[Edit Condition]";
		mes "Select condition to edit:";
		next;
		.@edit_menu$ = "";
		for (.@i = 0; .@i < .@count; .@i++) {
			.@edit_menu$ = .@edit_menu$ + callfunc("F_ConditionDesc", .@types$[.@i], .@params$[.@i]) + ":";
		}
		.@edit_menu$ = .@edit_menu$ + "Cancel";
		.@edit_sel = select(.@edit_menu$) - 1;
		if (.@edit_sel < .@count) {
			callfunc "F_EditCondition", .@ids[.@edit_sel], .@types$[.@edit_sel], .@params$[.@edit_sel];
		}
		goto F_ManageConditions_Loop;
	case 3:
		// Delete - select which condition
		clear;
		mes "[Delete Condition]";
		mes "Select condition to delete:";
		next;
		.@del_menu$ = "";
		for (.@i = 0; .@i < .@count; .@i++) {
			.@del_menu$ = .@del_menu$ + callfunc("F_ConditionDesc", .@types$[.@i], .@params$[.@i]) + ":";
		}
		.@del_menu$ = .@del_menu$ + "Cancel";
		.@del_sel = select(.@del_menu$) - 1;
		if (.@del_sel < .@count) {
			callfunc "F_DeleteCondition", .@ids[.@del_sel], .@types$[.@del_sel];
		}
		goto F_ManageConditions_Loop;
	default:
		return;  // Back
	}
	return;
}

function	script	F_AddCond_NoConflict	{
	.@group_id = getarg(0);

	clear;
	mes "[No Conflicting Events]";
	mes " ";
	mes "Prevents trigger if certain other";
	mes "events are currently active.";
	mes " ";
	mes "Select events to avoid:";
	next;

	.@events$ = "";
	.@sel = select("Invasion:EXP Boost:Happy Hour:Done");

	while (.@sel != 4) {
		switch(.@sel) {
		case 1:
			if (countstr(.@events$, "invasion") == 0) {
				.@events$ = .@events$ + ((.@events$ != "") ? "," : "") + "invasion";
			}
			break;
		case 2:
			if (countstr(.@events$, "exp_boost") == 0) {
				.@events$ = .@events$ + ((.@events$ != "") ? "," : "") + "exp_boost";
			}
			break;
		case 3:
			if (countstr(.@events$, "happy_hour") == 0) {
				.@events$ = .@events$ + ((.@events$ != "") ? "," : "") + "happy_hour";
			}
			break;
		}

		clear;
		mes "[No Conflicting Events]";
		mes " ";
		mes "Selected: ^0055FF" + ((.@events$ == "") ? "none" : .@events$) + "^000000";
		mes " ";
		mes "Add more or Done:";
		next;
		.@sel = select("Invasion:EXP Boost:Happy Hour:Done");
	}

	if (.@events$ == "") {
		mes "^FF0000No events selected.^000000";
		next;
		return;
	}

	query_sql "INSERT INTO event_trigger_conditions (group_id, condition_type, params, required, eval_order) VALUES ("+.@group_id+", 'no_active_event', '{\"events\": \""+escape_sql(.@events$)+"\"}', 1, 15)";

	next;
	return;
}

function	script	F_DisplayEventParams	{
	.@event_code$ = getarg(0);

	if (.@event_code$ == "invasion") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.threat')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.duration_minutes'))", .@threat$, .@dur$;

			.@threat_name$ = "";
			if (.@threat$ == "1") .@threat_name$ = "Low";
			else if (.@threat$ == "2") .@threat_name$ = "Medium";
			else if (.@threat$ == "3") .@threat_name$ = "High";
			else if (.@threat$ == "4") .@threat_name$ = "Extreme";
			else .@threat_name$ = "Random";
			mes "  Threat: ^00AA00"+.@threat_name$+"^000000";
		}
		if (.@dur$ != "" && .@dur$ != "null") {
			mes "  Duration: ^00AA00"+.@dur$+" min^000000";
		}
	}
	else if (.@event_code$ == "exp_boost") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.multiplier')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.duration_minutes'))", .@mult$, .@dur$;

			.@mult = atoi(.@mult$);
			.@whole = .@mult / 100;
			.@dec = (.@mult % 100) / 10;
			if (.@dec > 0) {
				mes "  Multiplier: ^00AA00"+.@whole+"."+.@dec+"x^000000";
			} else {
				mes "  Multiplier: ^00AA00"+.@whole+"x^000000";
			}
		}
		if (.@dur$ != "" && .@dur$ != "null") {
			mes "  Duration: ^00AA00"+.@dur$+" min^000000";
		}
	}
	else if (.@event_code$ == "happy_hour") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.duration_minutes'))", .@dur$;
		if (.@dur$ != "" && .@dur$ != "null") {
			mes "  Duration: ^00AA00"+.@dur$+" min^000000";
		}
	}
	else {
		mes "  ^777777("+.@params$+")^000000";
	}
	return;
}

			}
			if (.@current$ == "" || .@current$ == "null") {
				query_sql "UPDATE event_trigger_groups SET event_params = JSON_OBJECT('duration_minutes', "+.@dur+") WHERE id = "+.@group_id;
			} else {
				query_sql "UPDATE event_trigger_groups SET event_params = JSON_SET(event_params, '$.duration_minutes', "+.@dur+") WHERE id = "+.@group_id;
			}
			mes "^00AA00Duration set to "+.@dur+" min.^000000";
			next;
			break;
		case 3:
			query_sql "UPDATE event_trigger_groups SET event_params = NULL WHERE id = "+.@group_id;
			mes "^00AA00Settings cleared.^000000";
			next;
			break;
		case 4:
			return;
		}
	}
	else if (.@event_code$ == "happy_hour") {
		// Happy Hour settings: duration only
		mes "Current settings:";
		if (.@current$ != "" && .@current$ != "null") {
			callfunc "F_DisplayEventParams", .@event_code$, .@current$;
		} else {
			mes "  ^777777(using defaults)^000000";
		}
		next;

		switch(select(
			"Set Duration",
			"Clear (use default)",
			"Back"
		)) {
		case 1:
			clear;
			mes "[Duration]";
			mes " ";
			mes "Enter happy hour duration (minutes):";
			mes "^777777Default: 120^000000";
			next;
			input .@dur;
			if (.@dur < 5 || .@dur > 480) {
				mes "^FF0000Must be 5-480 minutes.^000000";
				next;
				return;
			}
			query_sql "UPDATE event_trigger_groups SET event_params = JSON_OBJECT('duration_minutes', "+.@dur+") WHERE id = "+.@group_id;
			mes "^00AA00Duration set to "+.@dur+" min.^000000";
			next;
			break;
		case 2:
			query_sql "UPDATE event_trigger_groups SET event_params = NULL WHERE id = "+.@group_id;
			mes "^00AA00Settings cleared.^000000";
			next;
			break;
		case 3:
			return;
		}
	}
	else {
		mes "^777777No configurable settings for";
		mes "this event type.^000000";
		next;
		select("Back");
	}
	return;
}

// ============================================================
// View Check History (debug log)
// Shows last N evaluations of this trigger group
// ============================================================
function	script	F_ViewCheckHistory	{
	.@group_id = getarg(0);
	.@group_name$ = getarg(1);

F_CheckHistory_Loop:
	clear;
	mes "[Check History]";
	mes "^0055FF"+.@group_name$+"^000000";
	mes " ";

	// Get last 10 checks (use 'h' instead of ':' in time to avoid menu breaking)
	query_sql "SELECT DATE_FORMAT(checked_at, '%m/%d %Hh%i'), group_passed, condition_results, online_count FROM trigger_check_log WHERE group_id = "+.@group_id+" ORDER BY checked_at DESC LIMIT 10", .@times$, .@passed, .@results$, .@online;

	.@count = getarraysize(.@times$);
	if (.@count == 0) {
		mes "No check history yet.";
		mes " ";
		mes "^777777Checks are logged each time";
		mes "this trigger group is evaluated.^000000";
		next;
		select("Back");
		return;
	}

	mes "Last "+.@count+" checks:";
	mes " ";

	// Pre-compute fail reasons for display
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@passed[.@i]) {
			.@reason$[.@i] = "all passed";
		} else {
			// Find first failing required condition
			.@reason$[.@i] = "";
			query_sql "SELECT JSON_LENGTH('"+escape_sql(.@results$[.@i])+"')", .@num_conds;
			for (.@c = 0; .@c < .@num_conds && .@reason$[.@i] == ""; .@c++) {
				query_sql "SELECT JSON_EXTRACT('"+escape_sql(.@results$[.@i])+"', '$["+.@c+"].passed'), JSON_EXTRACT('"+escape_sql(.@results$[.@i])+"', '$["+.@c+"].required'), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@results$[.@i])+"', '$["+.@c+"].type')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@results$[.@i])+"', '$["+.@c+"].detail'))", .@c_passed$, .@c_req$, .@c_type$, .@c_detail$;
				if (.@c_passed$ == "0" && .@c_req$ == "1") {
					if (.@c_detail$ != "" && .@c_detail$ != "null") {
						.@reason$[.@i] = .@c_detail$;
					} else {
						.@reason$[.@i] = .@c_type$ + " failed";
					}
				}
			}
			if (.@reason$[.@i] == "") .@reason$[.@i] = "unknown";
		}
	}

	for (.@i = 0; .@i < .@count && .@i < 5; .@i++) {
		.@status$ = (.@passed[.@i]) ? "^00AA00PASS^000000" : "^FF0000FAIL^000000";
		mes .@times$[.@i] + " " + .@status$ + " (" + .@reason$[.@i] + ")";
	}

	next;
	// Build menu for detail view
	.@menu$ = "";
	for (.@i = 0; .@i < .@count; .@i++) {
		.@status$ = (.@passed[.@i]) ? "PASS" : "FAIL";
		.@menu$ = .@menu$ + .@times$[.@i] + " " + .@status$ + ":";
	}
	.@menu$ = .@menu$ + "Back";

	.@sel = select(.@menu$) - 1;
	if (.@sel >= .@count) return;

	// Show detail for selected check
	callfunc "F_ShowCheckDetail", .@times$[.@sel], .@passed[.@sel], .@results$[.@sel], .@online[.@sel];
	goto F_CheckHistory_Loop;
}

// ============================================================
// Show detail of a single check
// ============================================================
function	script	F_ShowCheckDetail	{
	.@time$ = getarg(0);
	.@passed = getarg(1);
	.@results$ = getarg(2);
	.@online = getarg(3);

	clear;
	mes "[Check Detail]";
	mes "^0055FF"+.@time$+"^000000";
	mes " ";
	mes "Result: " + ((.@passed) ? "^00AA00PASSED^000000" : "^FF0000FAILED^000000");
	mes "Online: ^0055FF"+.@online+"^000000 players";
	mes " ";
	mes "Conditions:";

	// Parse JSON array of condition results
	// Format: [{"type":"random_chance","passed":0,"detail":"rolled 45 vs 5%"},...]
	// Use SQL to parse JSON array elements
	query_sql "SELECT JSON_LENGTH('"+escape_sql(.@results$)+"')", .@num_conds;

	for (.@i = 0; .@i < .@num_conds; .@i++) {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@results$)+"', '$["+.@i+"].type')), JSON_EXTRACT('"+escape_sql(.@results$)+"', '$["+.@i+"].passed'), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@results$)+"', '$["+.@i+"].detail')), JSON_EXTRACT('"+escape_sql(.@results$)+"', '$["+.@i+"].required')", .@type$, .@cond_passed$, .@detail$, .@required$;

		.@cond_passed = atoi(.@cond_passed$);
		.@required = atoi(.@required$);

		.@status$ = (.@cond_passed) ? "^00AA00OK^000000" : "^FF0000X^000000";
		.@req$ = (.@required) ? "" : " ^777777(opt)^000000";

		if (.@detail$ != "" && .@detail$ != "null") {
			mes .@status$ + " " + .@type$ + " - " + .@detail$ + .@req$;
		} else {
			mes .@status$ + " " + .@type$ + .@req$;
		}
	}

	next;
	select("Back");
	return;
}
