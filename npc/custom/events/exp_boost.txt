//===== rAthena Script =======================================
//= Custom EXP Boost Event
//= GM-configurable server-wide EXP multiplier
//===== Description: =========================================
//= Server-wide EXP boost with configurable multiplier.
//= - Triggered by GMs or automatically via scheduler
//= - Multiplier is configurable (1.5x to 5x)
//= - Always has a duration (auto-ends when time expires)
//= - Stacks with other events (Happy Hour, MOTD)
//= - Uses setbattleflag for true server-wide effect
//============================================================

-	script	ExpBoost	FAKE_NPC,{
	end;

OnInit:
	// Initialize state
	set $@ExpBoostActive, 0;
	set $@ExpBoostMultiplier, 0;
	set $@ExpBoostDuration, 0;
	set $@ExpBoostStartTime, 0;
	set $@ExpBoostEndTime, 0;
	end;

// Timer to check if boost should end (runs every 60 seconds when active)
OnTimer60000:
	stopnpctimer;
	if ($@ExpBoostActive && $@ExpBoostEndTime > 0 && gettimetick(2) >= $@ExpBoostEndTime) {
		donpcevent strnpcinfo(0)+"::OnEventEnd";
	} else if ($@ExpBoostActive) {
		// Continue timer
		setnpctimer 0;
		initnpctimer;
	}
	end;

// ============================================================
// Event Start - Called by Events Manager or Scheduler
// Expects $@ExpBoostMultiplier to be set (e.g., 200 = 2x, 300 = 3x)
// Expects $@ExpBoostDuration in minutes
// ============================================================
OnEventStart:
	if ($@ExpBoostActive) {
		end;
	}

	// Validate multiplier (minimum 150 = 1.5x, maximum 500 = 5x)
	if ($@ExpBoostMultiplier < 150 || $@ExpBoostMultiplier > 500) {
		set $@ExpBoostMultiplier, 200;
	}

	// Store original rates
	set $@EB_OrigBaseExp, getbattleflag("base_exp_rate");
	set $@EB_OrigJobExp, getbattleflag("job_exp_rate");

	// Calculate new rates
	// Multiplier is in percentage (200 = 2x), so: orig_rate * multiplier / 100
	set .@new_base, $@EB_OrigBaseExp * $@ExpBoostMultiplier / 100;
	set .@new_job, $@EB_OrigJobExp * $@ExpBoostMultiplier / 100;

	// Apply new rates
	setbattleflag("base_exp_rate", .@new_base);
	setbattleflag("job_exp_rate", .@new_job, true);  // true = reload

	// Mark as active
	set $@ExpBoostActive, 1;
	set $@ExpBoostStartTime, gettimetick(2);

	// Set end time and start NPC timer
	if ($@ExpBoostDuration > 0) {
		set $@ExpBoostEndTime, gettimetick(2) + ($@ExpBoostDuration * 60);
		// NPC timer to auto-end (doesn't require player context)
		stopnpctimer;
		setnpctimer 0;
		initnpctimer;
	} else {
		set $@ExpBoostEndTime, 0;
	}

	// Format multiplier for display (e.g., 200 -> "2x", 150 -> "1.5x")
	set .@whole, $@ExpBoostMultiplier / 100;
	set .@decimal, ($@ExpBoostMultiplier % 100) / 10;
	if (.@decimal > 0) {
		set .@mult$, .@whole + "." + .@decimal + "x";
	} else {
		set .@mult$, .@whole + "x";
	}

	// Note: event_instances is created by EventScheduler when it triggers this event
	// $@ExpBoostInstanceId is set by scheduler before calling OnEventStart

	// Format duration for display
	set .@dur$, callfunc("F_FormatDuration", $@ExpBoostDuration);

	// Announce
	announce "===========================================", bc_all, 0x00BFFF;
	announce "      EXP BOOST ACTIVATED!", bc_all, 0x00BFFF;
	announce "   "+.@mult$+" EXP rates for "+.@dur$+"!", bc_all, 0x00BFFF;
	announce "===========================================", bc_all, 0x00BFFF;

	debugmes "[ExpBoost] Started - "+.@mult$+" for "+.@dur$;
	end;

// ============================================================
// Event End - Called by Events Manager or auto-timer
// ============================================================
OnEventEnd:
	if (!$@ExpBoostActive) {
		end;
	}

	// Stop the NPC timer
	stopnpctimer;

	// Restore original rates
	setbattleflag("base_exp_rate", $@EB_OrigBaseExp);
	setbattleflag("job_exp_rate", $@EB_OrigJobExp, true);  // true = reload

	// Mark the schedule entry as completed (for one-time events)
	if ($@ExpBoostScheduleId > 0) {
		query_sql "UPDATE event_schedule SET status = 'completed', actual_end = NOW() WHERE id = "+$@ExpBoostScheduleId;
	}

	// Update instance and cache via helper function
	callfunc "F_EndEventInstance", $@ExpBoostInstanceId;

	// Mark as inactive
	set $@ExpBoostActive, 0;
	set $@ExpBoostMultiplier, 0;
	set $@ExpBoostDuration, 0;
	set $@ExpBoostStartTime, 0;
	set $@ExpBoostEndTime, 0;
	set $@ExpBoostScheduleId, 0;
	set $@ExpBoostInstanceId, 0;

	// Announce
	announce "===========================================", bc_all, 0x00BFFF;
	announce "      EXP BOOST DEACTIVATED!", bc_all, 0x00BFFF;
	announce "   EXP rates have returned to normal.", bc_all, 0x00BFFF;
	announce "===========================================", bc_all, 0x00BFFF;

	debugmes "[ExpBoost] Ended";
	end;
}

// ============================================================
// Helper function to check if EXP Boost is active
// ============================================================
function	script	F_IsExpBoostActive	{
	return $@ExpBoostActive;
}

// ============================================================
// Helper function to get current EXP Boost multiplier
// Returns multiplier as percentage (e.g., 200 = 2x)
// ============================================================
function	script	F_GetExpBoostMultiplier	{
	if (!$@ExpBoostActive) return 100;
	return $@ExpBoostMultiplier;
}

// ============================================================
// Helper function to get EXP Boost elapsed time (minutes since start)
// ============================================================
function	script	F_ExpBoostDuration	{
	if (!$@ExpBoostActive) return 0;
	return (gettimetick(2) - $@ExpBoostStartTime) / 60;
}

// ============================================================
// Helper function to get EXP Boost remaining time (minutes until end)
// ============================================================
function	script	F_ExpBoostRemaining	{
	if (!$@ExpBoostActive || $@ExpBoostEndTime == 0) return 0;
	set .@remaining, ($@ExpBoostEndTime - gettimetick(2)) / 60;
	if (.@remaining < 0) return 0;
	return .@remaining;
}

// ============================================================
// Helper function to format multiplier for display
// ============================================================
function	script	F_FormatExpMultiplier	{
	set .@mult, getarg(0, 100);
	set .@whole, .@mult / 100;
	set .@decimal, (.@mult % 100) / 10;
	if (.@decimal > 0) {
		return .@whole + "." + .@decimal + "x";
	}
	return .@whole + "x";
}

// ============================================================
// Helper function to format duration for display
// Converts minutes to human-readable format
// ============================================================
function	script	F_FormatDuration	{
	set .@mins, getarg(0, 0);

	if (.@mins >= 1440) {
		// Days
		set .@days, .@mins / 1440;
		set .@remaining_hours, (.@mins % 1440) / 60;
		if (.@remaining_hours > 0) {
			return .@days + " day" + (.@days > 1 ? "s" : "") + " " + .@remaining_hours + "h";
		}
		return .@days + " day" + (.@days > 1 ? "s" : "");
	} else if (.@mins >= 60) {
		// Hours
		set .@hours, .@mins / 60;
		set .@remaining_mins, .@mins % 60;
		if (.@remaining_mins > 0) {
			return .@hours + " hour" + (.@hours > 1 ? "s" : "") + " " + .@remaining_mins + "m";
		}
		return .@hours + " hour" + (.@hours > 1 ? "s" : "");
	}
	// Minutes only
	return .@mins + " minutes";
}
