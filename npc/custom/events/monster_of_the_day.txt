//===== rAthena Script =======================================
//= Monster of the Day Display NPC
//= Shows current and upcoming MOTD monsters
//===== Description: =========================================
//= Player-facing NPC that displays:
//= - Current featured monster with time remaining and location
//= - Next 5 upcoming monsters
//= All monsters are from the Bounty Board
//= Note: The actual 3x bonus is applied in bounty_board.txt
//============================================================

prontera,164,175,5	script	Monster Hunter Board	4_BOARD3,{
	mes "[Monster Hunter Board]";
	mes "^0055FFToday's Featured Hunts^000000";
	mes "Featured monsters grant";
	mes "^FF00003x EXP^000000 and ^FF00003x Hunt Points^000000!";
	next;

	// Get current MOTD with location from motd_monsters
	query_sql "SELECT ms.mob_id, ms.mob_name, ms.tier, TIMESTAMPDIFF(MINUTE, NOW(), ms.end_time), COALESCE(mm.map_display, '') FROM motd_schedule ms LEFT JOIN motd_monsters mm ON ms.mob_id = mm.mob_id WHERE NOW() BETWEEN ms.start_time AND ms.end_time LIMIT 1", .@cur_id, .@cur_name$, .@cur_tier$, .@cur_mins, .@cur_loc$;

	if (.@cur_id > 0) {
		// Format time remaining
		if (.@cur_mins >= 60) {
			set .@time$, (.@cur_mins / 60)+"h "+(.@cur_mins % 60)+"m";
		} else {
			set .@time$, .@cur_mins+"m";
		}

		mes "^FF0000>> NOW FEATURED <<^000000";
		mes "^0000FF"+.@cur_name$+"^000000 ("+.@cur_id+")";
		mes "Tier: "+.@cur_tier$;
		mes "Time left: ^00AA00"+.@time$+"^000000";
		if (.@cur_loc$ != "") {
			mes "Location: ^777777"+.@cur_loc$+"^000000";
		}
		next;
	} else {
		mes "^888888No monster currently featured.^000000";
		mes "Schedule may be generating...";
		next;
	}

	// Get upcoming monsters with locations
	mes "[Upcoming Monsters]";
	query_sql "SELECT ms.mob_id, ms.mob_name, TIMESTAMPDIFF(MINUTE, NOW(), ms.start_time), COALESCE(mm.map_display, '') FROM motd_schedule ms LEFT JOIN motd_monsters mm ON ms.mob_id = mm.mob_id WHERE ms.start_time > NOW() ORDER BY ms.start_time LIMIT 5", .@up_ids, .@up_names$, .@up_mins, .@up_locs$;

	if (getarraysize(.@up_ids) == 0) {
		mes "^888888No upcoming monsters scheduled.^000000";
	} else {
		for (set .@i, 0; .@i < getarraysize(.@up_ids); set .@i, .@i + 1) {
			// Format time until
			if (.@up_mins[.@i] >= 60) {
				set .@until$, (.@up_mins[.@i] / 60)+"h "+(.@up_mins[.@i] % 60)+"m";
			} else {
				set .@until$, .@up_mins[.@i]+"m";
			}

			mes "^0055FF"+.@up_names$[.@i]+"^000000 ("+.@up_ids[.@i]+") - in "+.@until$;
			if (.@up_locs$[.@i] != "") {
				mes "  ^777777@ "+.@up_locs$[.@i]+"^000000";
			}
		}
	}
	next;

	mes "[Monster Hunter Board]";
	mes "All featured monsters have";
	mes "^00AA00Bounty Board quests^000000!";
	mes " ";
	mes "Hunt the featured monster and";
	mes "complete your bounty for 3x rewards!";
	close;
}

// ============================================================
// MOTD Seeder (now handled by migration 043)
// Monster pool is populated from bounty board monsters only
// ============================================================
-	script	MOTDSeeder	FAKE_NPC,{
	end;

OnSeedMonsters:
	// Monster pool is now seeded via migration 043_seed_motd_from_bounty_board.sql
	// This ensures only bounty board monsters (level 50+) are used
	// To refresh: delete from motd_monsters, then re-run the migration
	query_sql "SELECT COUNT(*) FROM motd_monsters", .@count;
	announce "[MOTD] Pool has "+.@count+" monsters (from bounty board).", bc_self;
	debugmes "[MOTDSeeder] Pool has "+.@count+" monsters (seeded via migration)";
	end;
}
