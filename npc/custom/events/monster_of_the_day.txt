//===== rAthena Script =======================================
//= Monster of the Day Display NPC
//= Shows current and upcoming MOTD monsters
//===== Description: =========================================
//= Player-facing NPC that displays:
//= - Current featured monster with time remaining and location
//= - Next 5 upcoming monsters
//= All monsters are from the Bounty Board
//= Note: The actual 3x bonus is applied in bounty_board.txt
//============================================================

prontera,164,175,5	script	Monster Hunter Board	4_BOARD3,{
	mes "[Monster Hunter Board]";

	// Check if MOTD is enabled
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@enabled$;
	if (.@enabled$ == "0") {
		mes "^888888Monster of the Day^000000";
		mes "^888888is currently disabled.^000000";
		mes " ";
		mes "Check back later for";
		mes "featured hunts!";

		// Allow GMs to still access management
		if (getgmlevel() >= 80) {
			next;
			mes "[Monster Hunter Board]";
			mes "^FF8800[GM Access]^000000";
			mes "MOTD is disabled but you";
			mes "can manage it from here.";
			next;
			if (select("^FF8800[GM] Manage MOTD^000000:Close") == 1) {
				goto L_GMMenu;
			}
		}
		close;
	}

	mes "^0055FFToday's Featured Hunts^000000";
	mes "Featured monsters grant";
	mes "^FF00003x EXP^000000, ^FF00003x Drops^000000,";
	mes "and ^FF00003x Hunt Points^000000!";
	next;

	// Get current MOTD with location from event_motd_monsters
	query_sql "SELECT ms.mob_id, ms.mob_name, ms.tier, TIMESTAMPDIFF(MINUTE, NOW(), ms.end_time), COALESCE(mm.map_display, '') FROM event_motd_schedule ms LEFT JOIN event_motd_monsters mm ON ms.mob_id = mm.mob_id WHERE NOW() BETWEEN ms.start_time AND ms.end_time LIMIT 1", .@cur_id, .@cur_name$, .@cur_tier$, .@cur_mins, .@cur_loc$;

	if (.@cur_id > 0) {
		// Get monster level for display
		.@mob_lv = getmonsterinfo(.@cur_id, MOB_LV);

		// Format time remaining
		if (.@cur_mins >= 60) {
			set .@time$, (.@cur_mins / 60)+"h "+(.@cur_mins % 60)+"m";
		} else {
			set .@time$, .@cur_mins+"m";
		}

		mes "^FF0000>> NOW FEATURED <<^000000";
		mes "^0000FF"+.@cur_name$+" ["+.@cur_id+"]^000000";
		mes "Level: "+.@mob_lv+" | Tier: "+.@cur_tier$;
		mes "Time left: ^00AA00"+.@time$+"^000000";
		if (.@cur_loc$ != "") {
			mes "Location: ^777777"+.@cur_loc$+"^000000";
		}
		mes " ";
		mes "^FF8800Bounty Rewards (3x):^000000";
		// Calculate sample rewards for this monster (reuse .@mob_lv from above)
		.@exp_scale = (.@mob_lv > 10) ? ((.@mob_lv - 10) * (.@mob_lv - 10) / 98) : 1;
		if (.@exp_scale < 1) .@exp_scale = 1;
		.@base_exp = ((100 + .@mob_lv * 5) * 50 * 100 * .@exp_scale) / 100;  // 50 kills
		.@quest_rate = getbattleflag("quest_exp_rate");
		if (.@quest_rate < 1) .@quest_rate = 100;
		.@base_exp = .@base_exp * (.@quest_rate / 100) * 3;  // Apply 3x MOTD
		.@zeny = ((20 + .@mob_lv) * 50 * 100) / 5;
		.@points = ((.@mob_lv / 10 + 1) * 100 * 50 / 25) / 100 * 3;  // Apply 3x MOTD
		if (.@points < 1) .@points = 1;
		.@next_exp = (NextBaseExp > 0) ? NextBaseExp : 1;
		.@pct = .@base_exp * 100 / .@next_exp;
		mes "~"+.@pct+"% EXP, "+.@points+" Hunt Pts";
		next;
	} else {
		mes "^888888No monster currently featured.^000000";
		mes "Schedule may be generating...";
		next;
	}

	// Get upcoming monsters with locations (configurable, up to 50, paginated)
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_upcoming_count'", .@upcoming_count$;
	if (.@upcoming_count$ == "") set .@max_upcoming, 20; else set .@max_upcoming, atoi(.@upcoming_count$);
	if (.@max_upcoming < 10) set .@max_upcoming, 10;
	if (.@max_upcoming > 50) set .@max_upcoming, 50;
	.@per_page = 10;

	query_sql "SELECT mob_id, mob_name, TIMESTAMPDIFF(MINUTE, NOW(), start_time) FROM event_motd_schedule WHERE start_time > NOW() ORDER BY start_time LIMIT " + .@max_upcoming, .@up_ids, .@up_names$, .@up_mins;

	.@total = getarraysize(.@up_ids);
	.@page = 0;

L_ShowUpcoming:
	mes "[Upcoming Monsters]";

	if (.@total == 0) {
		mes "^888888No upcoming monsters scheduled.^000000";
		next;
		goto L_AfterUpcoming;
	}

	.@start = .@page * .@per_page;
	.@end = .@start + .@per_page;
	if (.@end > .@total) .@end = .@total;
	.@total_pages = (.@total + .@per_page - 1) / .@per_page;

	mes "Page " + (.@page + 1) + " of " + .@total_pages + " ^888888(" + .@total + " total)^000000";
	mes " ";

	for (set .@i, .@start; .@i < .@end; set .@i, .@i + 1) {
		// Format time until
		if (.@up_mins[.@i] >= 60) {
			set .@until$, (.@up_mins[.@i] / 60)+"h "+(.@up_mins[.@i] % 60)+"m";
		} else {
			set .@until$, .@up_mins[.@i]+"m";
		}

		// Get level for display
		.@up_lv = getmonsterinfo(.@up_ids[.@i], MOB_LV);
		mes "Lv"+.@up_lv+" ^0055FF"+.@up_names$[.@i]+"^000000 ("+.@up_ids[.@i]+") - in "+.@until$;
	}
	next;

	// Pagination menu
	.@menu$ = "";
	if (.@page > 0) .@menu$ = .@menu$ + "< Previous Page:";
	if (.@page < .@total_pages - 1) .@menu$ = .@menu$ + "Next Page >:";
	.@menu$ = .@menu$ + "Continue";

	.@choice = select(.@menu$);
	.@idx = 1;

	if (.@page > 0) {
		if (.@choice == .@idx) {
			.@page--;
			goto L_ShowUpcoming;
		}
		.@idx++;
	}
	if (.@page < .@total_pages - 1) {
		if (.@choice == .@idx) {
			.@page++;
			goto L_ShowUpcoming;
		}
		.@idx++;
	}
	// Continue selected

L_AfterUpcoming:

	// Menu options
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'crowdfund_motd_enabled'", .@reshuffle_enabled$;

	mes "[Monster Hunter Board]";
	mes "What would you like to do?";
	next;

	.@menu$ = "Hunt Information";
	if (.@reshuffle_enabled$ == "1" && .@cur_id > 0) {
		.@menu$ = .@menu$ + ":^FF8800Reshuffle Monster^000000";
		.@has_reshuffle = 1;
	} else {
		.@has_reshuffle = 0;
	}
	.@menu$ = .@menu$ + ":Reshuffle Hall of Fame";
	if (getgmlevel() >= 80) {
		.@menu$ = .@menu$ + ":^FF8800[GM] Manage MOTD^000000";
	}
	.@menu$ = .@menu$ + ":Close";

	.@choice = select(.@menu$);
	.@idx = 1;

	// Hunt Information
	if (.@choice == .@idx) {
		mes "[Monster Hunter Board]";
		mes "All featured monsters have";
		mes "^00AA00Bounty Board quests^000000!";
		mes " ";
		mes "Hunt the featured monster for";
		mes "^FF00003x EXP and 3x Drops^000000, then";
		mes "complete your bounty for bonus!";
		close;
	}
	.@idx++;

	// Reshuffle (if enabled)
	if (.@has_reshuffle) {
		if (.@choice == .@idx) goto L_Reshuffle;
		.@idx++;
	}

	// Hall of Fame
	if (.@choice == .@idx) goto L_HallOfFame;
	.@idx++;

	// GM Menu (if GM)
	if (getgmlevel() >= 80 && .@choice == .@idx) goto L_GMMenu;
	.@idx++;

	close;

// ============================================================
// MOTD Reshuffle
// ============================================================
L_Reshuffle:
	// Get current MOTD
	query_sql "SELECT mob_id, mob_name, TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM event_motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@cur_mob_id, .@cur_mob$, .@mins_left;

	if (.@cur_mob_id == 0) {
		mes "[Monster Hunter Board]";
		mes "No Monster of the Day is active";
		mes "right now. Check back later!";
		close;
	}

	// Get reshuffle cost
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'crowdfund_motd_base_cost'", .@base$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'crowdfund_motd_cost_multiplier'", .@mult$;
	.@base_cost = atoi(.@base$);
	.@cost_mult = atoi(.@mult$);

	// Get today's reshuffle count
	query_sql "SELECT reshuffle_count FROM event_motd_reshuffle WHERE reshuffle_date = CURDATE()", .@count;

	// Cap at 14 reshuffles to prevent integer overflow (14th reshuffle = ~819M zeny)
	if (.@count >= 14) {
		mes "[Monster Hunter Board]";
		mes "^FF0000Maximum reshuffles reached!^000000";
		mes " ";
		mes "The monster has been reshuffled";
		mes "14 times today. Please wait";
		mes "until tomorrow to reshuffle again.";
		close;
	}

	// Calculate cost: base * (multiplier/100)^count
	.@cost = .@base_cost;
	for (.@i = 0; .@i < .@count; .@i++) {
		.@cost = .@cost * .@cost_mult / 100;
	}

	mes "[Reshuffle Monster]";
	mes "Current target:";
	mes "^0000FF" + .@cur_mob$ + "^000000";
	mes "Time remaining: " + .@mins_left + " minutes";
	mes " ";
	mes "Reshuffle cost: ^FF0000" + callfunc("F_InsertComma", .@cost) + "z^000000";
	if (.@count > 0) {
		mes "^888888(Reshuffled " + .@count + "x today)^000000";
	}
	mes " ";
	mes "Your zeny: " + callfunc("F_InsertComma", Zeny) + "z";
	next;

	if (select("Pay to Reshuffle:Cancel") == 2) {
		close;
	}

	if (Zeny < .@cost) {
		mes "[Monster Hunter Board]";
		mes "^FF0000You don't have enough zeny!^000000";
		close;
	}

	// Confirm
	mes "[Confirm Reshuffle]";
	mes "Pay ^FF0000" + callfunc("F_InsertComma", .@cost) + "z^000000 to reshuffle?";
	mes " ";
	mes "A new random monster will be";
	mes "selected for the current slot.";
	next;

	if (select("Yes, reshuffle!:Cancel") == 2) {
		close;
	}

	// Deduct zeny
	Zeny -= .@cost;

	// Update daily reshuffle count
	query_sql "INSERT INTO event_motd_reshuffle (reshuffle_date, reshuffle_count, last_reshuffle_by, last_reshuffle_at) VALUES (CURDATE(), 1, " + getcharid(0) + ", NOW()) ON DUPLICATE KEY UPDATE reshuffle_count = reshuffle_count + 1, last_reshuffle_by = " + getcharid(0) + ", last_reshuffle_at = NOW()";

	// Get current slot ID
	query_sql "SELECT id, tier FROM event_motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@slot_id, .@tier$;

	// Pick a new random monster (different from current)
	query_sql "SELECT mob_id, mob_name FROM event_motd_monsters WHERE enabled = 1 AND mob_id != " + .@cur_mob_id + " ORDER BY RAND() LIMIT 1", .@new_mob_id, .@new_mob_name$;

	if (.@new_mob_id == 0) {
		// Fallback: just pick any
		query_sql "SELECT mob_id, mob_name FROM event_motd_monsters WHERE enabled = 1 ORDER BY RAND() LIMIT 1", .@new_mob_id, .@new_mob_name$;
	}

	// Log the reshuffle for leaderboard
	query_sql "INSERT INTO event_motd_reshuffle_log (char_id, char_name, cost_paid, old_mob_id, old_mob_name, new_mob_id, new_mob_name) VALUES (" + getcharid(0) + ", '" + escape_sql(strcharinfo(0)) + "', " + .@cost + ", " + .@cur_mob_id + ", '" + escape_sql(.@cur_mob$) + "', " + .@new_mob_id + ", '" + escape_sql(.@new_mob_name$) + "')";

	// Update totals for leaderboard
	query_sql "INSERT INTO event_motd_reshuffle_totals (char_id, char_name, total_spent, reshuffle_count) VALUES (" + getcharid(0) + ", '" + escape_sql(strcharinfo(0)) + "', " + .@cost + ", 1) ON DUPLICATE KEY UPDATE total_spent = total_spent + " + .@cost + ", reshuffle_count = reshuffle_count + 1, char_name = '" + escape_sql(strcharinfo(0)) + "', last_reshuffle = NOW()";

	// Update the schedule
	query_sql "UPDATE event_motd_schedule SET mob_id = " + .@new_mob_id + ", mob_name = '" + escape_sql(.@new_mob_name$) + "' WHERE id = " + .@slot_id;

	// Update global variables
	set $@LastMOTDMobID, .@new_mob_id;
	set $@CurrentMOTDName$, .@new_mob_name$;

	// Announce
	announce "[Monster Hunter] " + strcharinfo(0) + " reshuffled the target! New Monster of the Day: " + .@new_mob_name$ + " - 3x EXP & Drops for " + .@mins_left + " minutes!", bc_all, 0x00FF00;

	mes "[Monster Hunter Board]";
	mes "^00FF00Monster reshuffled!^000000";
	mes " ";
	mes "New Monster of the Day:";
	mes "^0000FF" + .@new_mob_name$ + "^000000";
	mes " ";
	mes "Happy hunting!";
	close;

// ============================================================
// Reshuffle Hall of Fame (Leaderboard)
// ============================================================
L_HallOfFame:
	mes "[Reshuffle Hall of Fame]";
	mes "Honoring hunters who shaped";
	mes "the daily hunt!";
	next;

	switch(select("Top Spenders (All-Time):Most Reshuffles (All-Time):Recent Reshuffles:Back")) {
		case 1:
			mes "[Top Spenders]";
			mes "Most zeny spent on reshuffles:";
			mes " ";

			query_sql "SELECT char_name, total_spent, reshuffle_count FROM event_motd_reshuffle_totals ORDER BY total_spent DESC LIMIT 10", .@names$, .@spent, .@counts;

			if (getarraysize(.@names$) == 0) {
				mes "^888888No reshuffles recorded yet.^000000";
			} else {
				for (.@i = 0; .@i < getarraysize(.@names$); .@i++) {
					if (.@i == 0) {
						mes "^FFD700#1 " + .@names$[.@i] + "^000000";
					} else if (.@i == 1) {
						mes "^C0C0C0#2 " + .@names$[.@i] + "^000000";
					} else if (.@i == 2) {
						mes "^CD7F32#3 " + .@names$[.@i] + "^000000";
					} else {
						mes "#" + (.@i+1) + " " + .@names$[.@i];
					}
					mes "   " + callfunc("F_InsertComma", .@spent[.@i]) + "z (" + .@counts[.@i] + " reshuffles)";
				}
			}
			close;

		case 2:
			mes "[Most Active Reshufflers]";
			mes "Most reshuffles performed:";
			mes " ";

			query_sql "SELECT char_name, reshuffle_count, total_spent FROM event_motd_reshuffle_totals ORDER BY reshuffle_count DESC LIMIT 10", .@names$, .@counts, .@spent;

			if (getarraysize(.@names$) == 0) {
				mes "^888888No reshuffles recorded yet.^000000";
			} else {
				for (.@i = 0; .@i < getarraysize(.@names$); .@i++) {
					if (.@i == 0) {
						mes "^FFD700#1 " + .@names$[.@i] + "^000000";
					} else if (.@i == 1) {
						mes "^C0C0C0#2 " + .@names$[.@i] + "^000000";
					} else if (.@i == 2) {
						mes "^CD7F32#3 " + .@names$[.@i] + "^000000";
					} else {
						mes "#" + (.@i+1) + " " + .@names$[.@i];
					}
					mes "   " + .@counts[.@i] + " reshuffles (" + callfunc("F_InsertComma", .@spent[.@i]) + "z total)";
				}
			}
			close;

		case 3:
			mes "[Recent Reshuffles]";
			mes "Latest monster changes:";
			mes " ";

			query_sql "SELECT char_name, old_mob_name, new_mob_name, DATE_FORMAT(reshuffled_at, '%m/%d %H:%i') FROM event_motd_reshuffle_log ORDER BY reshuffled_at DESC LIMIT 10", .@names$, .@old$, .@new$, .@times$;

			if (getarraysize(.@names$) == 0) {
				mes "^888888No reshuffles recorded yet.^000000";
			} else {
				for (.@i = 0; .@i < getarraysize(.@names$); .@i++) {
					mes "^0055FF" + .@names$[.@i] + "^000000 [" + .@times$[.@i] + "]";
					mes "  " + .@old$[.@i] + " -> " + .@new$[.@i];
				}
			}
			close;

		case 4:
			close;
	}
	end;

// ============================================================
// GM MENU - MOTD Management
// ============================================================
L_GMMenu:
	mes "[Monster Hunter Board]";
	mes "^FF8800[GM Mode]^000000";
	mes " ";

	// Show current status
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@motd_enabled$;
	query_sql "SELECT mob_name, TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM event_motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@cur_mob$, .@cur_mins;

	mes "MOTD Status: " + ((.@motd_enabled$ == "1") ? "^00AA00Enabled^000000" : "^FF0000Disabled^000000");
	if (.@cur_mob$ != "") {
		mes "Current: ^0055FF" + .@cur_mob$ + "^000000";
		mes "Time left: " + .@cur_mins + " min";
	} else {
		mes "^888888No monster currently active^000000";
	}
	next;

	switch(select("Change Current Monster:Regenerate Schedule:Configure:About MOTD:Back")) {
		case 1:
			goto L_GM_ChangeMonster;
		case 2:
			goto L_GM_Regenerate;
		case 3:
			goto L_GM_Config;
		case 4:
			goto L_GM_About;
		case 5:
			close;
	}
	end;

L_GM_ChangeMonster:
	mes "[Change Current Monster]";
	mes "This will replace the current";
	mes "featured monster with a new one.";
	mes " ";
	mes "Select source:";
	next;

	switch(select("Random from Pool:Specific Mob ID:Cancel")) {
		case 1:
			// Get random monster from pool
			query_sql "SELECT mob_id, mob_name FROM event_motd_monsters ORDER BY RAND() LIMIT 1", .@new_mob_id, .@new_mob_name$;
			if (.@new_mob_id == 0) {
				mes "Monster pool is empty!";
				close;
			}

			// Update current schedule entry
			query_sql "UPDATE event_motd_schedule SET mob_id = " + .@new_mob_id + ", mob_name = '" + escape_sql(.@new_mob_name$) + "' WHERE NOW() BETWEEN start_time AND end_time";

			mes "[Monster Hunter Board]";
			mes "Monster changed to:";
			mes "^0055FF" + .@new_mob_name$ + "^000000";

			// Announce
			announce "[MOTD] Featured monster changed: " + .@new_mob_name$ + "!", bc_all, 0xFFD700;
			close;

		case 2:
			mes "Enter Mob ID:";
			input .@mob_id;

			// Validate mob exists
			.@mob_name$ = strmobinfo(1, .@mob_id);
			if (.@mob_name$ == "") {
				mes "Invalid Mob ID!";
				close;
			}

			// Update current schedule entry
			query_sql "UPDATE event_motd_schedule SET mob_id = " + .@mob_id + ", mob_name = '" + escape_sql(.@mob_name$) + "' WHERE NOW() BETWEEN start_time AND end_time";

			mes "[Monster Hunter Board]";
			mes "Monster changed to:";
			mes "^0055FF" + .@mob_name$ + "^000000";

			announce "[MOTD] Featured monster changed: " + .@mob_name$ + "!", bc_all, 0xFFD700;
			close;

		case 3:
			goto L_GMMenu;
	}
	end;

L_GM_Regenerate:
	mes "[Regenerate Schedule]";
	mes "This will regenerate the MOTD";
	mes "schedule for today and tomorrow.";
	mes " ";
	mes "^FF0000Current schedule will be lost!^000000";
	next;

	if (select("Regenerate:Cancel") == 2) goto L_GMMenu;

	// Call the scheduler to regenerate
	donpcevent "EventScheduler::OnGenerateMOTDSchedule";
	mes "[Monster Hunter Board]";
	mes "Schedule regenerated!";
	mes "Check upcoming monsters.";
	close;

L_GM_Config:
	mes "[Configuration]";

	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@enabled$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_duration_hours'", .@duration$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_upcoming_count'", .@upcoming$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_selection_mode'", .@mode$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_min_level'", .@min_lv$;

	mes "MOTD: " + ((.@enabled$ == "1") ? "^00AA00Enabled^000000" : "^FF0000Disabled^000000");
	mes "Duration: ^0055FF" + .@duration$ + " hours^000000 per slot";
	mes "Upcoming: ^0055FF" + .@upcoming$ + "^000000 shown";
	mes "Mode: ^0055FF" + .@mode$ + "^000000";
	mes "Min Level: ^0055FF" + .@min_lv$ + "^000000";
	next;

	// Dynamic menu text for enable/disable
	.@toggle_text$ = (.@enabled$ == "1") ? "Disable MOTD" : "Enable MOTD";
	switch(select(.@toggle_text$ + ":Set Duration:Set Upcoming Count:Set Selection Mode:Set Min Level:Back")) {
		case 1:
			.@new_val$ = (.@enabled$ == "1") ? "0" : "1";
			// Update both event_config and event_types (scheduler checks event_types)
			query_sql "UPDATE event_config SET config_value = '" + .@new_val$ + "' WHERE config_key = 'motd_enabled'";
			query_sql "UPDATE event_types SET enabled = " + .@new_val$ + " WHERE code = 'motd'";

			if (.@new_val$ == "1") {
				// Enabling - generate schedule and activate
				mes "[Monster Hunter Board]";
				mes "^00AA00MOTD Enabled!^000000";
				mes " ";
				mes "Generating schedule...";
				// Reset tracking so the first monster gets announced
				set $@LastMOTDMobID, 0;
				announce "[MOTD] Monster of the Day has been enabled!", bc_all, 0x00FF00;
				donpcevent "EventScheduler::OnGenerateMOTDSchedule";
				mes "Schedule generated!";
				mes "Monster rotation is now active.";
			} else {
				// Disabling - announce cancellation
				mes "[Monster Hunter Board]";
				mes "^FF0000MOTD Disabled!^000000";
				mes " ";
				mes "Monster of the Day has been";
				mes "cancelled until re-enabled.";
				announce "[MOTD] Monster of the Day has been disabled.", bc_all, 0xFF6600;
			}
			close;
		case 2:
			mes "Enter duration (1, 2, 3, 4, or 6 hours):";
			input .@val;
			if (.@val != 1 && .@val != 2 && .@val != 3 && .@val != 4 && .@val != 6) {
				mes "Invalid value!";
				close;
			}
			query_sql "UPDATE event_config SET config_value = '" + .@val + "' WHERE config_key = 'motd_duration_hours'";
			mes "[Monster Hunter Board]";
			mes "Duration set to " + .@val + " hours.";
			close;

		case 3:
			mes "Enter upcoming count (10-50):";
			input .@val;
			if (.@val < 10) .@val = 10;
			if (.@val > 50) .@val = 50;
			query_sql "UPDATE event_config SET config_value = '" + .@val + "' WHERE config_key = 'motd_upcoming_count'";
			mes "[Monster Hunter Board]";
			mes "Upcoming count set to " + .@val + ".";
			close;

		case 4:
			mes "Select mode:";
			next;
			switch(select("random:tier_rotation:Cancel")) {
				case 1:
					query_sql "UPDATE event_config SET config_value = 'random' WHERE config_key = 'motd_selection_mode'";
					mes "Mode set to random.";
					close;
				case 2:
					query_sql "UPDATE event_config SET config_value = 'tier_rotation' WHERE config_key = 'motd_selection_mode'";
					mes "Mode set to tier_rotation.";
					close;
				case 3:
					goto L_GM_Config;
			}
			end;

		case 5:
			mes "Enter minimum level (1-99):";
			input .@val;
			if (.@val < 1) .@val = 1;
			if (.@val > 99) .@val = 99;
			query_sql "UPDATE event_config SET config_value = '" + .@val + "' WHERE config_key = 'motd_min_level'";
			mes "[Monster Hunter Board]";
			mes "Min level set to " + .@val + ".";
			close;

		case 6:
			goto L_GMMenu;
	}
	end;

L_GM_About:
	mes "[About MOTD]";
	mes "Monster of the Day features";
	mes "rotating monsters with 3x bonuses.";
	mes " ";
	mes "The schedule auto-generates daily";
	mes "and rotates based on configured";
	mes "duration (default 2 hours).";
	mes " ";
	mes "Monsters are drawn from the";
	mes "Bounty Board pool (level 50+).";
	close;
