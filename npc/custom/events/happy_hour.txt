//===== rAthena Script =======================================
//= Happy Hour Event
//= 2x EXP rates for a scheduled duration
//===== Description: =========================================
//= EXP-only bonus event (no drop rate changes to protect economy)
//= - Triggered by scheduler or manually by GM
//= - Doubles base and job EXP rates
//= - Announces on start, every 30 min, and on end
//============================================================

-	script	HappyHour	FAKE_NPC,{
	end;

OnInit:
	// Don't reset if already restored by scheduler
	// (scheduler restore runs before this OnInit)
	end;

// ============================================================
// Event Start - Called by scheduler or GM
// ============================================================
OnEventStart:
	if ($@HappyHourActive) {
		debugmes "[HappyHour] Already active, ignoring start request";
		end;
	}

	// Get duration (default 120 minutes if not set)
	if ($@HappyHourDuration <= 0) set $@HappyHourDuration, 120;

	// Store original rates
	set $@HH_OrigBaseExp, getbattleflag("base_exp_rate");
	set $@HH_OrigJobExp, getbattleflag("job_exp_rate");

	// Double EXP rates (no drop changes!)
	setbattleflag("base_exp_rate", $@HH_OrigBaseExp * 2);
	setbattleflag("job_exp_rate", $@HH_OrigJobExp * 2, true);  // true = reload

	// Mark as active
	set $@HappyHourActive, 1;
	set $@HappyHourEndTime, gettimetick(2) + ($@HappyHourDuration * 60);
	set $@HappyHourStartTime, gettimetick(2);

	// Note: event_instances is created by EventScheduler when it triggers this event
	// $@HappyHourInstanceId is set by scheduler before calling OnEventStart

	// Announce
	announce "[Happy Hour] 2x EXP rates activated for "+$@HappyHourDuration+" minutes!", bc_all, 0xFFD700;

	// Set timer for periodic checks (every 60 seconds to check end time)
	initnpctimer;

	debugmes "[HappyHour] Started - Duration: "+$@HappyHourDuration+" minutes";
	end;

// ============================================================
// Check End Time (every 60 seconds)
// ============================================================
OnTimer60000:
	if (!$@HappyHourActive) {
		stopnpctimer;
		end;
	}

	// Check if event should end
	if (gettimetick(2) >= $@HappyHourEndTime) {
		goto OnEventEnd;
	}

	// Continue timer
	initnpctimer;
	end;

// ============================================================
// Periodic Reminder (every 30 minutes from start)
// ============================================================
OnTimer1800000:  // 30 minutes
OnTimer3600000:  // 60 minutes
OnTimer5400000:  // 90 minutes
OnTimer7200000:  // 120 minutes
	if (!$@HappyHourActive) {
		end;
	}

	// Calculate remaining time
	set .@remaining, ($@HappyHourEndTime - gettimetick(2)) / 60;

	if (.@remaining > 5) {
		announce "[Happy Hour] "+.@remaining+" minutes remaining! 2x EXP rates!", bc_all, 0xFFD700;
	}
	end;

// ============================================================
// Event End
// ============================================================
OnEventEnd:
	// Always announce end (even if NPC state was desynced from DB)
	// This ensures players know the event ended regardless of internal state
	announce "[Happy Hour] Event ended - EXP rates returned to normal.", bc_all, 0xFFD700;

	if (!$@HappyHourActive) {
		debugmes "[HappyHour] End called but was not active (announced anyway)";
		end;
	}

	// Restore original rates
	setbattleflag("base_exp_rate", $@HH_OrigBaseExp);
	setbattleflag("job_exp_rate", $@HH_OrigJobExp, true);  // true = reload

	// Mark as inactive
	set $@HappyHourActive, 0;
	set $@HappyHourEndTime, 0;

	// Update instance and cache via helper function
	callfunc "F_EndEventInstance", $@HappyHourInstanceId;
	set $@HappyHourInstanceId, 0;
	set $@HappyHourScheduleId, 0;

	// Stop timer
	stopnpctimer;

	debugmes "[HappyHour] Ended";
	end;

// ============================================================
// Manual Stop (for GM use)
// ============================================================
OnManualStop:
	// Always call OnEventEnd to announce (even if NPC state desynced)
	// OnEventEnd will handle stopping the NPC timer via stopnpctimer
	donpcevent strnpcinfo(0)+"::OnEventEnd";
	end;
}

// ============================================================
// Helper function to check if Happy Hour is active
// ============================================================
function	script	F_IsHappyHourActive	{
	return $@HappyHourActive;
}

// ============================================================
// Helper function to get Happy Hour remaining time (minutes)
// ============================================================
function	script	F_HappyHourRemaining	{
	if (!$@HappyHourActive) return 0;
	return ($@HappyHourEndTime - gettimetick(2)) / 60;
}
