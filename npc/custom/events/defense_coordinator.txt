//===== rAthena Script =======================================
//= Defense Coordinator NPC
//= Allows players to join town invasions and claim rewards
//===== Description: =========================================
//= - Check current invasion status
//= - Warp to active invasion (unlocks Quest Warper!)
//= - Claim pending rewards from past invasions
//= - View invasion history
//============================================================

prontera,149,193,5	script	Defense Coordinator	4_M_JOB_KNIGHT2,{
	mes "[Defense Coordinator]";
	mes "Greetings, adventurer!";
	mes "I coordinate the defense";
	mes "of our cities against";
	mes "monster invasions.";
	next;

	// Check for active invasion
	if ($@InvasionActive) {
		mes "[Defense Coordinator]";
		mes "^FF0000[!] ALERT! [!]^000000";
		mes " ";
		.@threat_name$ = callfunc("GetInvasionThreatName", $@InvasionThreat);
		.@threat_icon$ = callfunc("GetInvasionThreatIcon", $@InvasionThreat);
		mes .@threat_icon$ + " ^FF0000" + .@threat_name$ + "^000000 threat!";
		mes "^0000FF" + $@InvasionThemeName$ + "^000000";
		mes "attacking ^00AA00" + $@InvasionTownName$ + "^000000!";
		mes " ";
		.@remaining = ($@InvasionEndTime - gettimetick(2)) / 60;
		mes "Wave: " + $@InvasionWave + "/" + $@InvasionTotalWaves;
		mes "Time left: ^FF8800" + .@remaining + " minutes^000000";
		mes "Kills: " + $@InvasionMonstersKilled + "/" + $@InvasionMonstersSpawned;
		next;
	} else {
		mes "[Defense Coordinator]";
		mes "^00AA00All towns are safe.^000000";
		mes "No active invasions at";
		mes "this time.";
		next;
	}

	// Check for pending rewards (exclude active invasion)
	// Points are stored as 10x for decimal precision, divide by 10 for display
	.@active_id = $@InvasionID;  // 0 if no active invasion
	query_sql "SELECT SUM(points_earned) FROM event_invasion_contributions WHERE char_id = " + getcharid(0) + " AND claimed = 0 AND invasion_id != " + .@active_id, .@pending_raw;
	.@pending_points = .@pending_raw / 10;  // Convert from 10x storage to actual points

	// Build menu
	.@menu$ = "";
	if ($@InvasionActive) {
		.@menu$ = "^FF0000[!] Warp to Invasion^000000:";
	}
	if (.@pending_points > 0) {
		.@menu$ = .@menu$ + "^00AA00[!] Claim Rewards (" + .@pending_points + " pts)^000000:";
	}
	.@menu$ = .@menu$ + "View Invasion Status:View Recent Invasions:About Invasions";
	// Add GM menu if authorized
	if (getgmlevel() >= 80) {
		.@menu$ = .@menu$ + ":^FF8800[GM] Manage Invasions^000000";
	}
	.@menu$ = .@menu$ + ":Leave";

	mes "[Defense Coordinator]";
	mes "What would you like to do?";
	next;

	.@choice = select(.@menu$);

	// Adjust choice based on menu availability
	.@idx = 1;
	if ($@InvasionActive) {
		if (.@choice == .@idx) goto L_WarpToInvasion;
		.@idx++;
	}
	if (.@pending_points > 0) {
		if (.@choice == .@idx) goto L_ClaimRewards;
		.@idx++;
	}
	if (.@choice == .@idx) goto L_ViewStatus;
	.@idx++;
	if (.@choice == .@idx) goto L_ViewHistory;
	.@idx++;
	if (.@choice == .@idx) goto L_About;
	.@idx++;
	if (getgmlevel() >= 80 && .@choice == .@idx) goto L_GMMenu;
	close;

// ============================================================
// WARP TO INVASION
// ============================================================
L_WarpToInvasion:
	if (!$@InvasionActive) {
		mes "[Defense Coordinator]";
		mes "The invasion has ended.";
		close;
	}

	mes "[Defense Coordinator]";
	mes "I'll send you directly to";
	mes "^00AA00" + $@InvasionTownName$ + "^000000 to help defend!";
	mes " ";
	mes "^888888No EXP loss while the";
	mes "invasion is active.^000000";
	next;

	// Check if player already has this town unlocked in Quest Warper
	.@qw_index = 0;
	query_sql "SELECT qw_index FROM event_invasion_towns WHERE map_name = '" + escape_sql($@InvasionMap$) + "'", .@qw_index;

	// Check Quest Warper unlock status (using the QWS_T_getbin logic)
	// ##QWS_T_Unlock is an account variable storing a bitmask
	.@unlocked = 0;
	.@bit = 1;
	if (.@qw_index > 0) {
		for (.@i = 0; .@i < .@qw_index; .@i++) {
			.@bit = .@bit * 2;
		}
	}
	if ((##QWS_T_Unlock & .@bit) == .@bit) {
		.@unlocked = 1;
	}

	// Unlock if not already unlocked
	if (!.@unlocked && .@qw_index >= 0) {
		##QWS_T_Unlock = ##QWS_T_Unlock | .@bit;
		mes "[Defense Coordinator]";
		mes "^00FF00" + $@InvasionTownName$ + " unlocked";
		mes "for your Warpra!^000000";
		mes " ";
		mes "You can now warp here";
		mes "anytime using the Warpra.";
		next;
	}

	mes "[Defense Coordinator]";
	mes "For the kingdom!";
	close2;
	warp $@InvasionMap$, $@InvasionSafeX, $@InvasionSafeY;
	end;

// ============================================================
// CLAIM REWARDS
// ============================================================
L_ClaimRewards:
	// Get unclaimed rewards from COMPLETED invasions only
	// Exclude active invasion (can only claim after it ends)
	.@active_id = $@InvasionID;  // 0 if no active invasion

	query_sql "SELECT id, points_earned FROM event_invasion_contributions WHERE char_id = " + getcharid(0) + " AND claimed = 0 AND invasion_id != " + .@active_id, .@ids, .@points;

	if (getarraysize(.@ids) == 0) {
		mes "[Defense Coordinator]";
		if (.@active_id > 0) {
			// Check if player has contribution in active invasion
			query_sql "SELECT points_earned FROM event_invasion_contributions WHERE char_id = " + getcharid(0) + " AND invasion_id = " + .@active_id, .@active_pts;
			if (.@active_pts > 0) {
				.@display_pts = .@active_pts / 10;
				.@display_dec = .@active_pts % 10;
				mes "You have ^FFAA00" + .@display_pts + "." + .@display_dec + "^000000 pts pending";
				mes "from the current invasion.";
				mes " ";
				mes "Come back after it ends";
				mes "to claim your rewards!";
			} else {
				mes "You have no pending rewards.";
			}
		} else {
			mes "You have no pending rewards.";
		}
		close;
	}

	// Sum all points and mark as claimed
	// Points are stored as 10x for decimal precision
	.@total_raw = 0;
	for (.@i = 0; .@i < getarraysize(.@ids); .@i++) {
		.@total_raw += .@points[.@i];
		query_sql "UPDATE event_invasion_contributions SET claimed = 1 WHERE id = " + .@ids[.@i];
	}

	// Convert from 10x storage to actual points
	.@total = .@total_raw / 10;

	#HUNT_POINTS += .@total;

	mes "[Defense Coordinator]";
	mes "Rewards claimed!";
	mes " ";
	mes "^00FF00+" + .@total + " Hunt Points^000000";
	mes " ";
	mes "Thank you for defending";
	mes "our cities!";
	close;

// ============================================================
// VIEW STATUS
// ============================================================
L_ViewStatus:
	mes "[Defense Coordinator]";

	if ($@InvasionActive) {
		.@threat_name$ = callfunc("GetInvasionThreatName", $@InvasionThreat);
		.@threat_icon$ = callfunc("GetInvasionThreatIcon", $@InvasionThreat);
		.@remaining = ($@InvasionEndTime - gettimetick(2)) / 60;

		mes "^FF0000INVASION ACTIVE^000000";
		mes "---------------------";
		mes "Threat: " + .@threat_icon$ + " " + .@threat_name$;
		mes "Theme: " + $@InvasionThemeName$;
		mes "Location: " + $@InvasionTownName$;
		mes "Wave: " + $@InvasionWave + "/" + $@InvasionTotalWaves;
		mes "Time: " + .@remaining + " min remaining";
		mes "Kills: " + $@InvasionMonstersKilled;
	} else {
		mes "^00AA00NO ACTIVE INVASION^000000";
		mes "---------------------";

		// Check cooldown
		query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_last_ended'", .@last$;
		query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_cooldown_minutes'", .@cooldown$;
		.@last = (.@last$ != "") ? atoi(.@last$) : 0;
		.@cooldown = (.@cooldown$ != "") ? atoi(.@cooldown$) : 120;

		if (.@last > 0) {
			.@elapsed = (gettimetick(2) - .@last) / 60;
			.@until_ready = .@cooldown - .@elapsed;
			if (.@until_ready > 0) {
				mes "Cooldown: " + .@until_ready + " min";
			} else {
				mes "Status: ^00AA00Ready^000000";
			}
		}
		mes " ";
		mes "Invasions occur randomly";
		mes "throughout the day when";
		mes "players are online.";
	}
	close;

// ============================================================
// VIEW HISTORY
// ============================================================
L_ViewHistory:
	mes "[Defense Coordinator]";
	mes "^0000FFRecent Invasions^000000";
	mes "---------------------";

	query_sql "SELECT t.display_name, th.name, i.threat_level, i.status, i.monsters_killed, DATE_FORMAT(i.started_at, '%m/%d %H:%i') FROM event_invasion_instances i JOIN event_invasion_towns t ON i.town_id = t.id JOIN event_invasion_themes th ON i.theme_id = th.id ORDER BY i.started_at DESC LIMIT 5", .@towns$, .@themes$, .@threats, .@statuses$, .@kills, .@dates$;

	if (getarraysize(.@towns$) == 0) {
		mes "No invasion records yet.";
		close;
	}

	for (.@i = 0; .@i < getarraysize(.@towns$); .@i++) {
		.@threat_icon$ = callfunc("GetInvasionThreatIcon", .@threats[.@i]);
		.@status_color$ = (.@statuses$[.@i] == "victory") ? "00FF00" : "FF0000";
		.@status_text$ = (.@statuses$[.@i] == "victory") ? "WIN" : (.@statuses$[.@i] == "failed") ? "FAIL" : "---";

		mes .@threat_icon$ + " " + .@towns$[.@i];
		mes "  " + .@themes$[.@i] + " [^" + .@status_color$ + .@status_text$ + "^000000]";
		mes "  " + .@kills[.@i] + " kills - " + .@dates$[.@i];
		if (.@i < getarraysize(.@towns$) - 1) {
			mes " ";
		}
	}

	next;
	mes "[Defense Coordinator]";
	mes "Your contribution:";

	// Get player's stats
	query_sql "SELECT COUNT(*), SUM(kills), SUM(points_earned) FROM event_invasion_contributions WHERE char_id = " + getcharid(0), .@count, .@total_kills, .@total_points;

	mes "Invasions joined: " + .@count;
	mes "Total kills: " + .@total_kills;
	mes "Total points earned: " + .@total_points;
	close;

// ============================================================
// ABOUT INVASIONS
// ============================================================
L_About:
	mes "[Defense Coordinator]";
	mes "^0000FFTown Invasions^000000";
	mes "---------------------";
	mes "Monsters periodically";
	mes "invade our cities! Help";
	mes "defend to earn rewards.";
	next;

	mes "[Defense Coordinator]";
	mes "^0000FFThreat Levels^000000";
	mes "---------------------";
	mes "WOLF - Easy (weak mobs)";
	mes "TIGER - Medium";
	mes "DEMON - Hard (elites)";
	mes "DRAGON - Very Hard (MVPs)";
	mes "GOD - Extreme (GM only)";
	next;

	mes "[Defense Coordinator]";
	mes "^0000FFRewards^000000";
	mes "---------------------";
	mes "- Points earned per kill";
	mes "- +10 pts per wave cleared";
	mes "- ^00FF00x1.5 bonus for victory!^000000";
	mes "- Claim rewards after invasion";
	mes "- No EXP loss during invasion";
	next;

	mes "[Defense Coordinator]";
	mes "^0000FFTip^000000";
	mes "---------------------";
	mes "Warping to an invasion";
	mes "also ^00AA00unlocks that town^000000";
	mes "for your Warpra NPC!";
	mes " ";
	mes "Great way to explore!";
	close;

// ============================================================
// GM MENU - Invasion Management
// ============================================================
L_GMMenu:
	mes "[Defense Coordinator]";
	mes "^FF8800[GM Mode]^000000";
	mes " ";

	// Show current status
	if ($@InvasionActive) {
		.@threat_name$ = callfunc("GetInvasionThreatName", $@InvasionThreat);
		.@threat_icon$ = callfunc("GetInvasionThreatIcon", $@InvasionThreat);
		.@remaining = ($@InvasionEndTime - gettimetick(2)) / 60;

		mes "^FF0000INVASION ACTIVE^000000";
		mes .@threat_icon$ + " " + .@threat_name$ + " - " + $@InvasionThemeName$;
		mes "Location: " + $@InvasionTownName$;
		mes "Wave: " + $@InvasionWave + "/" + $@InvasionTotalWaves;
		mes "Time left: " + .@remaining + " min";
		mes "Kills: " + $@InvasionMonstersKilled;
	} else {
		mes "^00AA00Status: No active invasion^000000";

		// Check cooldown
		query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_last_ended'", .@last$;
		query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_cooldown_minutes'", .@cooldown$;
		.@last = (.@last$ != "") ? atoi(.@last$) : 0;
		.@cooldown = (.@cooldown$ != "") ? atoi(.@cooldown$) : 120;

		if (.@last > 0) {
			.@elapsed = (gettimetick(2) - .@last) / 60;
			.@until_ready = .@cooldown - .@elapsed;
			if (.@until_ready > 0) {
				mes "Cooldown: " + .@until_ready + " min remaining";
			} else {
				mes "Cooldown: ^00AA00Ready^000000";
			}
		}
	}
	next;

	// Consistent menu structure
	switch(select("Start Event:Stop Event:View Schedules:Configure:Manage Triggers:Invasion History:Back")) {
		case 1: goto L_GM_StartMenu;
		case 2: goto L_GM_StopMenu;
		case 3: goto L_GM_ViewSched;
		case 4: goto L_GM_Config;
		case 5: goto L_GM_Triggers;
		case 6: goto L_GM_History;
		case 7: close;
	}
	end;

// ============================================================
// GM: Start Event Submenu
// ============================================================
L_GM_StartMenu:
	if ($@InvasionActive) {
		mes "[Start Invasion]";
		mes "^FF0000An invasion is already active!^000000";
		mes " ";
		mes "Wait for it to end or cancel it";
		mes "before starting a new one.";
		close;
	}

	mes "[Start Invasion]";
	mes "How would you like to start?";
	next;

	switch(select("Now:Schedule (One-Time):Recurring (Weekly):Back")) {
		case 1: goto L_GM_Start;
		case 2: goto L_GM_SchedOneTime;
		case 3: goto L_GM_SchedRecur;
		case 4: goto L_GMMenu;
	}
	end;

// ============================================================
// GM: Stop Event Submenu
// ============================================================
L_GM_StopMenu:
	if (!$@InvasionActive) {
		mes "[Stop Invasion]";
		mes "^888888No invasion is currently active.^000000";
		close;
	}

	mes "[Active Invasion]";
	.@threat_name$ = callfunc("GetInvasionThreatName", $@InvasionThreat);
	.@remaining = ($@InvasionEndTime - gettimetick(2)) / 60;
	mes "Town: " + $@InvasionTownName$;
	mes "Theme: " + $@InvasionThemeName$;
	mes "Threat: " + .@threat_name$;
	mes "Wave: " + $@InvasionWave + "/" + $@InvasionTotalWaves;
	mes "Time: " + .@remaining + " min left";
	mes "Kills: " + $@InvasionMonstersKilled + "/" + $@InvasionMonstersSpawned;
	next;

	switch(select("Force Wave Clear:Cancel Invasion:Back")) {
		case 1: goto L_GM_ForceWave;
		case 2: goto L_GM_Cancel;
		case 3: goto L_GMMenu;
	}
	end;

L_GM_Details:
	mes "[Invasion Details]";
	mes "ID: " + $@InvasionID;
	mes "Map: " + $@InvasionMap$;
	mes "Theme ID: " + $@InvasionTheme;
	mes "Threat: " + $@InvasionThreat;
	mes "Spawned: " + $@InvasionMonstersSpawned;
	mes "Killed: " + $@InvasionMonstersKilled;

	query_sql "SELECT COUNT(*), SUM(kills) FROM event_invasion_contributions WHERE invasion_id = " + $@InvasionID, .@defenders, .@total_kills;
	mes "Defenders: " + .@defenders;
	mes "Total kills: " + .@total_kills;
	close;

L_GM_Cancel:
	mes "[Cancel Invasion]";
	mes "Are you sure you want to";
	mes "cancel the current invasion?";
	mes " ";
	mes "^FF0000Players will NOT receive";
	mes "rewards for this invasion.^000000";
	next;

	if (select("Yes, cancel:No, keep running") == 1) {
		donpcevent "TownInvasion::OnGMCancelInvasion";
		mes "[Defense Coordinator]";
		mes "Invasion cancelled.";
	} else {
		mes "[Defense Coordinator]";
		mes "Invasion continues.";
	}
	close;

L_GM_ForceWave:
	mes "[Force Wave Clear]";
	mes "Current wave: " + $@InvasionWave + "/" + $@InvasionTotalWaves;
	mes "Wave kills: " + $@InvasionWaveKills + "/" + $@InvasionWaveMonsters;
	mes " ";
	mes "This will immediately clear";
	mes "the current wave.";
	next;

	if (select("Clear Wave Now:Cancel") == 1) {
		mes "[Defense Coordinator]";
		mes "Wave cleared!";
		donpcevent "TownInvasion::OnWaveCleared";
	} else {
		mes "[Defense Coordinator]";
		mes "Cancelled.";
	}
	close;

L_GM_Start:
	mes "[Start Invasion]";
	mes "Select threat level:";
	next;

	.@threat = select("Wolf (Easy):Tiger (Medium):Demon (Hard):Dragon (Very Hard):God (Extreme):Cancel") - 1;
	if (.@threat == 5) goto L_GMMenu;

	mes "[Start Invasion]";
	mes "Select town:";
	next;

	query_sql "SELECT id, display_name FROM event_invasion_towns WHERE enabled = 1 ORDER BY display_name", .@town_ids, .@town_names$;

	.@menu$ = "";
	for (.@i = 0; .@i < getarraysize(.@town_ids); .@i++) {
		.@menu$ = .@menu$ + .@town_names$[.@i] + ":";
	}
	.@menu$ = .@menu$ + "Cancel";

	.@town_choice = select(.@menu$);
	if (.@town_choice > getarraysize(.@town_ids)) goto L_GMMenu;
	.@town_id = .@town_ids[.@town_choice - 1];

	mes "[Start Invasion]";
	mes "Select theme:";
	next;

	query_sql "SELECT id, name FROM event_invasion_themes ORDER BY id", .@theme_ids, .@theme_names$;

	.@menu$ = "";
	for (.@i = 0; .@i < getarraysize(.@theme_ids); .@i++) {
		.@menu$ = .@menu$ + .@theme_names$[.@i] + ":";
	}
	.@menu$ = .@menu$ + "Random:Cancel";

	.@theme_choice = select(.@menu$);
	if (.@theme_choice > getarraysize(.@theme_ids) + 1) goto L_GMMenu;

	if (.@theme_choice == getarraysize(.@theme_ids) + 1) {
		query_sql "SELECT id FROM event_invasion_themes ORDER BY RAND() LIMIT 1", .@theme_id;
	} else {
		.@theme_id = .@theme_ids[.@theme_choice - 1];
	}

	// Confirm
	query_sql "SELECT display_name FROM event_invasion_towns WHERE id = " + .@town_id, .@town_name$;
	query_sql "SELECT name FROM event_invasion_themes WHERE id = " + .@theme_id, .@theme_name$;
	.@threat_name$ = callfunc("GetInvasionThreatName", .@threat);
	.@threat_icon$ = callfunc("GetInvasionThreatIcon", .@threat);

	mes "[Confirm Invasion]";
	mes "Threat: " + .@threat_icon$ + " " + .@threat_name$;
	mes "Town: " + .@town_name$;
	mes "Theme: " + .@theme_name$;
	next;

	if (select("Start Invasion:Cancel") == 1) {
		query_sql "SELECT id FROM event_types WHERE code = 'invasion'", .@event_type_id;
		query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_duration_minutes'", .@duration$;
		.@duration = (.@duration$ != "") ? atoi(.@duration$) : 45;

		query_sql "INSERT INTO event_instances (event_type_id, event_code, trigger_type, triggered_by, scheduled_start, scheduled_end, status, event_data) VALUES (" + .@event_type_id + ", 'invasion', 'manual', " + getcharid(3) + ", NOW(), DATE_ADD(NOW(), INTERVAL " + .@duration + " MINUTE), 'active', JSON_OBJECT('threat', " + .@threat + ", 'town_id', " + .@town_id + ", 'theme_id', " + .@theme_id + "))";

		$@InvasionPendingThreat = .@threat;
		$@InvasionPendingTown = .@town_id;
		$@InvasionPendingTheme = .@theme_id;
		donpcevent "TownInvasion::OnGMStartInvasion";
		mes "[Defense Coordinator]";
		mes "Invasion started!";
	} else {
		mes "[Defense Coordinator]";
		mes "Cancelled.";
	}
	close;

L_GM_SchedOneTime:
	mes "[Schedule One-Time Invasion]";
	mes "Enter start date/time:";
	mes "^888888Format: YYYY-MM-DD HH:MM^000000";
	input .@start_dt$;
	next;

	if (.@start_dt$ == "") {
		mes "No date entered. Cancelled.";
		close;
	}

	mes "[Schedule Invasion]";
	mes "Select threat level:";
	next;

	.@threat = select("Wolf:Tiger:Demon:Dragon:God:Random:Cancel") - 1;
	if (.@threat == 6) goto L_GMMenu;
	if (.@threat == 5) .@threat = -1;

	mes "[Schedule Invasion]";
	mes "Select town:";
	next;

	query_sql "SELECT id, display_name FROM event_invasion_towns WHERE enabled = 1 ORDER BY display_name", .@town_ids, .@town_names$;
	.@menu$ = "Random:";
	for (.@i = 0; .@i < getarraysize(.@town_ids); .@i++) {
		.@menu$ = .@menu$ + .@town_names$[.@i] + ":";
	}
	.@menu$ = .@menu$ + "Cancel";

	.@town_choice = select(.@menu$);
	if (.@town_choice > getarraysize(.@town_ids) + 1) goto L_GMMenu;
	.@town_id = (.@town_choice == 1) ? 0 : .@town_ids[.@town_choice - 2];

	mes "[Schedule Invasion]";
	mes "Select theme:";
	next;

	query_sql "SELECT id, name FROM event_invasion_themes ORDER BY id", .@theme_ids, .@theme_names$;
	.@menu$ = "Random:";
	for (.@i = 0; .@i < getarraysize(.@theme_ids); .@i++) {
		.@menu$ = .@menu$ + .@theme_names$[.@i] + ":";
	}
	.@menu$ = .@menu$ + "Cancel";

	.@theme_choice = select(.@menu$);
	if (.@theme_choice > getarraysize(.@theme_ids) + 1) goto L_GMMenu;
	.@theme_id = (.@theme_choice == 1) ? 0 : .@theme_ids[.@theme_choice - 2];

	query_sql "SELECT id FROM event_types WHERE code = 'invasion'", .@event_type_id;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_duration_minutes'", .@duration$;
	.@duration = (.@duration$ != "") ? atoi(.@duration$) : 45;

	query_sql "INSERT INTO event_instances (event_type_id, event_code, trigger_type, triggered_by, scheduled_start, scheduled_end, status, event_data) VALUES (" + .@event_type_id + ", 'invasion', 'one_time', " + getcharid(3) + ", '" + .@start_dt$ + "', DATE_ADD('" + .@start_dt$ + "', INTERVAL " + .@duration + " MINUTE), 'scheduled', JSON_OBJECT('threat', " + .@threat + ", 'town_id', " + .@town_id + ", 'theme_id', " + .@theme_id + "))";

	mes "[Defense Coordinator]";
	mes "Invasion scheduled!";
	mes "Start: " + .@start_dt$;
	close;

L_GM_SchedRecur:
	mes "[Recurring Invasion]";
	mes "Select day of week:";
	next;

	.@dow = select("Sunday:Monday:Tuesday:Wednesday:Thursday:Friday:Saturday:Cancel") - 1;
	if (.@dow == 7) goto L_GMMenu;

	mes "[Recurring Invasion]";
	mes "Select hour (server time):";
	next;

	.@hour_sel = select("12 PM:2 PM:4 PM:6 PM:8 PM:10 PM:Custom:Cancel") - 1;
	if (.@hour_sel == 7) goto L_GMMenu;

	switch(.@hour_sel) {
		case 0: .@start_hour = 12; break;
		case 1: .@start_hour = 14; break;
		case 2: .@start_hour = 16; break;
		case 3: .@start_hour = 18; break;
		case 4: .@start_hour = 20; break;
		case 5: .@start_hour = 22; break;
		case 6:
			mes "Enter hour (0-23):";
			input .@start_hour;
			if (.@start_hour < 0 || .@start_hour > 23) {
				mes "Invalid hour.";
				close;
			}
			break;
	}

	mes "[Recurring Invasion]";
	mes "Select threat level:";
	next;

	.@threat = select("Wolf:Tiger:Demon:Random each time:Cancel") - 1;
	if (.@threat == 4) goto L_GMMenu;
	if (.@threat == 3) .@threat = -1;

	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_duration_minutes'", .@duration$;
	.@duration = (.@duration$ != "") ? atoi(.@duration$) : 45;
	query_sql "SELECT id FROM event_types WHERE code = 'invasion'", .@event_type_id;

	explode(.@day_arr$, "Sunday:Monday:Tuesday:Wednesday:Thursday:Friday:Saturday", ":");
	.@day_name$ = .@day_arr$[.@dow];

	query_sql "INSERT INTO event_schedule (event_type_id, event_code, schedule_type, start_hour, start_minute, day_of_week, duration_minutes, event_params, description, enabled) VALUES (" + .@event_type_id + ", 'invasion', 'weekly', " + .@start_hour + ", 0, " + .@dow + ", " + .@duration + ", JSON_OBJECT('threat', " + .@threat + "), 'Weekly Invasion - " + .@day_name$ + "', 1)";

	query_sql "SELECT LAST_INSERT_ID()", .@sched_id;
	if (.@dow == gettime(4) && gettime(3) < .@start_hour) {
		query_sql "UPDATE event_schedule SET next_run = DATE_ADD(CURDATE(), INTERVAL " + .@start_hour + " HOUR) WHERE id = " + .@sched_id;
	} else {
		.@days_until = (.@dow - gettime(4) + 7) % 7;
		if (.@days_until == 0) .@days_until = 7;
		query_sql "UPDATE event_schedule SET next_run = DATE_ADD(DATE_ADD(CURDATE(), INTERVAL " + .@days_until + " DAY), INTERVAL " + .@start_hour + " HOUR) WHERE id = " + .@sched_id;
	}

	mes "[Defense Coordinator]";
	mes "Weekly invasion scheduled!";
	mes "Every " + .@day_name$ + " at " + .@start_hour + ":00";
	close;

L_GM_ViewSched:
	mes "[Scheduled Invasions]";
	mes " ";

	query_sql "SELECT id, day_of_week, start_hour, enabled, next_run FROM event_schedule WHERE event_code = 'invasion' ORDER BY day_of_week, start_hour", .@sched_ids, .@dows, .@hours, .@enabled, .@nexts$;

	if (getarraysize(.@sched_ids) > 0) {
		mes "^0000FFRecurring Schedules:^000000";
		explode(.@day_arr$, "Sun:Mon:Tue:Wed:Thu:Fri:Sat", ":");
		for (.@i = 0; .@i < getarraysize(.@sched_ids); .@i++) {
			.@day$ = .@day_arr$[.@dows[.@i]];
			.@status$ = (.@enabled[.@i]) ? "^00AA00ON^000000" : "^FF0000OFF^000000";
			mes .@day$ + " " + .@hours[.@i] + ":00 [" + .@status$ + "]";
		}
	} else {
		mes "^888888No recurring schedules^000000";
	}

	mes " ";

	query_sql "SELECT id, scheduled_start FROM event_instances WHERE event_type_id = (SELECT id FROM event_types WHERE code = 'invasion') AND status = 'scheduled' AND trigger_type = 'one_time' ORDER BY scheduled_start LIMIT 5", .@ot_ids, .@ot_times$;

	if (getarraysize(.@ot_ids) > 0) {
		mes "^0000FFUpcoming One-Time:^000000";
		for (.@i = 0; .@i < getarraysize(.@ot_ids); .@i++) {
			mes "- " + .@ot_times$[.@i];
		}
	} else {
		mes "^888888No one-time scheduled^000000";
	}

	next;

	switch(select("Delete a Schedule:Back")) {
		case 1: goto L_GM_DeleteSched;
		case 2: goto L_GMMenu;
	}
	end;

L_GM_DeleteSched:
	mes "[Delete Schedule]";
	mes "What type of schedule?";
	next;

	switch(select("Recurring (weekly):One-Time:Cancel")) {
		case 1: goto L_GM_DelRecur;
		case 2: goto L_GM_DelOneTime;
		case 3: goto L_GMMenu;
	}
	end;

L_GM_DelRecur:
	mes "[Delete Recurring]";
	mes "Select schedule to delete:";
	next;

	query_sql "SELECT id, CONCAT(CASE day_of_week WHEN 0 THEN 'Sun' WHEN 1 THEN 'Mon' WHEN 2 THEN 'Tue' WHEN 3 THEN 'Wed' WHEN 4 THEN 'Thu' WHEN 5 THEN 'Fri' WHEN 6 THEN 'Sat' END, ' ', start_hour, ':00') FROM event_schedule WHERE event_code = 'invasion'", .@ids, .@labels$;

	if (getarraysize(.@ids) == 0) {
		mes "No recurring schedules.";
		close;
	}

	.@menu$ = "";
	for (.@i = 0; .@i < getarraysize(.@ids); .@i++) {
		.@menu$ = .@menu$ + .@labels$[.@i] + ":";
	}
	.@menu$ = .@menu$ + "Cancel";

	.@choice = select(.@menu$);
	if (.@choice > getarraysize(.@ids)) goto L_GMMenu;

	query_sql "DELETE FROM event_schedule WHERE id = " + .@ids[.@choice - 1];
	mes "[Defense Coordinator]";
	mes "Recurring schedule deleted.";
	close;

L_GM_DelOneTime:
	mes "[Delete One-Time]";
	mes "Select scheduled invasion:";
	next;

	query_sql "SELECT ei.id, DATE_FORMAT(ei.scheduled_start, '%Y-%m-%d %H:%i') FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'invasion' AND ei.trigger_type = 'one_time' AND ei.status = 'scheduled' ORDER BY ei.scheduled_start", .@ids, .@labels$;

	if (getarraysize(.@ids) == 0) {
		mes "No scheduled invasions.";
		close;
	}

	.@menu$ = "";
	for (.@i = 0; .@i < getarraysize(.@ids); .@i++) {
		.@menu$ = .@menu$ + .@labels$[.@i] + ":";
	}
	.@menu$ = .@menu$ + "Cancel";

	.@choice = select(.@menu$);
	if (.@choice > getarraysize(.@ids)) goto L_GMMenu;

	query_sql "DELETE FROM event_instances WHERE id = " + .@ids[.@choice - 1];
	mes "[Defense Coordinator]";
	mes "Scheduled invasion deleted.";
	close;

L_GM_Config:
	mes "[Invasion Configuration]";

	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_enabled'", .@enabled$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_duration_minutes'", .@duration$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_max_threat'", .@max_threat$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_monster_multiplier'", .@multiplier$;
	if (.@multiplier$ == "") .@multiplier$ = "100";
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_boss_multiplier'", .@boss_mult$;
	if (.@boss_mult$ == "") .@boss_mult$ = "100";

	mes "Auto-Invasion: " + (.@enabled$ == "1" ? "^00AA00Enabled^000000" : "^FF0000Disabled^000000");
	mes "Duration: ^0055FF" + .@duration$ + " min^000000";
	mes "Max Threat: ^0055FF" + .@max_threat$ + "^000000 (0=Wolf, 3=Dragon)";
	mes "Monster mult: ^0000FF" + .@multiplier$ + "%^000000";
	mes "Boss mult: ^FF0000" + .@boss_mult$ + "%^000000";
	mes " ";
	mes "^777777Cooldown, time window, and player";
	mes "requirements are in Triggers menu.^000000";
	next;

	// Dynamic menu text for enable/disable
	.@toggle_text$ = (.@enabled$ == "1") ? "Disable Auto-Invasion" : "Enable Auto-Invasion";
	switch(select(.@toggle_text$ + ":Set Duration:Set Max Threat:Set Monster Mult:Set Boss Mult:Back")) {
		case 1:
			.@new_val$ = (.@enabled$ == "1") ? "0" : "1";
			query_sql "UPDATE event_config SET config_value = '" + .@new_val$ + "' WHERE config_key = 'invasion_enabled'";
			// Also update event_types.enabled
			query_sql "UPDATE event_types SET enabled = " + .@new_val$ + " WHERE code = 'invasion'";

			if (.@new_val$ == "1") {
				mes "[Defense Coordinator]";
				mes "^00AA00Auto-Invasion Enabled!^000000";
				mes " ";
				mes "Invasions will start when";
				mes "trigger conditions are met.";
				mes " ";
				mes "Use ^0055FFTriggers^000000 to configure";
				mes "when invasions can start.";
			} else {
				mes "[Defense Coordinator]";
				mes "^FF0000Auto-Invasion Disabled!^000000";
				mes " ";
				if ($@InvasionActive) {
					mes "^FF8800Cancelling active invasion...^000000";
					donpcevent "TownInvasion::OnGMCancelInvasion";
					mes "Active invasion cancelled.";
				} else {
					mes "No active invasion to cancel.";
				}
				mes " ";
				mes "Auto-invasions will not start";
				mes "until re-enabled.";
			}
			close;
		case 2:
			mes "[Set Duration]";
			mes "Current: " + .@duration$ + " min";
			mes "Enter new duration (15-120):";
			next;
			input .@val;
			if (.@val < 15) .@val = 15;
			if (.@val > 120) .@val = 120;
			query_sql "UPDATE event_config SET config_value = '" + .@val + "' WHERE config_key = 'invasion_duration_minutes'";
			mes "[Defense Coordinator]";
			mes "Duration set to " + .@val + " min.";
			close;
		case 3:
			mes "[Set Max Threat]";
			mes "Current: " + .@max_threat$ + " (0=Wolf, 3=Dragon)";
			mes "Max threat level for random selection.";
			mes "Enter max threat (0-3):";
			next;
			input .@val;
			if (.@val < 0) .@val = 0;
			if (.@val > 3) .@val = 3;
			query_sql "UPDATE event_config SET config_value = '" + .@val + "' WHERE config_key = 'invasion_max_threat'";
			mes "[Defense Coordinator]";
			mes "Max threat set to " + .@val + ".";
			close;
		case 4:
			mes "[Set Monster Multiplier]";
			mes "Current: " + .@multiplier$ + "%";
			mes "^888888Affects weak/medium/strong/elite^000000";
			mes "^888888100 = 1x, 200 = 2x^000000";
			mes "Enter multiplier (10-1000):";
			next;
			input .@val;
			if (.@val < 10) .@val = 10;
			if (.@val > 1000) .@val = 1000;
			query_sql "REPLACE INTO event_config (config_key, config_value) VALUES ('invasion_monster_multiplier', '" + .@val + "')";
			mes "[Defense Coordinator]";
			mes "Monster multiplier set to " + .@val + "%.";
			close;
		case 5:
			mes "[Set Boss Multiplier]";
			mes "Current: " + .@boss_mult$ + "%";
			mes "^888888Affects miniboss/MVP only^000000";
			mes "^888888100 = 1x, 200 = 2x^000000";
			mes "Enter multiplier (10-1000):";
			next;
			input .@val;
			if (.@val < 10) .@val = 10;
			if (.@val > 1000) .@val = 1000;
			query_sql "REPLACE INTO event_config (config_key, config_value) VALUES ('invasion_boss_multiplier', '" + .@val + "')";
			mes "[Defense Coordinator]";
			mes "Boss multiplier set to " + .@val + "%.";
			close;
		case 6:
			goto L_GMMenu;
	}
	end;

L_GM_Triggers:
	callfunc "F_TriggerMenu", "invasion";
	goto L_GMMenu;

L_GM_History:
	mes "[Invasion History]";

	query_sql "SELECT t.display_name, th.name, i.threat_level, i.status, i.monsters_killed, DATE_FORMAT(i.started_at, '%m/%d %H:%i') FROM event_invasion_instances i JOIN event_invasion_towns t ON i.town_id = t.id JOIN event_invasion_themes th ON i.theme_id = th.id ORDER BY i.started_at DESC LIMIT 10", .@towns$, .@themes$, .@threats, .@statuses$, .@kills, .@dates$;

	if (getarraysize(.@towns$) == 0) {
		mes "No invasion records yet.";
		close;
	}

	for (.@i = 0; .@i < getarraysize(.@towns$); .@i++) {
		.@threat_icon$ = callfunc("GetInvasionThreatIcon", .@threats[.@i]);
		.@status_color$ = (.@statuses$[.@i] == "victory") ? "00FF00" : (.@statuses$[.@i] == "failed") ? "FF0000" : "888888";
		mes .@threat_icon$ + " " + .@towns$[.@i] + " - " + .@themes$[.@i];
		mes "   [^" + .@status_color$ + .@statuses$[.@i] + "^000000] " + .@kills[.@i] + " kills - " + .@dates$[.@i];
	}
	close;
}

// ============================================================
// HELPER FUNCTIONS (globally accessible)
// ============================================================
function	script	GetInvasionThreatName	{
	switch(getarg(0)) {
		case 0: return "WOLF";
		case 1: return "TIGER";
		case 2: return "DEMON";
		case 3: return "DRAGON";
		case 4: return "GOD";
	}
	return "UNKNOWN";
}

function	script	GetInvasionThreatIcon	{
	switch(getarg(0)) {
		case 0: return "[WOLF]";
		case 1: return "[TIGER]";
		case 2: return "[DEMON]";
		case 3: return "[DRAGON]";
		case 4: return "[GOD]";
	}
	return "[?]";
}
