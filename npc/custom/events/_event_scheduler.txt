//===== rAthena Script =======================================
//= Event Scheduler
//= Core scheduler that manages all automated events
//===== Description: =========================================
//= Timer-based event scheduler that:
//= - Checks for scheduled events every minute
//= - Triggers event start/end via donpcevent
//= - Manages MOTD daily schedule generation
//============================================================

-	script	EventScheduler	FAKE_NPC,{
	end;

OnInit:
	// Generate MOTD schedule on server start
	donpcevent "EventScheduler::OnGenerateMOTDSchedule";

	// Check for active events immediately (e.g., boost still running after restart)
	callsub S_CheckScheduledEvents;

	// Start the scheduler timer (check every 60 seconds)
	initnpctimer;
	end;

OnTimer60000:
	// Reset timer for next check
	stopnpctimer;
	initnpctimer;

	// Check for scheduled events
	callsub S_CheckScheduledEvents;

	// Check if MOTD needs rotation announcement
	callsub S_CheckMOTDRotation;
	end;

// ============================================================
// Check for scheduled events (Happy Hour, EXP Boost)
// ============================================================
S_CheckScheduledEvents:
	// Check if Happy Hour should start
	query_sql "SELECT id, duration_minutes FROM event_schedule WHERE event_code = 'happy_hour' AND enabled = 1 AND schedule_type = 'fixed' AND day_of_week = WEEKDAY(NOW()) AND start_hour = HOUR(NOW()) AND start_minute = MINUTE(NOW())", .@hh_id, .@hh_duration;

	if (.@hh_id > 0 && !$@HappyHourActive) {
		// Start Happy Hour
		set $@HappyHourDuration, .@hh_duration;
		donpcevent "HappyHour::OnEventStart";
	}

	// Check if scheduled EXP Boost should start
	// Uses exp_boost_schedule table for one-time datetime-based schedules
	debugmes "[Scheduler] Checking EXP Boost - $@ExpBoostActive = "+$@ExpBoostActive;
	if (!$@ExpBoostActive) {
		query_sql "SELECT id, multiplier, TIMESTAMPDIFF(MINUTE, NOW(), end_datetime) FROM exp_boost_schedule WHERE enabled = 1 AND NOW() BETWEEN start_datetime AND end_datetime LIMIT 1", .@eb_id, .@eb_mult, .@eb_remaining;
		debugmes "[Scheduler] Found boost: id="+.@eb_id+", mult="+.@eb_mult+", remaining="+.@eb_remaining;

		if (.@eb_id > 0 && .@eb_remaining > 0) {
			// Start EXP Boost with scheduled multiplier and remaining duration
			debugmes "[Scheduler] Starting EXP Boost via donpcevent";
			set $@ExpBoostMultiplier, .@eb_mult;
			set $@ExpBoostDuration, .@eb_remaining;
			set $@ExpBoostScheduleId, .@eb_id;
			donpcevent "ExpBoost::OnEventStart";
		}
	}
	return;

// ============================================================
// Check MOTD rotation (announce new monster)
// ============================================================
S_CheckMOTDRotation:
	// Check if MOTD is enabled
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@enabled$;
	if (.@enabled$ == "0") {
		// MOTD disabled - clear the active mob ID to stop bonuses
		if ($@LastMOTDMobID > 0) {
			set $@LastMOTDMobID, 0;
			set $@CurrentMOTDName$, "";
		}
		return;
	}

	// Get current MOTD
	query_sql "SELECT mob_id, mob_name FROM motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@mob_id, .@mob_name$;

	// Check if this is a new rotation (different from last announced)
	if (.@mob_id > 0 && .@mob_id != $@LastMOTDMobID) {
		set $@LastMOTDMobID, .@mob_id;
		set $@CurrentMOTDName$, .@mob_name$;

		// Calculate time remaining
		query_sql "SELECT TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM motd_schedule WHERE mob_id = "+.@mob_id+" AND NOW() BETWEEN start_time AND end_time", .@mins_left;

		// Check if this monster has a bounty
		query_sql "SELECT has_bounty FROM motd_monsters WHERE mob_id = "+.@mob_id, .@has_bounty;

		if (.@has_bounty) {
			announce "Monster of the Day: "+.@mob_name$+"! 3x EXP, Drops & Hunt Points for "+.@mins_left+" minutes!", bc_all, 0x00FF00;
		} else {
			announce "Monster of the Day: "+.@mob_name$+"! 3x EXP & Drops for "+.@mins_left+" minutes!", bc_all, 0x00FF00;
		}
	}
	return;

// ============================================================
// Generate MOTD Schedule (called daily at midnight and on init)
// ============================================================
OnClock0000:
OnGenerateMOTDSchedule:
	// Get configuration
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_selection_mode'", .@mode$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_min_level'", .@min_level$;

	if (.@mode$ == "") set .@mode$, "random";
	if (.@min_level$ == "") set .@min_level, 50; else set .@min_level, atoi(.@min_level$);

	// Clear today's schedule
	query_sql "DELETE FROM motd_schedule WHERE DATE(start_time) = CURDATE()";

	// Generate 8 slots (every 3 hours)
	set .@tier_index, 0;
	setarray .@tiers$[0], "medium", "hard";

	for (set .@i, 0; .@i < 8; set .@i, .@i + 1) {
		set .@hour, .@i * 3;

		if (.@mode$ == "tier_rotation") {
			// Alternate between medium and hard
			set .@tier$, .@tiers$[.@tier_index % 2];
			set .@tier_index, .@tier_index + 1;

			// Select random monster from this tier
			query_sql "SELECT mob_id, mob_name FROM motd_monsters WHERE tier = '"+.@tier$+"' AND enabled = 1 AND mob_level >= "+.@min_level+" ORDER BY RAND() LIMIT 1", .@mob_id, .@mob_name$;
		} else {
			// Random mode - pick any monster
			query_sql "SELECT mob_id, mob_name, tier FROM motd_monsters WHERE enabled = 1 AND mob_level >= "+.@min_level+" ORDER BY RAND() LIMIT 1", .@mob_id, .@mob_name$, .@tier$;
		}

		if (.@mob_id > 0) {
			// Insert into schedule
			query_sql "INSERT INTO motd_schedule (mob_id, mob_name, start_time, end_time, tier) VALUES ("+.@mob_id+", '"+escape_sql(.@mob_name$)+"', DATE_ADD(CURDATE(), INTERVAL "+.@hour+" HOUR), DATE_ADD(CURDATE(), INTERVAL "+(.@hour+3)+" HOUR), '"+.@tier$+"')";
		}
	}

	debugmes "[EventScheduler] Generated MOTD schedule for today";
	end;
}

// ============================================================
// Helper function to get current MOTD info
// Called by other NPCs to check if a mob is current MOTD
// ============================================================
function	script	F_GetCurrentMOTD	{
	// Check if MOTD is enabled
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@enabled$;
	if (.@enabled$ == "0") return 0;

	query_sql "SELECT mob_id, mob_name, TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@mob_id, .@mob_name$, .@mins_left;

	// Return mob_id (0 if none active)
	return .@mob_id;
}

// ============================================================
// Helper function to check if a specific mob is current MOTD
// ============================================================
function	script	F_IsMOTD	{
	// Check if MOTD is enabled
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@enabled$;
	if (.@enabled$ == "0") return 0;

	set .@check_mob_id, getarg(0, 0);

	query_sql "SELECT 1 FROM motd_schedule WHERE mob_id = "+.@check_mob_id+" AND NOW() BETWEEN start_time AND end_time LIMIT 1", .@is_motd;

	return .@is_motd;
}

// ============================================================
// Helper function to check if MOTD system is enabled
// ============================================================
function	script	F_IsMOTDEnabled	{
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@enabled$;
	return (.@enabled$ != "0");
}
