//===== rAthena Script =======================================
//= Event Scheduler v3
//= Unified scheduler for all automated events
//===== Description: =========================================
//= Timer-based event scheduler that:
//= - Uses next_run based scheduling (efficient)
//= - Supports: interval, daily, weekly, biweekly, monthly
//= - One-time events are stored directly in event_instances
//= - Uses event_type_id + JOINs (not event_code)
//= - Updates event_current_state cache for hot-path queries
//= - Manages MOTD rotation with event_instances
//============================================================

-	script	EventScheduler	FAKE_NPC,{
	end;

OnInit:
	// Initialize next_run for all enabled schedules
	callsub S_InitializeNextRun;

	// Generate MOTD schedule on server start
	donpcevent "EventScheduler::OnGenerateMOTDSchedule";

	// Restore active events after restart (e.g., boost still running)
	callsub S_RestoreActiveEvents;

	// Start the scheduler timer (check every 60 seconds)
	initnpctimer;
	debugmes "[EventScheduler] Initialized";
	end;

OnTimer60000:
	// Reset timer for next check
	stopnpctimer;
	initnpctimer;

	// Check for events ready to trigger (recurring + one-time)
	callsub S_CheckScheduledEvents;

	// Check if MOTD needs rotation announcement
	callsub S_CheckMOTDRotation;

	// Check conditional triggers (replaces hardcoded S_CheckInvasionTrigger)
	callsub S_CheckCondTriggers;
	end;

// ============================================================
// Initialize next_run for schedules that don't have it set
// ============================================================
S_InitializeNextRun:
	// Get schedules without next_run
	query_sql "SELECT id, schedule_type, interval_minutes, day_of_week, day_of_month, start_hour, start_minute, week_interval, anchor_date FROM event_schedule WHERE enabled = 1 AND next_run IS NULL", .@ids, .@types$, .@intervals, .@dow, .@dom, .@hours, .@mins, .@week_int, .@anchors$;

	for (set .@i, 0; .@i < getarraysize(.@ids); set .@i, .@i + 1) {
		set .@next$, "";

		if (.@types$[.@i] == "interval") {
			// Interval: next aligned boundary
			// For 180 min interval, align to 0:00, 3:00, 6:00, etc.
			set .@interval, .@intervals[.@i];
			if (.@interval > 0) {
				query_sql "SELECT DATE_ADD(NOW(), INTERVAL ("+.@interval+" - (MINUTE(NOW()) % "+.@interval+")) MINUTE)", .@next$;
			}
		}
		else if (.@types$[.@i] == "daily") {
			// Daily: next occurrence of this time
			query_sql "SELECT IF(CURTIME() < MAKETIME("+.@hours[.@i]+", "+.@mins[.@i]+", 0), CONCAT(CURDATE(), ' ', MAKETIME("+.@hours[.@i]+", "+.@mins[.@i]+", 0)), CONCAT(DATE_ADD(CURDATE(), INTERVAL 1 DAY), ' ', MAKETIME("+.@hours[.@i]+", "+.@mins[.@i]+", 0)))", .@next$;
		}
		else if (.@types$[.@i] == "weekly" || .@types$[.@i] == "fixed") {
			// Weekly: next occurrence of this day/time
			query_sql "SELECT DATE_ADD(DATE_ADD(CURDATE(), INTERVAL (("+.@dow[.@i]+" - WEEKDAY(CURDATE()) + 7) % 7) DAY), INTERVAL "+.@hours[.@i]+" * 60 + "+.@mins[.@i]+" MINUTE)", .@next$;
			// If that time has passed today, add a week
			query_sql "SELECT IF('"+.@next$+"' <= NOW(), DATE_ADD('"+.@next$+"', INTERVAL 7 DAY), '"+.@next$+"')", .@next$;
		}
		else if (.@types$[.@i] == "biweekly") {
			// Biweekly: similar to weekly but every 2 weeks
			query_sql "SELECT DATE_ADD(DATE_ADD(CURDATE(), INTERVAL (("+.@dow[.@i]+" - WEEKDAY(CURDATE()) + 7) % 7) DAY), INTERVAL "+.@hours[.@i]+" * 60 + "+.@mins[.@i]+" MINUTE)", .@next$;
			query_sql "SELECT IF('"+.@next$+"' <= NOW(), DATE_ADD('"+.@next$+"', INTERVAL 14 DAY), '"+.@next$+"')", .@next$;
		}
		else if (.@types$[.@i] == "monthly") {
			// Monthly: next occurrence of this day of month
			if (.@dom[.@i] > 0) {
				query_sql "SELECT IF(DAY(NOW()) < "+.@dom[.@i]+" OR (DAY(NOW()) = "+.@dom[.@i]+" AND CURTIME() < MAKETIME("+.@hours[.@i]+", "+.@mins[.@i]+", 0)), CONCAT(DATE_FORMAT(CURDATE(), '%Y-%m-'), LPAD("+.@dom[.@i]+", 2, '0'), ' ', MAKETIME("+.@hours[.@i]+", "+.@mins[.@i]+", 0)), CONCAT(DATE_FORMAT(DATE_ADD(CURDATE(), INTERVAL 1 MONTH), '%Y-%m-'), LPAD("+.@dom[.@i]+", 2, '0'), ' ', MAKETIME("+.@hours[.@i]+", "+.@mins[.@i]+", 0)))", .@next$;
			}
		}

		if (.@next$ != "") {
			query_sql "UPDATE event_schedule SET next_run = '"+.@next$+"' WHERE id = "+.@ids[.@i];
			debugmes "[EventScheduler] Set next_run for schedule "+.@ids[.@i]+": "+.@next$;
		}
	}
	return;

// ============================================================
// Restore active events after server restart
// One-time events are now stored in event_instances directly
// ============================================================
S_RestoreActiveEvents:
	// Check for EXP Boost (one-time event in event_instances)
	// Status can be 'scheduled' (not yet started) or 'active' (was running)
	if (!$@ExpBoostActive) {
		// Query event_instances for one-time EXP boost within its time window
		query_sql "SELECT ei.id, CAST(JSON_EXTRACT(ei.event_data, '$.multiplier') AS UNSIGNED), TIMESTAMPDIFF(MINUTE, NOW(), ei.scheduled_end), ei.status FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'exp_boost' AND et.enabled = 1 AND ei.trigger_type = 'one_time' AND ei.status IN ('scheduled', 'active') AND NOW() BETWEEN COALESCE(ei.scheduled_start, ei.actual_start) AND ei.scheduled_end LIMIT 1", .@ei_id, .@eb_mult, .@eb_remaining, .@eb_status$;

		if (.@ei_id > 0 && .@eb_remaining > 0) {
			// If status was 'scheduled', mark as active now (server just started during its window)
			if (.@eb_status$ == "scheduled") {
				query_sql "SELECT event_type_id FROM event_instances WHERE id = "+.@ei_id, .@type_id;
				query_sql "UPDATE event_instances SET actual_start = NOW(), status = 'active' WHERE id = "+.@ei_id;

				// Update cache
				query_sql "UPDATE event_current_state SET instance_id = "+.@ei_id+", is_active = 1, cache_data = JSON_OBJECT('multiplier', "+.@eb_mult+") WHERE event_type_id = "+.@type_id;
			}

			set $@ExpBoostInstanceId, .@ei_id;
			set $@ExpBoostMultiplier, .@eb_mult;
			set $@ExpBoostDuration, .@eb_remaining;
			donpcevent "ExpBoost::OnEventStart";
			debugmes "[EventScheduler] Restored EXP Boost (was "+.@eb_status$+")";
		}
	}

	// Check for Happy Hour - restore if we're within an active schedule window
	if (!$@HappyHourActive) {
		// Find any weekly schedule where we're currently in the time window
		query_sql "SELECT es.id, es.duration_minutes, es.start_hour, es.start_minute, (es.start_hour * 60 + es.start_minute + es.duration_minutes) - (HOUR(NOW()) * 60 + MINUTE(NOW())) as remaining FROM event_schedule es JOIN event_types et ON es.event_type_id = et.id WHERE et.code = 'happy_hour' AND et.enabled = 1 AND es.enabled = 1 AND es.day_of_week = WEEKDAY(NOW()) AND (HOUR(NOW()) * 60 + MINUTE(NOW())) >= (es.start_hour * 60 + es.start_minute) AND (HOUR(NOW()) * 60 + MINUTE(NOW())) < (es.start_hour * 60 + es.start_minute + es.duration_minutes) LIMIT 1", .@hh_sched_id, .@hh_duration, .@hh_hour, .@hh_min, .@hh_remaining;

		if (.@hh_sched_id > 0 && .@hh_remaining > 0) {
			set $@HappyHourDuration, .@hh_remaining;
			set $@HappyHourScheduleId, .@hh_sched_id;
			donpcevent "HappyHour::OnEventStart";
			debugmes "[EventScheduler] Restored Happy Hour - "+.@hh_remaining+" minutes remaining";
		}
	}

	// MOTD is handled by S_CheckMOTDRotation
	return;

// ============================================================
// Main scheduler: check for events ready to trigger
// Uses next_run for efficient lookup
// Also checks event_instances for pending one-time events
// ============================================================
S_CheckScheduledEvents:
	// Part 1: Check recurring schedules (next_run based)
	// Uses JOIN to get event code from event_types (not event_schedule.event_code)
	query_sql "SELECT es.id, es.event_type_id, et.code, es.schedule_type, es.interval_minutes, es.duration_minutes, es.day_of_week, es.start_hour, es.start_minute, es.week_interval, COALESCE(JSON_EXTRACT(es.event_params, '$.multiplier'), 100) FROM event_schedule es JOIN event_types et ON es.event_type_id = et.id WHERE es.enabled = 1 AND et.enabled = 1 AND es.next_run IS NOT NULL AND es.next_run <= NOW() AND es.schedule_type != 'once' ORDER BY es.next_run LIMIT 10", .@ids, .@type_ids, .@et_codes$, .@sched_types$, .@intervals, .@durations, .@dow, .@hours, .@mins, .@week_int, .@mults;

	for (set .@i, 0; .@i < getarraysize(.@ids); set .@i, .@i + 1) {
		set .@schedule_id, .@ids[.@i];
		set .@event_code$, .@et_codes$[.@i];
		set .@schedule_type$, .@sched_types$[.@i];
		set .@duration, .@durations[.@i];

		// Check if this event type is already active (prevent overlap)
		// Must clear variable first - query_sql doesn't clear it if no rows returned
		.@already_active = 0;
		query_sql "SELECT 1 FROM event_instances WHERE event_type_id = "+.@type_ids[.@i]+" AND status = 'active' LIMIT 1", .@already_active;
		if (.@already_active) {
			debugmes "[EventScheduler] Skipping "+.@event_code$+" - already active (overlap prevention)";
			// Still update next_run to prevent repeated attempts
			callsub S_CalculateNextRun, .@schedule_id, .@schedule_type$, .@intervals[.@i], .@dow[.@i], .@hours[.@i], .@mins[.@i], .@week_int[.@i];
			continue;
		}

		debugmes "[EventScheduler] Triggering recurring event: "+.@event_code$+" (schedule "+.@schedule_id+")";

		// Create event instance (event_code kept for backwards compat, will be removed later)
		query_sql "INSERT INTO event_instances (event_type_id, event_code, schedule_id, actual_start, status, trigger_type) VALUES ("+.@type_ids[.@i]+", '"+.@event_code$+"', "+.@schedule_id+", NOW(), 'active', 'scheduled')";
		query_sql "SELECT LAST_INSERT_ID()", .@instance_id;

		// Update event_current_state cache
		query_sql "UPDATE event_current_state SET instance_id = "+.@instance_id+", is_active = 1, cache_data = JSON_OBJECT('schedule_id', "+.@schedule_id+", 'duration', "+.@duration+") WHERE event_type_id = "+.@type_ids[.@i];

		// Update schedule last_run (status stays as-is for recurring schedules)
		query_sql "UPDATE event_schedule SET last_run = NOW() WHERE id = "+.@schedule_id;

		// Trigger the event NPC
		if (.@event_code$ == "happy_hour") {
			// Calculate actual remaining time (schedule may have started earlier)
			// For weekly: end time = today's scheduled start + duration
			query_sql "SELECT TIMESTAMPDIFF(MINUTE, NOW(), DATE_ADD(DATE_ADD(CURDATE(), INTERVAL "+.@hours[.@i]+" HOUR), INTERVAL "+.@mins[.@i]+" + "+.@duration+" MINUTE))", .@remaining;
			if (.@remaining < 1) set .@remaining, 1;  // At least 1 minute
			if (.@remaining > .@duration) set .@remaining, .@duration;  // Cap at full duration

			set $@HappyHourDuration, .@remaining;
			set $@HappyHourScheduleId, .@schedule_id;
			set $@HappyHourInstanceId, .@instance_id;
			donpcevent "HappyHour::OnEventStart";
		}
		else if (.@event_code$ == "motd") {
			// MOTD is handled separately by S_CheckMOTDRotation
			// This just marks the schedule as triggered
		}
		else if (.@event_code$ == "invasion") {
			// Town Invasion - read params from event_params JSON
			query_sql "SELECT event_params FROM event_schedule WHERE id = "+.@schedule_id, .@params$;
			if (.@params$ != "") {
				// Parse JSON params: {"threat":1,"town":1,"theme":1}
				.@threat = atoi(replacestr(replacestr(replacestr(.@params$, "{", ""), "}", ""), "\"threat\":", ""));
				// Default to random if not specified
				if (.@threat < 0 || .@threat > 4) .@threat = rand(0, 2);
			} else {
				.@threat = rand(0, 2); // Random easy-medium-hard
			}
			// Set trigger type and pending params for scheduled invasions
			// "scheduled" bypasses cooldown, time window, and player checks
			$@InvasionTriggerType$ = "scheduled";
			$@InvasionPendingThreat = .@threat;
			$@InvasionPendingTown = 0; // 0 = random selection
			$@InvasionPendingTheme = 0; // 0 = random selection
			donpcevent "TownInvasion::OnScheduledInvasion";
		}
		else {
			// Generic event - try to call OnEventStart
			donpcevent .@event_code$+"::OnEventStart";
		}

		// Calculate and set next_run
		callsub S_CalculateNextRun, .@schedule_id, .@schedule_type$, .@intervals[.@i], .@dow[.@i], .@hours[.@i], .@mins[.@i], .@week_int[.@i];
	}

	// Part 2: Check one-time events in event_instances (scheduled_start based)
	query_sql "SELECT ei.id, ei.event_type_id, et.code, TIMESTAMPDIFF(MINUTE, NOW(), ei.scheduled_end), CAST(JSON_EXTRACT(ei.event_data, '$.multiplier') AS UNSIGNED) FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE ei.status = 'scheduled' AND ei.trigger_type = 'one_time' AND et.enabled = 1 AND ei.scheduled_start IS NOT NULL AND ei.scheduled_start <= NOW() ORDER BY ei.scheduled_start LIMIT 5", .@ot_ids, .@ot_type_ids, .@ot_codes$, .@ot_durations, .@ot_mults;

	for (set .@j, 0; .@j < getarraysize(.@ot_ids); set .@j, .@j + 1) {
		set .@instance_id, .@ot_ids[.@j];
		set .@event_code$, .@ot_codes$[.@j];
		set .@duration, .@ot_durations[.@j];

		// Check if this event type is already active (prevent overlap)
		// Must clear variable first - query_sql doesn't clear it if no rows returned
		.@already_active = 0;
		query_sql "SELECT 1 FROM event_instances WHERE event_type_id = "+.@ot_type_ids[.@j]+" AND status = 'active' AND id != "+.@instance_id+" LIMIT 1", .@already_active;
		if (.@already_active) {
			debugmes "[EventScheduler] Skipping one-time "+.@event_code$+" - another instance active";
			continue;
		}

		// For invasions, also check the global variable (invasion might be running without DB update)
		if (.@event_code$ == "invasion" && $@InvasionActive) {
			debugmes "[EventScheduler] Skipping one-time invasion - $@InvasionActive is true";
			continue;
		}

		debugmes "[EventScheduler] Triggering one-time event: "+.@event_code$+" (instance "+.@instance_id+")";

		// For invasions, don't mark as 'active' yet - TownInvasion will handle it
		// This prevents orphaned 'active' records if the invasion fails to start
		if (.@event_code$ != "invasion") {
			// Update instance to active
			query_sql "UPDATE event_instances SET actual_start = NOW(), status = 'active' WHERE id = "+.@instance_id;

			// Update event_current_state cache
			query_sql "UPDATE event_current_state SET instance_id = "+.@instance_id+", is_active = 1, cache_data = JSON_OBJECT('multiplier', "+.@ot_mults[.@j]+", 'duration', "+.@duration+") WHERE event_type_id = "+.@ot_type_ids[.@j];
		}

		// Trigger the event NPC
		if (.@event_code$ == "exp_boost") {
			set $@ExpBoostMultiplier, .@ot_mults[.@j];
			set $@ExpBoostDuration, .@duration;
			set $@ExpBoostInstanceId, .@instance_id;
			donpcevent "ExpBoost::OnEventStart";
		}
		else if (.@event_code$ == "invasion") {
			// One-time scheduled invasion - read params from event_data JSON
			query_sql "SELECT CAST(JSON_EXTRACT(event_data, '$.threat') AS UNSIGNED), CAST(JSON_EXTRACT(event_data, '$.town_id') AS UNSIGNED), CAST(JSON_EXTRACT(event_data, '$.theme_id') AS UNSIGNED) FROM event_instances WHERE id = "+.@instance_id, .@inv_threat, .@inv_town, .@inv_theme;
			// Default to random if not specified
			if (.@inv_threat < 0 || .@inv_threat > 4) .@inv_threat = rand(0, 2);
			if (.@inv_town < 1) .@inv_town = 0; // 0 = random
			if (.@inv_theme < 1) .@inv_theme = 0; // 0 = random
			$@InvasionPendingThreat = .@inv_threat;
			$@InvasionPendingTown = .@inv_town;
			$@InvasionPendingTheme = .@inv_theme;
			$@InvasionPendingInstanceId = .@instance_id;  // Pass instance ID for status update
			$@InvasionTriggerType$ = "scheduled";
			donpcevent "TownInvasion::OnScheduledInvasion";
		}
		else {
			// Generic one-time event
			donpcevent .@event_code$+"::OnEventStart";
		}
	}
	return;

// ============================================================
// Calculate next_run for a schedule after it triggers
// ============================================================
S_CalculateNextRun:
	set .@schedule_id, getarg(0);
	set .@type$, getarg(1);
	set .@interval, getarg(2);
	set .@dow, getarg(3);
	set .@hour, getarg(4);
	set .@minute, getarg(5);
	set .@week_int, getarg(6);

	set .@next$, "";

	if (.@type$ == "interval") {
		// Add interval minutes to now
		query_sql "SELECT DATE_ADD(NOW(), INTERVAL "+.@interval+" MINUTE)", .@next$;
	}
	else if (.@type$ == "daily") {
		// Tomorrow at same time
		query_sql "SELECT CONCAT(DATE_ADD(CURDATE(), INTERVAL 1 DAY), ' ', MAKETIME("+.@hour+", "+.@minute+", 0))", .@next$;
	}
	else if (.@type$ == "weekly" || .@type$ == "fixed") {
		// Next week same day/time
		query_sql "SELECT DATE_ADD(NOW(), INTERVAL 7 DAY)", .@next$;
		// Align to exact day/time
		query_sql "SELECT DATE_ADD(DATE_ADD(CURDATE(), INTERVAL (("+.@dow+" - WEEKDAY(CURDATE()) + 7) % 7 + 7) DAY), INTERVAL "+.@hour+" * 60 + "+.@minute+" MINUTE)", .@next$;
	}
	else if (.@type$ == "biweekly") {
		// Two weeks from now
		query_sql "SELECT DATE_ADD(NOW(), INTERVAL 14 DAY)", .@next$;
	}
	else if (.@type$ == "once") {
		// One-time event - disable schedule (it's done)
		query_sql "UPDATE event_schedule SET next_run = NULL, enabled = 0 WHERE id = "+.@schedule_id;
		return;
	}

	if (.@next$ != "") {
		query_sql "UPDATE event_schedule SET next_run = '"+.@next$+"' WHERE id = "+.@schedule_id;
		debugmes "[EventScheduler] Next run for schedule "+.@schedule_id+": "+.@next$;
	}
	return;

// ============================================================
// Check MOTD rotation (announce new monster)
// Updates event_current_state cache for C++ hot-path
// ============================================================
S_CheckMOTDRotation:
	// Check if MOTD is enabled (centralized in event_types)
	query_sql "SELECT id, enabled FROM event_types WHERE code = 'motd'", .@motd_type_id, .@enabled;
	if (!.@enabled) {
		// MOTD disabled - clear cache and globals
		if ($@LastMOTDMobID > 0) {
			set $@LastMOTDMobID, 0;
			set $@CurrentMOTDName$, "";
			query_sql "UPDATE event_current_state SET is_active = 0, instance_id = NULL, cache_data = NULL WHERE event_type_id = "+.@motd_type_id;
		}
		return;
	}

	// Get current MOTD from event_motd_schedule
	query_sql "SELECT mob_id, mob_name, tier, TIMESTAMPDIFF(MINUTE, NOW(), end_time), end_time FROM event_motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@mob_id, .@mob_name$, .@tier$, .@mins_left, .@end_time$;

	// Check if this is a new rotation (different from last announced)
	if (.@mob_id > 0 && .@mob_id != $@LastMOTDMobID) {
		// Complete ALL previous active MOTD instances (handles server restarts)
		query_sql "UPDATE event_instances SET actual_end = NOW(), status = 'completed' WHERE event_code = 'motd' AND status = 'active'";

		// Create new MOTD instance with event_data
		query_sql "INSERT INTO event_instances (event_type_id, event_code, actual_start, status, trigger_type, event_data) VALUES ("+.@motd_type_id+", 'motd', NOW(), 'active', 'scheduled', JSON_OBJECT('mob_id', "+.@mob_id+", 'mob_name', '"+escape_sql(.@mob_name$)+"', 'tier', '"+.@tier$+"'))";
		query_sql "SELECT LAST_INSERT_ID()", .@instance_id;
		set $@CurrentMOTDInstanceId, .@instance_id;

		// Update event_current_state cache (used by C++)
		query_sql "UPDATE event_current_state SET instance_id = "+.@instance_id+", is_active = 1, cache_data = JSON_OBJECT('mob_id', "+.@mob_id+", 'mob_name', '"+escape_sql(.@mob_name$)+"', 'tier', '"+.@tier$+"', 'end_time', '"+.@end_time$+"') WHERE event_type_id = "+.@motd_type_id;

		// Update globals
		set $@LastMOTDMobID, .@mob_id;
		set $@CurrentMOTDName$, .@mob_name$;

		// Check if this monster has a bounty
		query_sql "SELECT has_bounty FROM event_motd_monsters WHERE mob_id = "+.@mob_id, .@has_bounty;

		if (.@has_bounty) {
			announce "Monster of the Day: "+.@mob_name$+"! 3x EXP, Drops & Hunt Points for "+.@mins_left+" minutes!", bc_all, 0x00FF00;
		} else {
			announce "Monster of the Day: "+.@mob_name$+"! 3x EXP & Drops for "+.@mins_left+" minutes!", bc_all, 0x00FF00;
		}

		debugmes "[EventScheduler] MOTD rotated to: "+.@mob_name$+" ("+.@mob_id+")";
	}
	return;

// ============================================================
// Check Conditional Triggers
// Generic system that evaluates trigger groups for all events
// ============================================================
S_CheckCondTriggers:
	// Query event types that have enabled trigger groups
	query_sql "SELECT DISTINCT et.id, et.code FROM event_types et JOIN event_trigger_groups tg ON tg.event_type_id = et.id WHERE et.enabled = 1 AND tg.enabled = 1", .@type_ids, .@codes$;

	for (.@i = 0; .@i < getarraysize(.@type_ids); .@i++) {
		.@event_type_id = .@type_ids[.@i];
		.@event_code$ = .@codes$[.@i];

		// Skip if event is already active
		.@already_active = 0;
		query_sql "SELECT 1 FROM event_instances WHERE event_type_id = "+.@event_type_id+" AND status = 'active' LIMIT 1", .@already_active;
		if (.@already_active) continue;

		// Special check for invasion - also check global variable
		if (.@event_code$ == "invasion" && $@InvasionActive) continue;

		// Evaluate trigger groups for this event type
		callsub S_EvaluateTriggerGroups, .@event_type_id, .@event_code$;
	}
	return;

// ============================================================
// Evaluate all trigger groups for an event type
// Groups are evaluated in priority order (OR between groups)
// ============================================================
S_EvaluateTriggerGroups:
	.@event_type_id = getarg(0);
	.@event_code$ = getarg(1);

	// Get all enabled trigger groups for this event, ordered by priority
	query_sql "SELECT id, name, check_interval_minutes, last_checked, event_params, last_ended_config_key FROM event_trigger_groups WHERE event_type_id = "+.@event_type_id+" AND enabled = 1 ORDER BY priority DESC", .@group_ids, .@group_names$, .@intervals, .@last_checked$, .@event_params$, .@config_keys$;

	for (.@g = 0; .@g < getarraysize(.@group_ids); .@g++) {
		.@group_id = .@group_ids[.@g];
		.@interval = .@intervals[.@g];

		// Check if enough time has passed since last check
		if (.@last_checked$[.@g] != "" && .@last_checked$[.@g] != "NULL") {
			query_sql "SELECT TIMESTAMPDIFF(MINUTE, '"+.@last_checked$[.@g]+"', NOW())", .@minutes_since;
			if (.@minutes_since < .@interval) {
				// Not time to check this group yet
				continue;
			}
		}

		// Update last_checked timestamp
		query_sql "UPDATE event_trigger_groups SET last_checked = NOW() WHERE id = "+.@group_id;

		// Evaluate all conditions in this group (AND logic)
		.@group_passed = 1;

		query_sql "SELECT id, condition_type, params, required FROM event_trigger_conditions WHERE group_id = "+.@group_id+" ORDER BY eval_order", .@cond_ids, .@cond_types$, .@cond_params$, .@cond_required;

		for (.@c = 0; .@c < getarraysize(.@cond_ids); .@c++) {
			// Evaluate condition - result stored in .cond_result (NPC-scoped)
			.cond_result = 0;
			callsub S_EvaluateCondition, .@cond_types$[.@c], .@cond_params$[.@c], .@event_type_id, .@config_keys$[.@g];

			if (!.cond_result && .@cond_required[.@c]) {
				// Required condition failed - group fails
				.@group_passed = 0;
				break;
			}
		}

		if (.@group_passed) {
			// All conditions passed - trigger the event!
			debugmes "[ConditionalTrigger] Group '"+.@group_names$[.@g]+"' passed for "+.@event_code$;
			callsub S_TriggerCondEvent, .@event_type_id, .@event_code$, .@group_id, .@event_params$[.@g];
			return;  // Stop evaluating other groups (OR logic satisfied)
		}
	}
	return;

// ============================================================
// Evaluate a single condition
// Sets .cond_result to 1 if condition passes, 0 if fails
// ============================================================
S_EvaluateCondition:
	.@type$ = getarg(0);
	.@params$ = getarg(1);
	.@event_type_id = getarg(2);
	.@config_key$ = getarg(3, "");

	// Random chance - check FIRST for performance (short-circuits most checks)
	if (.@type$ == "random_chance") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.percent'))", .@percent$;
		.@percent = atoi(.@percent$);
		if (.@percent <= 0) {
			.cond_result = 0;
			return;
		}
		.cond_result = (rand(100) < .@percent);
		return;
	}

	// Outside WoE - pass if WoE is NOT active
	if (.@type$ == "outside_woe") {
		// Check agit flag (WoE active indicator)
		// agitcheck() returns 1 if WoE is active
		if (agitcheck() || agitcheck2()) {
			.cond_result = 0;
			return;
		}
		.cond_result = 1;
		return;
	}

	// No active event - pass if specified events are NOT running
	if (.@type$ == "no_active_event") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.events'))", .@events$;
		// Parse JSON array - simple approach: check each known event
		// Format: ["invasion", "exp_boost"]
		if (countstr(.@events$, "invasion") > 0) {
			.@active = 0;
			query_sql "SELECT 1 FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'invasion' AND ei.status = 'active' LIMIT 1", .@active;
			if (.@active || $@InvasionActive) {
				.cond_result = 0;
				return;
			}
		}
		if (countstr(.@events$, "exp_boost") > 0) {
			.@active = 0;
			query_sql "SELECT 1 FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'exp_boost' AND ei.status = 'active' LIMIT 1", .@active;
			if (.@active || $@ExpBoostActive) {
				.cond_result = 0;
				return;
			}
		}
		if (countstr(.@events$, "happy_hour") > 0) {
			.@active = 0;
			query_sql "SELECT 1 FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'happy_hour' AND ei.status = 'active' LIMIT 1", .@active;
			if (.@active || $@HappyHourActive) {
				.cond_result = 0;
				return;
			}
		}
		.cond_result = 1;
		return;
	}

	// Cooldown - pass if enough time since last event ended
	if (.@type$ == "cooldown") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.minutes'))", .@minutes$;
		.@minutes = atoi(.@minutes$);
		if (.@minutes <= 0) {
			.cond_result = 1;
			return;
		}

		// Get last ended timestamp from config
		if (.@config_key$ == "") {
			.cond_result = 1;
			return;
		}
		query_sql "SELECT config_value FROM event_config WHERE config_key = '"+escape_sql(.@config_key$)+"'", .@last_ended$;
		.@last_ended = atoi(.@last_ended$);
		if (.@last_ended == 0) {
			.cond_result = 1;
			return;
		}

		.@elapsed_min = (gettimetick(2) - .@last_ended) / 60;
		.cond_result = (.@elapsed_min >= .@minutes);
		return;
	}

	// Time window - pass if current hour is within range
	if (.@type$ == "time_window") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.start_hour')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.end_hour'))", .@start$, .@end$;
		.@start = atoi(.@start$);
		.@end = atoi(.@end$);
		.@hour = gettime(3);  // Current hour (0-23)
		.cond_result = (.@hour >= .@start && .@hour < .@end);
		return;
	}

	// Player count - pass if online players within range
	if (.@type$ == "player_count") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.min')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.max'))", .@min$, .@max$;
		.@min = atoi(.@min$);
		.@max = atoi(.@max$);
		.@count = getusers(1);
		if (.@min > 0 && .@count < .@min) {
			.cond_result = 0;
			return;
		}
		if (.@max > 0 && .@count > .@max) {
			.cond_result = 0;
			return;
		}
		.cond_result = 1;
		return;
	}

	// Active player count - pass if enough players moved recently
	if (.@type$ == "active_player_count") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.min')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.window_minutes'))", .@min$, .@window$;
		.@min = atoi(.@min$);
		.@window = atoi(.@window$);
		if (.@window <= 0) .@window = 10;  // Default 10 minutes

		query_sql "SELECT COUNT(*) FROM player_activity_tracking WHERE position_changed = 1 AND last_seen >= DATE_SUB(NOW(), INTERVAL "+.@window+" MINUTE)", .@active_count;
		.cond_result = (.@active_count >= .@min);
		return;
	}

	// Day of week - pass if today is in the list
	if (.@type$ == "day_of_week") {
		query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@params$)+"', '$.days'))", .@days$;
		// gettime(4) returns day of week: 0=Sunday, 1=Monday, ..., 6=Saturday
		// Convert to 0=Monday format used in our config
		.@today = gettime(4);
		.@today_mon = (.@today == 0) ? 6 : (.@today - 1);  // Convert to 0=Mon

		// Check if today is in the days array (simple string search)
		.cond_result = (countstr(.@days$, ""+.@today_mon) > 0);
		return;
	}

	// Unknown condition type
	debugmes "[ConditionalTrigger] Unknown condition type: "+.@type$;
	.cond_result = 0;
	return;

// ============================================================
// Trigger an event via conditional trigger
// ============================================================
S_TriggerCondEvent:
	.@event_type_id = getarg(0);
	.@event_code$ = getarg(1);
	.@group_id = getarg(2);
	.@event_params$ = getarg(3, "");

	debugmes "[ConditionalTrigger] Triggering "+.@event_code$+" via group "+.@group_id;

	// Create event instance
	query_sql "INSERT INTO event_instances (event_type_id, event_code, trigger_type, trigger_group_id, actual_start, status, event_data) VALUES ("+.@event_type_id+", '"+.@event_code$+"', 'conditional', "+.@group_id+", NOW(), 'active', '"+escape_sql(.@event_params$)+"')";
	query_sql "SELECT LAST_INSERT_ID()", .@instance_id;

	// Update event_current_state cache
	query_sql "UPDATE event_current_state SET instance_id = "+.@instance_id+", is_active = 1, cache_data = '"+escape_sql(.@event_params$)+"' WHERE event_type_id = "+.@event_type_id;

	// Trigger the appropriate event NPC
	if (.@event_code$ == "invasion") {
		// For invasion, don't mark as active in DB yet - TownInvasion handles it
		// Set up invasion params from event_params JSON
		if (.@event_params$ != "" && .@event_params$ != "null") {
			query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@event_params$)+"', '$.threat'))", .@threat$;
			$@InvasionPendingThreat = atoi(.@threat$);
		} else {
			$@InvasionPendingThreat = rand(0, 2);  // Random easy-medium-hard
		}
		$@InvasionPendingTown = 0;   // Random
		$@InvasionPendingTheme = 0;  // Random
		$@InvasionPendingInstanceId = .@instance_id;
		$@InvasionTriggerType$ = "conditional";
		donpcevent "TownInvasion::OnScheduledInvasion";
	}
	else if (.@event_code$ == "exp_boost") {
		// Parse multiplier and duration from event_params
		.@multiplier = 150;  // Default 1.5x
		.@duration = 30;     // Default 30 min
		if (.@event_params$ != "" && .@event_params$ != "null") {
			query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@event_params$)+"', '$.multiplier')), JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@event_params$)+"', '$.duration_minutes'))", .@mult$, .@dur$;
			if (.@mult$ != "" && .@mult$ != "null") .@multiplier = atoi(.@mult$);
			if (.@dur$ != "" && .@dur$ != "null") .@duration = atoi(.@dur$);
		}
		$@ExpBoostMultiplier = .@multiplier;
		$@ExpBoostDuration = .@duration;
		$@ExpBoostInstanceId = .@instance_id;
		donpcevent "ExpBoost::OnEventStart";
	}
	else if (.@event_code$ == "happy_hour") {
		// Parse duration from event_params
		.@duration = 120;  // Default 2 hours
		if (.@event_params$ != "" && .@event_params$ != "null") {
			query_sql "SELECT JSON_UNQUOTE(JSON_EXTRACT('"+escape_sql(.@event_params$)+"', '$.duration_minutes'))", .@dur$;
			if (.@dur$ != "" && .@dur$ != "null") .@duration = atoi(.@dur$);
		}
		$@HappyHourDuration = .@duration;
		$@HappyHourInstanceId = .@instance_id;
		donpcevent "HappyHour::OnEventStart";
	}
	else {
		// Generic event - try to call OnEventStart
		donpcevent .@event_code$+"::OnEventStart";
	}
	return;

// ============================================================
// Generate MOTD Schedule (called daily at midnight and on init)
// ============================================================
OnClock0000:
OnGenerateMOTDSchedule:
	// Check if MOTD is enabled
	query_sql "SELECT enabled FROM event_types WHERE code = 'motd'", .@enabled;
	if (!.@enabled) {
		debugmes "[EventScheduler] MOTD disabled, skipping schedule generation";
		end;
	}

	// Get configuration
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_selection_mode'", .@mode$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_min_level'", .@min_level$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_duration_hours'", .@duration_str$;

	if (.@mode$ == "") set .@mode$, "random";
	if (.@min_level$ == "") set .@min_level, 50; else set .@min_level, atoi(.@min_level$);
	if (.@duration_str$ == "") set .@duration, 3; else set .@duration, atoi(.@duration_str$);
	// Clamp duration to 1-6 hours
	if (.@duration < 1) set .@duration, 1;
	if (.@duration > 6) set .@duration, 6;

	// Clear today's and tomorrow's schedule
	query_sql "DELETE FROM event_motd_schedule WHERE DATE(start_time) IN (CURDATE(), DATE_ADD(CURDATE(), INTERVAL 1 DAY))";

	// Generate slots based on duration (24 / duration = number of slots per day)
	set .@slots_per_day, 24 / .@duration;
	set .@tier_index, 0;
	setarray .@tiers$[0], "medium", "hard";

	// Generate 2 days worth of schedule (today + tomorrow)
	for (set .@day, 0; .@day < 2; set .@day, .@day + 1) {
		for (set .@i, 0; .@i < .@slots_per_day; set .@i, .@i + 1) {
			set .@hour, .@i * .@duration;

			if (.@mode$ == "tier_rotation") {
				// Alternate between medium and hard
				set .@tier$, .@tiers$[.@tier_index % 2];
				set .@tier_index, .@tier_index + 1;

				// Select random monster from this tier
				query_sql "SELECT mob_id, mob_name FROM event_motd_monsters WHERE tier = '"+.@tier$+"' AND enabled = 1 AND mob_level >= "+.@min_level+" ORDER BY RAND() LIMIT 1", .@mob_id, .@mob_name$;
			} else {
				// Random mode - pick any monster
				query_sql "SELECT mob_id, mob_name, tier FROM event_motd_monsters WHERE enabled = 1 AND mob_level >= "+.@min_level+" ORDER BY RAND() LIMIT 1", .@mob_id, .@mob_name$, .@tier$;
			}

			if (.@mob_id > 0) {
				// Insert into schedule (day offset added to CURDATE)
				query_sql "INSERT INTO event_motd_schedule (mob_id, mob_name, start_time, end_time, tier) VALUES ("+.@mob_id+", '"+escape_sql(.@mob_name$)+"', DATE_ADD(DATE_ADD(CURDATE(), INTERVAL "+.@day+" DAY), INTERVAL "+.@hour+" HOUR), DATE_ADD(DATE_ADD(CURDATE(), INTERVAL "+.@day+" DAY), INTERVAL "+(.@hour+.@duration)+" HOUR), '"+.@tier$+"')";
			}
		}
	}

	debugmes "[EventScheduler] Generated MOTD schedule for today and tomorrow";

	// Immediately check for MOTD rotation (in case we just generated for current slot)
	callsub S_CheckMOTDRotation;
	end;

// ============================================================
// Called by event NPCs when they end (to update instance)
// ============================================================
OnEventEnd:
	set .@event_code$, getarg(0, "");
	set .@instance_id, getarg(1, 0);

	if (.@instance_id > 0) {
		query_sql "UPDATE event_instances SET actual_end = NOW(), status = 'completed' WHERE id = "+.@instance_id;

		// Clear event_current_state cache
		query_sql "UPDATE event_current_state ecs JOIN event_instances ei ON ecs.instance_id = ei.id SET ecs.is_active = 0, ecs.instance_id = NULL, ecs.cache_data = NULL WHERE ei.id = "+.@instance_id;
	}
	return;
}

// ============================================================
// Helper function to get current MOTD info (uses cache)
// ============================================================
function	script	F_GetCurrentMOTD	{
	// Use global variable first (fastest)
	if ($@LastMOTDMobID > 0) return $@LastMOTDMobID;

	// Fall back to cache
	query_sql "SELECT JSON_EXTRACT(cache_data, '$.mob_id') FROM event_current_state ecs JOIN event_types et ON ecs.event_type_id = et.id WHERE et.code = 'motd' AND ecs.is_active = 1", .@mob_id;
	return .@mob_id;
}

// ============================================================
// Helper function to check if a specific mob is current MOTD
// ============================================================
function	script	F_IsMOTD	{
	set .@check_mob_id, getarg(0, 0);

	// Use global variable first (fastest)
	if ($@LastMOTDMobID == .@check_mob_id) return 1;

	return 0;
}

// ============================================================
// Helper function to check if MOTD system is enabled
// ============================================================
function	script	F_IsMOTDEnabled	{
	query_sql "SELECT enabled FROM event_types WHERE code = 'motd'", .@enabled;
	return .@enabled;
}

// ============================================================
// Helper function to get active instance ID for an event
// Uses JOIN with event_types instead of event_code column
// ============================================================
function	script	F_GetActiveInstance	{
	set .@event_code$, getarg(0, "");
	query_sql "SELECT ei.id FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = '"+.@event_code$+"' AND ei.status = 'active' ORDER BY ei.id DESC LIMIT 1", .@instance_id;
	return .@instance_id;
}

// ============================================================
// Helper function to end an event and update its instance
// Call this from event NPCs when they end
// ============================================================
function	script	F_EndEventInstance	{
	set .@instance_id, getarg(0, 0);
	if (.@instance_id <= 0) return;

	// Update instance
	query_sql "UPDATE event_instances SET actual_end = NOW(), status = 'completed' WHERE id = "+.@instance_id;

	// Clear cache
	query_sql "UPDATE event_current_state SET is_active = 0, instance_id = NULL WHERE instance_id = "+.@instance_id;
}
