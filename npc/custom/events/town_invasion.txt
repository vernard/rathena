//===== rAthena Script =======================================
//= Town Invasion Event System
//= Themed monster invasions with OPM-inspired threat levels
//===== Description: =========================================
//= Dynamic event where monsters invade towns:
//= - 16 themed invasions (race, element, special)
//= - 5 threat levels (Wolf → God)
//= - Wave-based progression with escalating difficulty
//= - Hunt Points rewards + bonus EXP
//= - Discord webhook notifications
//============================================================

-	script	TownInvasion	FAKE_NPC,{
	end;

// ============================================================
// INITIALIZATION
// ============================================================
OnInit:
	// Debug mode - set to 1 to enable verbose logging
	$@InvasionDebug = 1;

	// Announcement settings (for tuning/debugging)
	$@InvasionAnnounceEvery = 5;       // Announce remaining every X kills (default: 5, set to 1 for every kill)
	$@InvasionAnnounceLocBelow = 5;    // Show locations when remaining is below this (default: 5)
	$@InvasionAnnounceCountPercent = 50; // Announce every kill when remaining <= this % of wave (default: 50)
	$@InvasionAnnounceLocInterval = 15; // When overtime, announce locations every X seconds
	$@InvasionWaveOvertimeThreshold = 60; // After this many seconds, start announcing locations (default: 60 for testing, 300 for prod)
	$@InvasionWaveOvertimeAnnounced = 0; // Track if overtime announcement was made this wave

	// Global state variables
	$@InvasionActive = 0;
	$@InvasionID = 0;
	$@InvasionMap$ = "";
	$@InvasionTheme = 0;
	$@InvasionThreat = 0;
	$@InvasionWave = 0;
	$@InvasionTotalWaves = 0;
	$@InvasionMonstersSpawned = 0;
	$@InvasionMonstersKilled = 0;
	$@InvasionStartTime = 0;
	$@InvasionEndTime = 0;
	$@InvasionLastAnnounce = 0;
	$@InvasionNextWaveTime = 0;
	$@InvasionEventInstanceID = 0;
	// Wave tracking for required kills (MVP/miniboss must all die)
	$@InvasionWaveBossSpawns = 0;    // MVP + miniboss spawned this wave
	$@InvasionWaveBossKills = 0;     // MVP + miniboss killed this wave
	$@InvasionWaveRegularSpawns = 0; // Non-boss monsters spawned
	$@InvasionWaveRegularKills = 0;  // Non-boss monsters killed
	$@InvasionLastBossGather = 0;    // Last time bosses were announced (if outside radius)
	$@InvasionLastLocAnnounce = 0;   // Last time straggler locations were announced
	$@InvasionWaveStartTime = 0;     // When current wave started (for overtime detection)

	// Cancel any active invasions from before server restart
	// (We don't restore mid-invasion state - GMs can manually restart if needed)
	query_sql "UPDATE event_invasion_instances SET status = 'cancelled', ended_at = NOW() WHERE status = 'active'";

	// Also cancel in event_instances (unified event history)
	// Using subquery because rAthena query_sql may not support UPDATE with JOIN
	query_sql "UPDATE event_instances SET status = 'cancelled', actual_end = NOW() WHERE status = 'active' AND event_type_id = (SELECT id FROM event_types WHERE code = 'invasion')";

	// Cancel scheduled invasions that are past their start time (missed due to downtime)
	query_sql "UPDATE event_instances SET status = 'cancelled', actual_end = NOW() WHERE status = 'scheduled' AND scheduled_start < NOW() AND event_type_id = (SELECT id FROM event_types WHERE code = 'invasion')";

	debugmes "[TownInvasion] Initialized";
	end;

// NPC Timer - polls every 10 seconds to check timeout/wave spawn/reminders
// Using polling instead of addtimer because addtimer requires player attachment
OnTimer10000:
	if (!$@InvasionActive) {
		stopnpctimer;
		end;
	}

	// Check for pending wave spawn
	if ($@InvasionNextWaveTime > 0 && gettimetick(2) >= $@InvasionNextWaveTime) {
		$@InvasionNextWaveTime = 0;
		callsub S_SpawnWave;
		// IMPORTANT: End here to let monsters register before mobcount check
		// The next timer tick will handle the failsafe check
		initnpctimer;
		end;
	}

	// Check for timeout
	if (gettimetick(2) >= $@InvasionEndTime) {
		stopnpctimer;
		donpcevent "TownInvasion::OnInvasionTimeout";
		end;
	}

	// Use tracked counts for accurate invasion monster count (ignores non-invasion mobs like Wild Rose)
	.@tracked_remaining = $@InvasionWaveMonsters - $@InvasionWaveKills;
	if (.@tracked_remaining < 0) .@tracked_remaining = 0;

	// Also get getmapunits for GID array (needed for location lookups), but not for counting
	.@raw_count = getmapunits(BL_MOB, $@InvasionMap$, .@mob_gids[0]);

	// Check if we've hit the 90% clear threshold based on tracked count
	// Wave clears when: all bosses dead AND 90% of regular monsters dead
	if ($@InvasionWaveMonsters > 0 && $@InvasionActive) {
		// Get the clear threshold (default 90%)
		query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_wave_clear_percent'", .@threshold$;
		.@threshold = (.@threshold$ != "") ? atoi(.@threshold$) : 90;

		// Calculate how many can remain (10% if threshold is 90%)
		.@max_remaining = ($@InvasionWaveMonsters * (100 - .@threshold)) / 100;

		// Check if all bosses are dead
		.@bosses_clear = ($@InvasionWaveBossKills >= $@InvasionWaveBossSpawns);

		// If all bosses dead AND tracked remaining is at or below threshold, trigger wave clear
		if (.@bosses_clear && .@tracked_remaining <= .@max_remaining) {
			debugmes "[TownInvasion] Failsafe: bosses_clear=1, tracked_remaining=" + .@tracked_remaining + " <= " + .@max_remaining + " (threshold " + .@threshold + "%), forcing completion";
			donpcevent "TownInvasion::OnWaveCleared";
			end;
		} else if (!.@bosses_clear && .@tracked_remaining <= .@max_remaining) {
			// 90% cleared but bosses still alive - don't clear, just log
			if ($@InvasionDebug) {
				.@boss_remaining = $@InvasionWaveBossSpawns - $@InvasionWaveBossKills;
				debugmes "[TownInvasion] Failsafe skipped: " + .@boss_remaining + " boss(es) still alive";
			}
		}
	}

	// Periodic straggler location announcement when wave is overtime
	// Helps players find remaining monsters when they're struggling
	.@wave_overtime = ($@InvasionWaveStartTime > 0 && gettimetick(2) - $@InvasionWaveStartTime >= $@InvasionWaveOvertimeThreshold);

	// Global announcement when overtime starts (only once per wave) - timer check version
	if (.@wave_overtime && !$@InvasionWaveOvertimeAnnounced) {
		$@InvasionWaveOvertimeAnnounced = 1;
		// Calculate remaining breakdown for announcement
		.@ot_boss_remaining = $@InvasionWaveBossSpawns - $@InvasionWaveBossKills;
		.@ot_regular_remaining = .@tracked_remaining - .@ot_boss_remaining;
		if (.@ot_regular_remaining < 0) .@ot_regular_remaining = 0;
		.@ot_msg$ = "[TownInvasion] Wave " + $@InvasionWave + " OVERTIME! " + .@tracked_remaining + " remaining";
		if (.@ot_boss_remaining > 0 || .@ot_regular_remaining > 0) {
			.@ot_msg$ = .@ot_msg$ + " (";
			if (.@ot_boss_remaining > 0) {
				.@ot_msg$ = .@ot_msg$ + .@ot_boss_remaining + " boss";
				if (.@ot_boss_remaining != 1) .@ot_msg$ = .@ot_msg$ + "es";
			}
			if (.@ot_boss_remaining > 0 && .@ot_regular_remaining > 0) .@ot_msg$ = .@ot_msg$ + ", ";
			if (.@ot_regular_remaining > 0) {
				.@ot_msg$ = .@ot_msg$ + .@ot_regular_remaining + " mob";
				if (.@ot_regular_remaining != 1) .@ot_msg$ = .@ot_msg$ + "s";
			}
			.@ot_msg$ = .@ot_msg$ + ")";
		}
		.@ot_msg$ = .@ot_msg$ + " - Locations revealed every " + $@InvasionAnnounceLocInterval + "s!";
		announce .@ot_msg$, bc_all, 0xFF6600;
	}

	if ($@InvasionWaveMonsters > 0 && .@tracked_remaining > 0 && .@wave_overtime && gettimetick(2) - $@InvasionLastLocAnnounce >= $@InvasionAnnounceLocInterval) {
		// Get all monster GIDs on map (for location lookups), filter to invasion mobs below
		.@loc_count = getmapunits(BL_MOB, $@InvasionMap$, .@timer_gids[0]);
		.@locations$ = "";
		.@shown = 0;
		// Track navigation target (prioritize bosses)
		.@nav_x = 0;
		.@nav_y = 0;
		.@nav_name$ = "";
		.@nav_is_boss = 0;
		// Cap display at tracked remaining count (not raw map count) to avoid showing stale monsters
		.@max_show = (.@tracked_remaining < 5) ? .@tracked_remaining : 5;
		if ($@InvasionDebug) debugmes "[InvasionDebug] Periodic location announce (overtime): tracked=" + .@tracked_remaining + " max_show=" + .@max_show + " raw=" + .@loc_count;
		// Iterate through ALL monsters but stop once we've shown enough
		for (.@k = 0; .@k < .@loc_count && .@shown < .@max_show; .@k++) {
			.@gid = .@timer_gids[.@k];
			// Use array form of getunitdata
			deletearray .@mobdata;
			getunitdata .@gid, .@mobdata;
			.@mob_id = .@mobdata[UMOB_CLASS];
			.@mx = .@mobdata[UMOB_X];
			.@my = .@mobdata[UMOB_Y];
			// Skip monsters with invalid coordinates (teleporting MVPs may have 0,0)
			if (.@mx == 0 && .@my == 0) continue;
			// Check if this monster is part of our invasion theme (filter out non-invasion mobs)
			// IMPORTANT: Clear variable first - query_sql doesn't clear it when no rows returned
			.@mname$ = "";
			.@mtier$ = "";
			query_sql "SELECT mob_name, tier FROM event_invasion_monsters WHERE theme_id = " + $@InvasionTheme + " AND mob_id = " + .@mob_id + " LIMIT 1", .@mname$, .@mtier$;
			if (.@mname$ == "") continue;
			if (.@shown > 0) .@locations$ = .@locations$ + ", ";
			.@locations$ = .@locations$ + .@mname$ + " (" + .@mx + ", " + .@my + ")";
			// Track nav target - prioritize bosses (miniboss/mvp tier)
			.@is_boss = (.@mtier$ == "miniboss" || .@mtier$ == "mvp");
			if (.@nav_x == 0 || (.@is_boss && !.@nav_is_boss)) {
				.@nav_x = .@mx;
				.@nav_y = .@my;
				.@nav_name$ = .@mname$;
				.@nav_is_boss = .@is_boss;
			}
			.@shown++;
		}
		if (.@locations$ != "") {
			// Build message with boss/mob breakdown
			.@timer_boss_rem = $@InvasionWaveBossSpawns - $@InvasionWaveBossKills;
			.@timer_reg_rem = .@tracked_remaining - .@timer_boss_rem;
			if (.@timer_reg_rem < 0) .@timer_reg_rem = 0;
			.@loc_msg$ = "[Invasion] " + .@tracked_remaining + "/" + $@InvasionWaveMonsters + " remaining";
			if (.@timer_boss_rem > 0 || .@timer_reg_rem > 0) {
				.@loc_msg$ = .@loc_msg$ + " (";
				if (.@timer_boss_rem > 0) {
					.@loc_msg$ = .@loc_msg$ + .@timer_boss_rem + " boss";
					if (.@timer_boss_rem != 1) .@loc_msg$ = .@loc_msg$ + "es";
				}
				if (.@timer_boss_rem > 0 && .@timer_reg_rem > 0) .@loc_msg$ = .@loc_msg$ + ", ";
				if (.@timer_reg_rem > 0) {
					.@loc_msg$ = .@loc_msg$ + .@timer_reg_rem + " mob";
					if (.@timer_reg_rem != 1) .@loc_msg$ = .@loc_msg$ + "s";
				}
				.@loc_msg$ = .@loc_msg$ + ")";
			}
			.@loc_msg$ = .@loc_msg$ + ": " + .@locations$;
			mapannounce $@InvasionMap$, .@loc_msg$, bc_map, 0xFFAA00;

			// Auto-navigate all players on map to the target (prioritizes bosses)
			if (.@nav_x > 0) {
				.@pc_count = getmapunits(BL_PC, $@InvasionMap$, .@nav_pcs[0]);
				for (.@p = 0; .@p < .@pc_count; .@p++) {
					if (attachrid(.@nav_pcs[.@p])) {
						navigateto $@InvasionMap$, .@nav_x, .@nav_y;
						dispbottom "[Invasion] Auto-navigating to " + .@nav_name$ + "...", 0x00CCFF;
					}
				}
				// Detach to return to NPC context
				detachrid;
			}
		}
		$@InvasionLastLocAnnounce = gettimetick(2);
	}

	// Check for periodic reminder (every 5 min when no bosses)
	if (gettimetick(2) - $@InvasionLastAnnounce >= 300) {
		.@time_remaining = ($@InvasionEndTime - gettimetick(2)) / 60;
		if (.@time_remaining > 0 && .@tracked_remaining > 0) {
			// Use tracked count for accurate display
			announce "[TownInvasion] " + $@InvasionThemeName$ + " in " + $@InvasionTownName$ + "! " + .@time_remaining + " min left. " + .@tracked_remaining + " monsters remaining!", bc_all, 0xFF8800;
			$@InvasionLastAnnounce = gettimetick(2);
		}
	}

	// Continue polling
	initnpctimer;
	end;

// ============================================================
// SCHEDULED INVASION (called by EventScheduler)
// Can be triggered by:
// - Random auto trigger (S_CheckInvasionTrigger) - respects cooldown/time window/players
// - Scheduled trigger (S_CheckScheduledEvents) - bypasses cooldown/time window/players
// ============================================================
OnScheduledInvasion:
	// Check if already active
	if ($@InvasionActive) {
		debugmes "[TownInvasion] Scheduled trigger skipped - invasion already active";
		end;
	}

	// Check if enabled
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_enabled'", .@enabled$;
	if (.@enabled$ != "1") {
		debugmes "[TownInvasion] Scheduled trigger skipped - disabled";
		end;
	}

	// Get trigger type for logging
	.@trigger$ = $@InvasionTriggerType$;
	if (.@trigger$ == "") .@trigger$ = "auto";
	debugmes "[TownInvasion] Trigger type: " + .@trigger$;

	// Use pre-set parameters if available, otherwise randomize
	if ($@InvasionPendingTown > 0) {
		// Use parameters set by GM menu or scheduler
		.@threat = $@InvasionPendingThreat;
		.@town_id = $@InvasionPendingTown;
		.@theme_id = $@InvasionPendingTheme;
		debugmes "[TownInvasion] Using preset params - Threat:" + .@threat + " Town:" + .@town_id + " Theme:" + .@theme_id;

		// If threat/theme is 0, use random
		if (.@threat < 0 || .@threat > 4) {
			.@threat = rand(0, 2);
		}
		if (.@theme_id < 1) {
			query_sql "SELECT id FROM event_invasion_themes ORDER BY RAND() LIMIT 1", .@theme_id;
		}
	} else {
		// Random selection for auto triggers
		// Select random threat level (0-3, God=4 is GM-only)
		query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_max_threat'", .@max_threat$;
		.@max = (.@max_threat$ != "") ? atoi(.@max_threat$) : 3;
		if (.@max > 3) .@max = 3; // Cap at Dragon for auto
		.@threat = rand(0, .@max);

		// Select random town
		query_sql "SELECT id FROM event_invasion_towns WHERE enabled = 1 ORDER BY RAND() LIMIT 1", .@town_id;

		// Select random theme
		query_sql "SELECT id FROM event_invasion_themes ORDER BY RAND() LIMIT 1", .@theme_id;
	}

	if (.@town_id == 0) {
		debugmes "[TownInvasion] No enabled towns found!";
		end;
	}

	// Check minimum player requirements for threat levels (configurable)
	// Default: Wolf/Tiger = 1, Demon = 3, Dragon = 5, God = 10
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_threat_min_players_" + .@threat + "'", .@min_players$;
	if (.@min_players$ == "") {
		// Default minimums by threat level
		if (.@threat == 0) .@min_players = 1;       // Wolf
		else if (.@threat == 1) .@min_players = 1;  // Tiger
		else if (.@threat == 2) .@min_players = 3;  // Demon
		else if (.@threat == 3) .@min_players = 5;  // Dragon
		else .@min_players = 10;                     // God
	} else {
		.@min_players = atoi(.@min_players$);
	}

	.@online_players = getusers(1);

	// For auto/scheduled triggers, downgrade threat if not enough players
	if (.@trigger$ != "gm" && .@online_players < .@min_players) {
		debugmes "[TownInvasion] Not enough players (" + .@online_players + "/" + .@min_players + ") for threat " + .@threat + ", downgrading...";
		// Find highest threat level we can support
		.@original_threat = .@threat;
		while (.@threat > 0 && .@online_players < .@min_players) {
			.@threat--;
			query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_threat_min_players_" + .@threat + "'", .@min_players$;
			if (.@min_players$ == "") {
				if (.@threat == 0) .@min_players = 1;
				else if (.@threat == 1) .@min_players = 1;
				else if (.@threat == 2) .@min_players = 3;
				else .@min_players = 5;
			} else {
				.@min_players = atoi(.@min_players$);
			}
		}
		if (.@threat != .@original_threat) {
			debugmes "[TownInvasion] Downgraded from threat " + .@original_threat + " to " + .@threat;
		}
	}

	// Actually start it
	callsub S_BeginInvasion, .@threat, .@town_id, .@theme_id, .@trigger$;
	end;

// ============================================================
// GM MANUAL TRIGGER (routes through OnScheduledInvasion)
// ============================================================
OnGMStartInvasion:
	// GM sets $@InvasionPendingThreat, $@InvasionPendingTown, $@InvasionPendingTheme
	// Then routes through the common path with "gm" trigger type
	$@InvasionTriggerType$ = "gm";
	donpcevent "TownInvasion::OnScheduledInvasion";
	end;

// ============================================================
// BEGIN INVASION
// ============================================================
S_BeginInvasion:
	.@threat = getarg(0);
	.@town_id = getarg(1);
	.@theme_id = getarg(2);
	.@trigger$ = getarg(3, "auto");

	debugmes "[TownInvasion] S_BeginInvasion - threat:" + .@threat + " town_id:" + .@town_id + " theme_id:" + .@theme_id + " trigger:" + .@trigger$;

	// Get wave count for this threat level
	query_sql "SELECT MAX(wave_num) FROM event_invasion_wave_config WHERE threat_level = " + .@threat, .@total_waves;

	// Create invasion instance in DB
	query_sql "INSERT INTO event_invasion_instances (theme_id, town_id, threat_level, status, current_wave, total_waves, started_at, triggered_by) VALUES (" + .@theme_id + ", " + .@town_id + ", " + .@threat + ", 'active', 1, " + .@total_waves + ", NOW(), '" + .@trigger$ + "')";

	debugmes "[TownInvasion] S_BeginInvasion - Created DB record, loading invasion data...";

	// Load and start the invasion
	callsub S_LoadAndStartInvasion;

	// Set up timers AFTER returning from nested callsub (addtimer fails in nested calls)
	debugmes "[TownInvasion] S_BeginInvasion - Setting up timers";
	callsub S_SetupTimers;
	debugmes "[TownInvasion] S_BeginInvasion - Complete";
	return;

// ============================================================
// LOAD AND START INVASION
// ============================================================
S_LoadAndStartInvasion:
	// Load active invasion from database
	query_sql "SELECT i.id, t.map_name, i.theme_id, i.threat_level, i.current_wave, i.total_waves, i.monsters_spawned, i.monsters_killed, UNIX_TIMESTAMP(i.started_at), UNIX_TIMESTAMP(DATE_ADD(i.started_at, INTERVAL (SELECT config_value FROM event_config WHERE config_key = 'invasion_duration_minutes') MINUTE)), t.center_x, t.center_y, t.spawn_radius, t.safe_x, t.safe_y, t.display_name, th.name, th.icon FROM event_invasion_instances i JOIN event_invasion_towns t ON i.town_id = t.id JOIN event_invasion_themes th ON i.theme_id = th.id WHERE i.status = 'active' ORDER BY i.id DESC LIMIT 1", .@id, .@map$, .@theme, .@threat, .@wave, .@total, .@spawned, .@killed, .@start, .@end, .@cx, .@cy, .@radius, .@safe_x, .@safe_y, .@town_name$, .@theme_name$, .@theme_icon$;

	if (.@id == 0) {
		debugmes "[TownInvasion] S_LoadAndStartInvasion - No active invasion found in DB!";
		return;
	}

	debugmes "[TownInvasion] S_LoadAndStartInvasion - Loaded invasion " + .@id + " from DB";
	debugmes "[TownInvasion] S_LoadAndStartInvasion - Map:" + .@map$ + " Center:" + .@cx + "," + .@cy + " Radius:" + .@radius;

	// Set global state
	$@InvasionActive = 1;
	$@InvasionID = .@id;
	$@InvasionMap$ = .@map$;
	$@InvasionTheme = .@theme;
	$@InvasionThreat = .@threat;
	$@InvasionWave = .@wave;
	$@InvasionTotalWaves = .@total;
	$@InvasionMonstersSpawned = .@spawned;
	$@InvasionMonstersKilled = .@killed;
	$@InvasionStartTime = .@start;
	$@InvasionEndTime = .@end;
	$@InvasionLastAnnounce = gettimetick(2);
	$@InvasionCenterX = .@cx;
	$@InvasionCenterY = .@cy;
	$@InvasionRadius = .@radius;
	$@InvasionSafeX = .@safe_x;
	$@InvasionSafeY = .@safe_y;
	$@InvasionTownName$ = .@town_name$;
	$@InvasionThemeName$ = .@theme_name$;
	$@InvasionThemeIcon$ = .@theme_icon$;

	debugmes "[TownInvasion] S_LoadAndStartInvasion - Globals set, setting mapflag...";

	// Clean up any leftover monsters from previous invasions
	killmonsterall $@InvasionMap$;

	// Set invasion mapflags (no death penalty + allow teleport for mobility)
	setmapflag $@InvasionMap$, mf_nopenalty;
	removemapflag $@InvasionMap$, mf_noteleport;

	// Get duration for announcements
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_duration_minutes'", .@duration$;
	.@duration = (.@duration$ != "") ? atoi(.@duration$) : 45;

	// Announce invasion start
	.@threat_name$ = callsub(S_GetThreatName, .@threat);
	announce "[TownInvasion] " + .@threat_name$ + " threat! " + .@theme_name$ + " attacking " + .@town_name$ + "!", bc_all, 0xFF4444;
	announce "[TownInvasion] Defend for " + .@duration + " minutes! No EXP loss + Teleport enabled!", bc_all, 0xFF8800;

	// Queue Discord webhook
	callsub S_QueueDiscordStart;

	debugmes "[TownInvasion] Started invasion " + $@InvasionID + " in " + .@town_name$;

	// Update event_instances status to 'active'
	// Use pending instance ID if passed from scheduler, otherwise find the most recent scheduled record
	if ($@InvasionPendingInstanceId > 0) {
		.@ei_id = $@InvasionPendingInstanceId;
		$@InvasionPendingInstanceId = 0;  // Clear for next time
	} else {
		query_sql "SELECT ei.id FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'invasion' AND ei.status = 'scheduled' ORDER BY ei.id DESC LIMIT 1", .@ei_id;
	}
	if (.@ei_id > 0) {
		query_sql "UPDATE event_instances SET status = 'active', actual_start = NOW() WHERE id = " + .@ei_id;
		$@InvasionEventInstanceID = .@ei_id;
		debugmes "[TownInvasion] Updated event_instances " + .@ei_id + " to active";
	}

	// Schedule first wave with 15 second countdown (gives players time to prepare)
	$@InvasionNextWaveTime = gettimetick(2) + 15;
	announce "[TownInvasion] Wave 1 spawning in 15 seconds! Get ready!", bc_all, 0xFFAA00;
	debugmes "[TownInvasion] First wave scheduled at " + $@InvasionNextWaveTime;
	return;

// ============================================================
// SET UP TIMERS (starts NPC timer polling)
// ============================================================
S_SetupTimers:
	// Start the NPC timer for polling (OnTimer10000 checks timeout/wave spawn/reminders)
	initnpctimer;
	debugmes "[TownInvasion] NPC timer started for polling";
	return;

// ============================================================
// SPAWN WAVE
// ============================================================
S_SpawnWave:
	debugmes "[TownInvasion] S_SpawnWave called - Active:" + $@InvasionActive + " Map:" + $@InvasionMap$ + " Wave:" + $@InvasionWave;
	debugmes "[TownInvasion] S_SpawnWave - Center:" + $@InvasionCenterX + "," + $@InvasionCenterY + " Radius:" + $@InvasionRadius + " Theme:" + $@InvasionTheme + " Threat:" + $@InvasionThreat;

	if (!$@InvasionActive) {
		debugmes "[TownInvasion] S_SpawnWave - NOT ACTIVE, returning";
		return;
	}

	// Get wave configuration
	query_sql "SELECT tier, count FROM event_invasion_wave_config WHERE threat_level = " + $@InvasionThreat + " AND wave_num = " + $@InvasionWave, .@tiers$, .@counts;
	debugmes "[TownInvasion] S_SpawnWave - Wave config tiers:" + getarraysize(.@tiers$);

	// Count ACTIVE players on the invasion map
	.@active_players = getmapunits(BL_PC, $@InvasionMap$, .@pc_ids[0]);
	debugmes "[TownInvasion] S_SpawnWave - Active players on map: " + .@active_players;

	// Get player-scaled monster multiplier tiers (configurable)
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_player_tier1_max'", .@pt1_max$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_player_tier1_monster_mult'", .@pt1_mult$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_player_tier2_max'", .@pt2_max$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_player_tier2_monster_mult'", .@pt2_mult$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_player_tier3_monster_mult'", .@pt3_mult$;

	.@pt1_max = (.@pt1_max$ != "") ? atoi(.@pt1_max$) : 5;      // 1-5 players
	.@pt1_mult = (.@pt1_mult$ != "") ? atoi(.@pt1_mult$) : 100; // 100% base
	.@pt2_max = (.@pt2_max$ != "") ? atoi(.@pt2_max$) : 15;     // 6-15 players
	.@pt2_mult = (.@pt2_mult$ != "") ? atoi(.@pt2_mult$) : 150; // 150%
	.@pt3_mult = (.@pt3_mult$ != "") ? atoi(.@pt3_mult$) : 200; // 16+ = 200%

	// Determine multiplier based on active player count
	if (.@active_players <= .@pt1_max) {
		.@player_mult = .@pt1_mult;
	} else if (.@active_players <= .@pt2_max) {
		.@player_mult = .@pt2_mult;
	} else {
		.@player_mult = .@pt3_mult;
	}

	// Get base monster multiplier from config (can be overridden by GM)
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_monster_multiplier'", .@mult$;
	.@base_multiplier = (.@mult$ != "") ? atoi(.@mult$) : 100;

	// Combined multiplier: base × player-scaled
	.@multiplier = (.@base_multiplier * .@player_mult) / 100;
	if (.@multiplier < 10) .@multiplier = 10;   // Minimum 10%
	if (.@multiplier > 1000) .@multiplier = 1000; // Maximum 10x

	debugmes "[TownInvasion] S_SpawnWave - Player mult: " + .@player_mult + "%, Base mult: " + .@base_multiplier + "%, Combined: " + .@multiplier + "%";

	// Get separate boss multiplier for miniboss/MVP tiers
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_boss_multiplier'", .@boss_mult$;
	.@boss_multiplier = (.@boss_mult$ != "") ? atoi(.@boss_mult$) : 100;
	if (.@boss_multiplier < 10) .@boss_multiplier = 10;   // Minimum 10%
	if (.@boss_multiplier > 1000) .@boss_multiplier = 1000; // Maximum 10x

	debugmes "[TownInvasion] S_SpawnWave - Monster mult: " + .@multiplier + "%, Boss mult: " + .@boss_multiplier + "%";

	.@total_spawned = 0;
	.@boss_spawned = 0;    // MVP + miniboss count
	.@regular_spawned = 0; // Everything else
	.@miniboss_idx = 0;    // Index for miniboss names array
	.@mvp_idx = 0;         // Index for MVP names array

	// Calculate spawn area (only walkable cells will be used by areamonster)
	.@x1 = $@InvasionCenterX - $@InvasionRadius;
	.@y1 = $@InvasionCenterY - $@InvasionRadius;
	.@x2 = $@InvasionCenterX + $@InvasionRadius;
	.@y2 = $@InvasionCenterY + $@InvasionRadius;
	debugmes "[TownInvasion] S_SpawnWave - Spawn area: (" + .@x1 + "," + .@y1 + ") to (" + .@x2 + "," + .@y2 + ")";

	for (.@i = 0; .@i < getarraysize(.@tiers$); .@i++) {
		.@tier$ = .@tiers$[.@i];
		.@base_count = .@counts[.@i];

		// Track if this tier is a "boss" tier (required kills)
		.@is_boss_tier = (.@tier$ == "mvp" || .@tier$ == "miniboss");

		// Apply appropriate multiplier - boss multiplier for miniboss/MVP, regular for others
		if (.@is_boss_tier) {
			.@count = (.@base_count * .@boss_multiplier) / 100;
		} else {
			.@count = (.@base_count * .@multiplier) / 100;
		}
		if (.@count < 1 && .@base_count > 0) .@count = 1; // At least 1 if base > 0

		// Get random monsters of this tier from the theme
		// IMPORTANT: Clear arrays first to prevent leftover values from previous tier iterations
		deletearray .@mob_ids;
		deletearray .@mob_names$;
		query_sql "SELECT mob_id, mob_name FROM event_invasion_monsters WHERE theme_id = " + $@InvasionTheme + " AND tier = '" + .@tier$ + "' ORDER BY RAND()", .@mob_ids, .@mob_names$;

		if (getarraysize(.@mob_ids) == 0) {
			debugmes "[TownInvasion] Warning: No monsters found for tier " + .@tier$ + " in theme " + $@InvasionTheme;
			continue;
		}

		// Distribute spawns across available monsters of this tier
		// Use areamonster to ensure monsters spawn on WALKABLE cells only
		.@num_types = getarraysize(.@mob_ids);
		.@per_type = .@count / .@num_types;
		.@remainder = .@count % .@num_types;

		for (.@j = 0; .@j < .@num_types; .@j++) {
			.@mob = .@mob_ids[.@j];
			.@name$ = .@mob_names$[.@j];
			.@spawn_count = .@per_type;
			if (.@j < .@remainder) .@spawn_count++;

			if (.@spawn_count > 0) {
				// Debug: log boss/miniboss/mvp spawns
				if ($@InvasionDebug && .@is_boss_tier) {
					debugmes "[TownInvasion] Spawning " + .@tier$ + ": " + .@spawn_count + "x " + .@name$ + " (ID:" + .@mob + ")";
				}
				// areamonster spawns on walkable cells only, preventing stuck monsters
				areamonster $@InvasionMap$, .@x1, .@y1, .@x2, .@y2, .@name$, .@mob, .@spawn_count, "TownInvasion::OnMonsterKilled";
				.@total_spawned += .@spawn_count;

				// Track boss vs regular spawns
				if (.@is_boss_tier) {
					.@boss_spawned += .@spawn_count;
					// Track boss names and counts for announcements
					if (.@tier$ == "miniboss") {
						.@miniboss_names$[.@miniboss_idx] = .@name$;
						.@miniboss_counts[.@miniboss_idx] = .@spawn_count;
						.@miniboss_idx++;
					} else if (.@tier$ == "mvp") {
						.@mvp_names$[.@mvp_idx] = .@name$;
						.@mvp_counts[.@mvp_idx] = .@spawn_count;
						.@mvp_idx++;
					}
				} else {
					.@regular_spawned += .@spawn_count;
				}
			}
		}
	}

	$@InvasionMonstersSpawned += .@total_spawned;
	$@InvasionWaveMonsters = .@total_spawned;
	$@InvasionWaveKills = 0;
	// Track boss vs regular for wave clear logic
	$@InvasionWaveBossSpawns = .@boss_spawned;
	$@InvasionWaveBossKills = 0;
	$@InvasionWaveRegularSpawns = .@regular_spawned;
	$@InvasionWaveRegularKills = 0;
	$@InvasionLastBossGather = gettimetick(2);  // Reset boss gather timer for new wave
	debugmes "[TownInvasion] S_SpawnWave - Bosses: " + .@boss_spawned + ", Regular: " + .@regular_spawned;

	// Make all spawned monsters aggressive using getmapunits
	// Mode flags: MD_CANMOVE(0x1) + MD_AGGRESSIVE(0x4) + MD_CANATTACK(0x80) + MD_CASTSENSOR_IDLE(0x10)
	// = 0x95 = basic aggressive monster that moves, attacks, and detects players
	// Note: getmapunits requires array[0] syntax
	.@mob_count = getmapunits(BL_MOB, $@InvasionMap$, .@gids[0]);
	debugmes "[TownInvasion] Found " + .@mob_count + " monsters on map, making aggressive";
	for (.@k = 0; .@k < .@mob_count; .@k++) {
		// Set mode directly - 0x95 = can move + aggressive + can attack + cast sensor
		setunitdata .@gids[.@k], UMOB_MODE, 0x95;
	}

	// Update database
	query_sql "UPDATE event_invasion_instances SET monsters_spawned = " + $@InvasionMonstersSpawned + ", current_wave = " + $@InvasionWave + " WHERE id = " + $@InvasionID;

	announce "[TownInvasion] Wave " + $@InvasionWave + "/" + $@InvasionTotalWaves + " begins! " + .@total_spawned + " monsters spawning in " + $@InvasionTownName$ + "!", bc_all, 0xFFAA00;

	// Announce minibosses with excitement!
	if (.@miniboss_idx > 0) {
		.@miniboss_list$ = "";
		.@total_minibosses = 0;
		for (.@k = 0; .@k < .@miniboss_idx; .@k++) {
			if (.@k > 0) .@miniboss_list$ = .@miniboss_list$ + ", ";
			if (.@miniboss_counts[.@k] > 1) {
				.@miniboss_list$ = .@miniboss_list$ + .@miniboss_counts[.@k] + "x " + .@miniboss_names$[.@k];
			} else {
				.@miniboss_list$ = .@miniboss_list$ + .@miniboss_names$[.@k];
			}
			.@total_minibosses += .@miniboss_counts[.@k];
		}
		if (.@total_minibosses == 1) {
			announce "[TownInvasion] << MINIBOSS ALERT >> " + .@miniboss_list$ + " has joined the invasion!", bc_all, 0xFF6600;
		} else {
			announce "[TownInvasion] << MINIBOSS ALERT >> " + .@miniboss_list$ + " have joined the invasion!", bc_all, 0xFF6600;
		}
	}

	// Announce MVPs with maximum hype!
	if (.@mvp_idx > 0) {
		.@mvp_list$ = "";
		.@total_mvps = 0;
		for (.@k = 0; .@k < .@mvp_idx; .@k++) {
			if (.@k > 0) .@mvp_list$ = .@mvp_list$ + ", ";
			if (.@mvp_counts[.@k] > 1) {
				.@mvp_list$ = .@mvp_list$ + .@mvp_counts[.@k] + "x " + .@mvp_names$[.@k];
			} else {
				.@mvp_list$ = .@mvp_list$ + .@mvp_names$[.@k];
			}
			.@total_mvps += .@mvp_counts[.@k];
		}
		if (.@total_mvps == 1) {
			announce "[TownInvasion] !! MVP SIGHTED !! " + .@mvp_list$ + " is leading the assault! HEROES NEEDED!", bc_all, 0xFF0000;
		} else {
			announce "[TownInvasion] !!! MULTIPLE MVPs !!! " + .@mvp_list$ + " are leading the assault! ALL HANDS ON DECK!", bc_all, 0xFF0000;
		}
	}

	$@InvasionLastAnnounce = gettimetick(2);
	$@InvasionWaveStartTime = gettimetick(2);  // Track when wave started for overtime detection
	$@InvasionWaveOvertimeAnnounced = 0;       // Reset overtime announcement flag for new wave

	debugmes "[TownInvasion] Spawned wave " + $@InvasionWave + " with " + .@total_spawned + " monsters (made aggressive)";
	return;

// ============================================================
// MONSTER KILLED CALLBACK
// ============================================================
OnMonsterKilled:
	if (!$@InvasionActive) end;

	// When killmonster is called (wave clear/invasion end), this triggers
	// for each monster but without a player attached. Skip player logic.
	if (!playerattached()) {
		// Still count the kill for wave tracking
		$@InvasionMonstersKilled++;
		$@InvasionWaveKills++;
		end;
	}

	if (strcharinfo(3) != $@InvasionMap$) end;

	$@InvasionMonstersKilled++;
	$@InvasionWaveKills++;

	// Track player contribution
	.@char_id = getcharid(0);
	.@char_name$ = strcharinfo(0);

	// Give Hunt Points for kill - scales with monster tier AND threat level
	// Tier base (stored as 10x for decimal support): weak=2, medium=2, strong=5, elite=7, miniboss=25, boss=40, mvp=125
	// These represent 0.2, 0.2, 0.5, 0.7, 2.5, 4, 12.5 actual points (quartered from original)
	// Threat multiplier: Wolf=1.0, Tiger=1.25, Demon=1.5, Dragon=2.0, God=3.0
	.@mob_id = killedrid;
	.@tier_pts = 2;  // default (weak/medium) = 0.2 pts
	query_sql "SELECT tier FROM event_invasion_monsters WHERE theme_id = " + $@InvasionTheme + " AND mob_id = " + .@mob_id + " LIMIT 1", .@tier$;
	if (.@tier$ == "strong") .@tier_pts = 5;       // 0.5 pt
	else if (.@tier$ == "elite") .@tier_pts = 7;   // 0.7 pts
	else if (.@tier$ == "miniboss") .@tier_pts = 25;  // 2.5 pts
	else if (.@tier$ == "boss") .@tier_pts = 40;   // 4 pts
	else if (.@tier$ == "mvp") .@tier_pts = 125;   // 12.5 pts

	// Track boss vs regular kills for wave clear logic
	if (.@tier$ == "mvp" || .@tier$ == "miniboss") {
		$@InvasionWaveBossKills++;
	} else {
		$@InvasionWaveRegularKills++;
	}

	// Apply threat multiplier (100=1x, 125=1.25x, etc)
	setarray .@threat_mult, 100, 125, 150, 200, 300;  // Wolf, Tiger, Demon, Dragon, God
	.@kill_pts = (.@tier_pts * .@threat_mult[$@InvasionThreat]) / 100;

	// Apply global point multiplier (configurable for easy tuning)
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_point_multiplier'", .@pt_mult$;
	.@point_mult = (.@pt_mult$ != "") ? atoi(.@pt_mult$) : 100;
	.@kill_pts = (.@kill_pts * .@point_mult) / 100;

	if (.@kill_pts < 1) .@kill_pts = 1;

	// Track contribution and points earned (NO immediate reward - claimed later)
	// Points stored as 10x for decimal support
	query_sql "INSERT INTO event_invasion_contributions (invasion_id, char_id, char_name, kills, waves_participated, points_earned) VALUES (" + $@InvasionID + ", " + .@char_id + ", '" + escape_sql(.@char_name$) + "', 1, 1, " + .@kill_pts + ") ON DUPLICATE KEY UPDATE kills = kills + 1, points_earned = points_earned + " + .@kill_pts;

	// Show pending points with decimals (stored as 10x)
	query_sql "SELECT points_earned FROM event_invasion_contributions WHERE invasion_id = " + $@InvasionID + " AND char_id = " + .@char_id, .@pending;
	// Format as X.Y for both kill pts and pending
	.@kill_whole = .@kill_pts / 10;
	.@kill_dec = .@kill_pts % 10;
	.@pend_whole = .@pending / 10;
	.@pend_dec = .@pending % 10;
	dispbottom "[Invasion] +" + .@kill_whole + "." + .@kill_dec + " pts | Pending: " + .@pend_whole + "." + .@pend_dec, 0x00CC00;

	// Map-local announcement for remaining monsters (every 5 kills, or every kill when < 10)
	// Use tracked counts instead of getmapunits for accuracy (avoids counting stray mobs)
	// Note: $@InvasionWaveKills was already incremented above, so this gives remaining AFTER this kill
	.@remaining = $@InvasionWaveMonsters - $@InvasionWaveKills;
	if (.@remaining < 0) .@remaining = 0;
	.@display_remaining = .@remaining;

	// Get GID array for location lookups (still needed for coordinates)
	.@killed_gid = killedgid;
	.@raw_count = getmapunits(BL_MOB, $@InvasionMap$, .@rem_gids[0]);

	if ($@InvasionDebug) debugmes "[InvasionDebug] OnKill: WaveKills=" + $@InvasionWaveKills + " WaveMonsters=" + $@InvasionWaveMonsters + " remaining=" + .@remaining + " raw_count=" + .@raw_count;
	.@boss_remaining = $@InvasionWaveBossSpawns - $@InvasionWaveBossKills;
	.@regular_remaining = .@remaining - .@boss_remaining;
	if (.@regular_remaining < 0) .@regular_remaining = 0;

	// Check if wave is overtime
	.@wave_overtime = ($@InvasionWaveStartTime > 0 && gettimetick(2) - $@InvasionWaveStartTime >= $@InvasionWaveOvertimeThreshold);

	// Global announcement when overtime starts (only once per wave)
	if (.@wave_overtime && !$@InvasionWaveOvertimeAnnounced) {
		$@InvasionWaveOvertimeAnnounced = 1;
		// Build message with boss/mob breakdown
		.@ot_msg$ = "[TownInvasion] Wave " + $@InvasionWave + " OVERTIME! " + .@remaining + " remaining";
		if (.@boss_remaining > 0 || .@regular_remaining > 0) {
			.@ot_msg$ = .@ot_msg$ + " (";
			if (.@boss_remaining > 0) {
				.@ot_msg$ = .@ot_msg$ + .@boss_remaining + " boss";
				if (.@boss_remaining != 1) .@ot_msg$ = .@ot_msg$ + "es";
			}
			if (.@boss_remaining > 0 && .@regular_remaining > 0) .@ot_msg$ = .@ot_msg$ + ", ";
			if (.@regular_remaining > 0) {
				.@ot_msg$ = .@ot_msg$ + .@regular_remaining + " mob";
				if (.@regular_remaining != 1) .@ot_msg$ = .@ot_msg$ + "s";
			}
			.@ot_msg$ = .@ot_msg$ + ")";
		}
		.@ot_msg$ = .@ot_msg$ + " - Locations revealed every " + $@InvasionAnnounceLocInterval + "s!";
		announce .@ot_msg$, bc_all, 0xFF6600;
	}

	// Calculate 50% threshold for per-kill announcements
	.@announce_threshold = ($@InvasionWaveMonsters * $@InvasionAnnounceCountPercent) / 100;
	if (.@announce_threshold < 1) .@announce_threshold = 1;

	// Announce every X kills OR every kill when under percentage threshold
	if (.@remaining > 0 && (.@remaining <= .@announce_threshold || $@InvasionWaveKills % $@InvasionAnnounceEvery == 0)) {
		// Show locations only when wave is overtime - helps find stragglers
		if (.@wave_overtime) {
			.@locations$ = "";
			.@shown = 0;
			// Cap display at tracked remaining count (not raw map count) to avoid showing stale monsters
			.@max_show = (.@remaining < 5) ? .@remaining : 5;
			if ($@InvasionDebug) debugmes "[InvasionDebug] Building locations for " + .@remaining + " monsters, max_show=" + .@max_show + ", iterating " + .@raw_count + " entries";
			for (.@k = 0; .@k < .@raw_count && .@shown < .@max_show; .@k++) {
				.@gid = .@rem_gids[.@k];
				if ($@InvasionDebug) debugmes "[InvasionDebug] Loop " + .@k + ": gid=" + .@gid;
				// Skip the dying monster if it's still in the list
				if (.@gid == .@killed_gid) {
					if ($@InvasionDebug) debugmes "[InvasionDebug] -> Skipping (killed monster)";
					continue;
				}
				// Use array form of getunitdata to get all mob data at once
				deletearray .@mobdata;
				getunitdata .@gid, .@mobdata;
				.@mob_id = .@mobdata[UMOB_CLASS];
				.@mx = .@mobdata[UMOB_X];
				.@my = .@mobdata[UMOB_Y];
				if ($@InvasionDebug) debugmes "[InvasionDebug] -> mob_id=" + .@mob_id + " pos=(" + .@mx + "," + .@my + ")";
				// Skip monsters with invalid coordinates (dead/despawning)
				if (.@mx == 0 && .@my == 0) {
					if ($@InvasionDebug) debugmes "[InvasionDebug] -> Skipping (0,0 coords)";
					continue;
				}
				// Check if this monster is part of our invasion theme (filter out non-invasion mobs)
				// IMPORTANT: Clear variable first - query_sql doesn't clear it when no rows returned
				.@mname$ = "";
				query_sql "SELECT mob_name FROM event_invasion_monsters WHERE theme_id = " + $@InvasionTheme + " AND mob_id = " + .@mob_id + " LIMIT 1", .@mname$;
				if (.@mname$ == "") {
					if ($@InvasionDebug) debugmes "[InvasionDebug] -> Skipping (not in invasion theme)";
					continue;
				}
				if (.@shown > 0) .@locations$ = .@locations$ + ", ";
				.@locations$ = .@locations$ + .@mname$ + " (" + .@mx + ", " + .@my + ")";
				.@shown++;
			}
			if ($@InvasionDebug) debugmes "[InvasionDebug] Final locations$: " + .@locations$;
			// Build message with boss/mob breakdown
			.@loc_msg$ = "[Invasion] " + .@display_remaining + "/" + $@InvasionWaveMonsters + " remaining";
			if (.@boss_remaining > 0 || .@regular_remaining > 0) {
				.@loc_msg$ = .@loc_msg$ + " (";
				if (.@boss_remaining > 0) {
					.@loc_msg$ = .@loc_msg$ + .@boss_remaining + " boss";
					if (.@boss_remaining != 1) .@loc_msg$ = .@loc_msg$ + "es";
				}
				if (.@boss_remaining > 0 && .@regular_remaining > 0) .@loc_msg$ = .@loc_msg$ + ", ";
				if (.@regular_remaining > 0) {
					.@loc_msg$ = .@loc_msg$ + .@regular_remaining + " mob";
					if (.@regular_remaining != 1) .@loc_msg$ = .@loc_msg$ + "s";
				}
				.@loc_msg$ = .@loc_msg$ + ")";
			}
			.@loc_msg$ = .@loc_msg$ + ": " + .@locations$;
			mapannounce $@InvasionMap$, .@loc_msg$, bc_map, 0xFFAA00;
		} else {
			.@msg$ = "[Invasion] " + .@display_remaining + "/" + $@InvasionWaveMonsters + " monsters remaining";
			if (.@boss_remaining > 0) {
				.@msg$ = .@msg$ + " (" + .@boss_remaining + " boss";
				if (.@boss_remaining != 1) .@msg$ = .@msg$ + "es";
				if (.@regular_remaining > 0) {
					.@msg$ = .@msg$ + ", " + .@regular_remaining + " mob";
					if (.@regular_remaining != 1) .@msg$ = .@msg$ + "s";
				}
				.@msg$ = .@msg$ + ")";
			}
			mapannounce $@InvasionMap$, .@msg$, bc_map, 0xFFAA00;
		}
	}

	// Party kill sharing - give credit to nearby party members
	.@killer_aid = getcharid(3);  // Save killer's account ID before loop (detachrid clears attachment)
	if (getcharid(1) > 0) {
		// Get party members on same map within range
		getpartymember getcharid(1), 0;
		getpartymember getcharid(1), 1;
		getpartymember getcharid(1), 2;
		for (.@i = 0; .@i < $@partymembercount; .@i++) {
			if ($@partymemberaid[.@i] != .@killer_aid && isloggedin($@partymemberaid[.@i])) {
				// Check if on same map
				if (attachrid($@partymemberaid[.@i])) {
					if (strcharinfo(3) == $@InvasionMap$) {
						// Check distance (within 15 cells)
						// Note: We can't easily check distance, so just give credit if on map
						query_sql "INSERT INTO event_invasion_contributions (invasion_id, char_id, char_name, kills, waves_participated) VALUES (" + $@InvasionID + ", " + getcharid(0) + ", '" + escape_sql(strcharinfo(0)) + "', 0, 1) ON DUPLICATE KEY UPDATE waves_participated = GREATEST(waves_participated, 1)";
					}
					detachrid;
					attachrid(.@killer_aid);  // Re-attach to killer for next iteration
				}
			}
		}
	}

	// Check wave clear conditions:
	// 1. ALL bosses (MVP + miniboss) must be killed
	// 2. 90% of regular monsters must be killed
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_wave_clear_percent'", .@threshold$;
	.@threshold = (.@threshold$ != "") ? atoi(.@threshold$) : 90;
	.@regular_needed = ($@InvasionWaveRegularSpawns * .@threshold) / 100;

	.@bosses_clear = ($@InvasionWaveBossKills >= $@InvasionWaveBossSpawns);
	.@regular_clear = ($@InvasionWaveRegularKills >= .@regular_needed);

	if (.@bosses_clear && .@regular_clear) {
		donpcevent "TownInvasion::OnWaveCleared";
	}
	end;

// ============================================================
// WAVE CLEARED
// ============================================================
OnWaveCleared:
	if (!$@InvasionActive) end;

	// IMPORTANT: Reset wave state FIRST to prevent failsafe/overtime from re-triggering
	// during the countdown period before the next wave spawns
	$@InvasionWaveMonsters = 0;
	$@InvasionWaveStartTime = 0;  // Reset to prevent overtime check during countdown

	// Clean up remaining monsters from this wave (both methods for robustness)
	killmonster $@InvasionMap$, "TownInvasion::OnMonsterKilled";
	killmonsterall $@InvasionMap$;

	// Award wave clear bonus to all participants (deferred to claim)
	// Base bonus: 50 stored = 5 pts displayed
	// Apply global point multiplier for consistency
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_point_multiplier'", .@pt_mult$;
	.@point_mult = (.@pt_mult$ != "") ? atoi(.@pt_mult$) : 100;
	.@wave_bonus = (50 * .@point_mult) / 100;
	if (.@wave_bonus < 10) .@wave_bonus = 10;  // Minimum 1 pt displayed
	.@wave_display = .@wave_bonus / 10;

	// Need account_id for attachrid, char_id for DB updates
	query_sql "SELECT c.char_id, ch.account_id FROM event_invasion_contributions c JOIN `char` ch ON c.char_id = ch.char_id WHERE c.invasion_id = " + $@InvasionID, .@char_ids, .@acc_ids;
	for (.@i = 0; .@i < getarraysize(.@char_ids); .@i++) {
		query_sql "UPDATE event_invasion_contributions SET points_earned = points_earned + " + .@wave_bonus + " WHERE invasion_id = " + $@InvasionID + " AND char_id = " + .@char_ids[.@i];
		if (attachrid(.@acc_ids[.@i])) {
			query_sql "SELECT points_earned FROM event_invasion_contributions WHERE invasion_id = " + $@InvasionID + " AND char_id = " + .@char_ids[.@i], .@pending;
			.@pend_whole = .@pending / 10;
			.@pend_dec = .@pending % 10;
			dispbottom "[Invasion] Wave Clear: +" + .@wave_display + " pts (Pending: " + .@pend_whole + "." + .@pend_dec + ")", 0x00FF00;
			detachrid;
		}
	}

	announce "[TownInvasion] Wave " + $@InvasionWave + " cleared!", bc_all, 0x00FF00;
	$@InvasionLastAnnounce = gettimetick(2);

	// Check if all waves complete
	if ($@InvasionWave >= $@InvasionTotalWaves) {
		donpcevent "TownInvasion::OnInvasionVictory";
		end;
	}

	// Start next wave after delay (using polling, not addtimer)
	$@InvasionWave++;
	$@InvasionNextWaveTime = gettimetick(2) + 15;
	announce "[TownInvasion] Wave " + $@InvasionWave + "/" + $@InvasionTotalWaves + " starting in 15 seconds...", bc_all, 0xFFAA00;
	debugmes "[TownInvasion] OnWaveCleared - next wave scheduled at " + $@InvasionNextWaveTime;
	// CRITICAL: Restart timer so OnTimer10000 fires again to check $@InvasionNextWaveTime
	initnpctimer;
	end;

OnNextWave:
	debugmes "[TownInvasion] OnNextWave triggered - Active:" + $@InvasionActive;
	if (!$@InvasionActive) {
		debugmes "[TownInvasion] OnNextWave aborted - invasion not active";
		end;
	}
	debugmes "[TownInvasion] OnNextWave calling S_SpawnWave";
	callsub S_SpawnWave;
	debugmes "[TownInvasion] OnNextWave complete";
	end;

// ============================================================
// INVASION VICTORY
// ============================================================
OnInvasionVictory:
	if (!$@InvasionActive) end;

	// Store map before any cleanup
	.@map$ = $@InvasionMap$;

	// Stop NPC timer and clear pending wave
	stopnpctimer;
	$@InvasionNextWaveTime = 0;

	// Kill any remaining monsters (both methods for robustness)
	killmonster .@map$, "TownInvasion::OnMonsterKilled";
	killmonsterall .@map$;

	// Calculate and distribute rewards
	callsub S_DistributeRewards, 1; // 1 = victory

	// Update database
	query_sql "UPDATE event_invasion_instances SET status = 'victory', ended_at = NOW(), monsters_killed = " + $@InvasionMonstersKilled + " WHERE id = " + $@InvasionID;

	// Update last ended time for cooldown
	query_sql "UPDATE event_config SET config_value = UNIX_TIMESTAMP() WHERE config_key = 'invasion_last_ended'";

	// Store values before reset
	.@town$ = $@InvasionTownName$;
	.@total = $@InvasionTotalWaves;
	.@id = $@InvasionID;

	// Announce victory
	announce "[TownInvasion] VICTORY! " + .@town$ + " DEFENDED! All " + .@total + " waves cleared!", bc_all, 0x00FF00;
	announce "[TownInvasion] Rewards distributed! Thank you brave defenders!", bc_all, 0x00FF00;

	// Queue Discord webhook
	callsub S_QueueDiscordEnd, "victory";

	// Update event_instances status
	if ($@InvasionEventInstanceID > 0) {
		query_sql "UPDATE event_instances SET status = 'completed', actual_end = NOW() WHERE id = " + $@InvasionEventInstanceID;
	}

	// Remove invasion mapflags (restore normal town behavior)
	removemapflag .@map$, mf_nopenalty;
	setmapflag .@map$, mf_noteleport;

	// Reset state
	callsub S_ResetState;

	debugmes "[TownInvasion] Invasion " + .@id + " ended in victory";
	end;

// ============================================================
// INVASION TIMEOUT (FAILURE)
// ============================================================
OnInvasionTimeout:
	if (!$@InvasionActive) end;

	// Store values before cleanup
	.@map$ = $@InvasionMap$;
	.@town$ = $@InvasionTownName$;
	.@theme$ = $@InvasionThemeName$;
	.@killed = $@InvasionMonstersKilled;
	.@id = $@InvasionID;

	// Stop NPC timer and clear pending wave
	stopnpctimer;
	$@InvasionNextWaveTime = 0;

	// Kill remaining monsters (both methods for robustness)
	killmonster .@map$, "TownInvasion::OnMonsterKilled";
	killmonsterall .@map$;

	// Distribute partial rewards (no defense bonus)
	callsub S_DistributeRewards, 0; // 0 = failure

	// Update database
	query_sql "UPDATE event_invasion_instances SET status = 'failed', ended_at = NOW(), monsters_killed = " + .@killed + " WHERE id = " + .@id;

	// Update last ended time
	query_sql "UPDATE event_config SET config_value = UNIX_TIMESTAMP() WHERE config_key = 'invasion_last_ended'";

	// Announce failure
	announce "[TownInvasion] DEFEAT! " + .@town$ + " was overwhelmed by " + .@theme$ + "...", bc_all, 0xFF4444;
	announce "[TownInvasion] Partial rewards distributed. The city will recover!", bc_all, 0xFF8800;

	// Queue Discord webhook
	callsub S_QueueDiscordEnd, "failed";

	// Update event_instances status
	if ($@InvasionEventInstanceID > 0) {
		query_sql "UPDATE event_instances SET status = 'failed', actual_end = NOW() WHERE id = " + $@InvasionEventInstanceID;
	}

	// Remove invasion mapflags (restore normal town behavior)
	removemapflag .@map$, mf_nopenalty;
	setmapflag .@map$, mf_noteleport;

	// Reset state
	callsub S_ResetState;

	debugmes "[TownInvasion] Invasion " + .@id + " timed out (failed)";
	end;

// ============================================================
// GM CANCEL INVASION
// ============================================================
OnGMCancelInvasion:
	debugmes "[TownInvasion] GM cancel triggered, $@InvasionActive=" + $@InvasionActive + ", $@InvasionID=" + $@InvasionID;

	// IMMEDIATELY set inactive to prevent race conditions with timeout handler
	// Store values first, then disable
	.@was_active = $@InvasionActive;
	.@map$ = $@InvasionMap$;
	.@id = $@InvasionID;
	.@town$ = $@InvasionTownName$;

	// Disable FIRST to prevent timeout from firing during cancel
	$@InvasionActive = 0;

	// Check both memory state and database for active invasion
	if (!.@was_active) {
		// Fallback: check database for any active invasion
		query_sql "SELECT i.id, t.map_name, t.display_name FROM event_invasion_instances i JOIN event_invasion_towns t ON i.town_id = t.id WHERE i.status = 'active' LIMIT 1", .@db_id, .@db_map$, .@db_town$;
		if (.@db_id > 0) {
			debugmes "[TownInvasion] GM cancel: memory state was stale, found active invasion " + .@db_id + " in database";
			.@map$ = .@db_map$;
			.@id = .@db_id;
			.@town$ = .@db_town$;
		} else {
			debugmes "[TownInvasion] GM cancel failed - no active invasion in memory or database";
			end;
		}
	}

	debugmes "[TownInvasion] GM cancelling invasion " + .@id + " in " + .@map$;

	// Note: We already set $@InvasionActive = 0 above, so timer handlers will exit early
	// deltimer/stopnpctimer cause issues, so we skip them
	debugmes "[TownInvasion] State disabled, proceeding with cleanup";

	// Kill monsters - try callback-specific first, then fallback to all
	debugmes "[TownInvasion] Killing monsters on " + .@map$;
	killmonster .@map$, "TownInvasion::OnMonsterKilled";
	// Fallback: kill ALL monsters on the map (catches any that slipped through)
	killmonsterall .@map$;
	debugmes "[TownInvasion] Monsters killed";

	// Update database - mark as cancelled
	debugmes "[TownInvasion] Updating database status to cancelled for ID " + .@id;
	query_sql "UPDATE event_invasion_instances SET status = 'cancelled', ended_at = NOW() WHERE id = " + .@id;
	// Also mark any other active invasions as cancelled (cleanup)
	query_sql "UPDATE event_invasion_instances SET status = 'cancelled', ended_at = NOW() WHERE status = 'active'";

	// Update last ended time for cooldown
	query_sql "UPDATE event_config SET config_value = UNIX_TIMESTAMP() WHERE config_key = 'invasion_last_ended'";

	// Announce cancellation
	announce "[TownInvasion] Event in " + .@town$ + " cancelled by a GM.", bc_all, 0xFFFF00;

	// Update event_instances status (general event tracking)
	if ($@InvasionEventInstanceID > 0) {
		query_sql "UPDATE event_instances SET status = 'cancelled', actual_end = NOW() WHERE id = " + $@InvasionEventInstanceID;
	}

	// Remove invasion mapflags (restore normal town behavior)
	removemapflag .@map$, mf_nopenalty;
	setmapflag .@map$, mf_noteleport;

	// Reset state
	callsub S_ResetState;

	debugmes "[TownInvasion] Invasion " + .@id + " cancelled by GM successfully";
	end;

// ============================================================
// DISTRIBUTE REWARDS (no multipliers - points are what you earned)
// ============================================================
S_DistributeRewards:
	.@victory = getarg(0);

	// Count defenders for display
	query_sql "SELECT COUNT(*) FROM event_invasion_contributions WHERE invasion_id = " + $@InvasionID, .@defenders;


	// Notify online players about pending rewards
	query_sql "SELECT c.char_id, c.points_earned, ch.account_id FROM event_invasion_contributions c JOIN `char` ch ON c.char_id = ch.char_id WHERE c.invasion_id = " + $@InvasionID, .@char_ids, .@points, .@acc_ids;

	for (.@i = 0; .@i < getarraysize(.@char_ids); .@i++) {
		if (attachrid(.@acc_ids[.@i])) {
			// Points stored as 10x, so divide by 10 for display
			.@raw = .@points[.@i];
			.@final = .@raw / 10;
			if (.@final < 1 && .@raw > 0) .@final = 1;  // Minimum 1 point
			if (.@victory) {
				dispbottom "[Invasion] Victory! You earned " + .@final + " Hunt Points! (" + .@defenders + " defenders)", 0x00FF00;
			} else {
				dispbottom "[Invasion] Defense ended. You earned " + .@final + " Hunt Points.", 0xFFAA00;
			}
			detachrid;
		}
	}

	debugmes "[TownInvasion] Rewards: " + .@defenders + " defenders, no multipliers applied";
	return;

// ============================================================
// RESET STATE
// ============================================================
S_ResetState:
	$@InvasionActive = 0;
	$@InvasionID = 0;
	$@InvasionMap$ = "";
	$@InvasionTheme = 0;
	$@InvasionThreat = 0;
	$@InvasionWave = 0;
	$@InvasionTotalWaves = 0;
	$@InvasionMonstersSpawned = 0;
	$@InvasionMonstersKilled = 0;
	$@InvasionStartTime = 0;
	$@InvasionEndTime = 0;
	$@InvasionNextWaveTime = 0;
	$@InvasionEventInstanceID = 0;
	$@InvasionLastBossGather = 0;
	$@InvasionLastLocAnnounce = 0;
	return;

// ============================================================
// DISCORD WEBHOOK - INVASION START
// ============================================================
S_QueueDiscordStart:
	.@threat_name$ = callsub(S_GetThreatName, $@InvasionThreat);
	.@threat_icon$ = callsub(S_GetThreatIcon, $@InvasionThreat);

	query_sql "SELECT config_value FROM event_config WHERE config_key = 'invasion_duration_minutes'", .@duration$;
	.@duration = (.@duration$ != "") ? atoi(.@duration$) : 45;

	// Build JSON payload
	.@payload$ = "{";
	.@payload$ += "\"type\":\"invasion_start\",";
	.@payload$ += "\"threat_level\":" + $@InvasionThreat + ",";
	.@payload$ += "\"threat_name\":\"" + .@threat_name$ + "\",";
	.@payload$ += "\"threat_icon\":\"" + .@threat_icon$ + "\",";
	.@payload$ += "\"theme_name\":\"" + $@InvasionThemeName$ + "\",";
	.@payload$ += "\"theme_icon\":\"" + $@InvasionThemeIcon$ + "\",";
	.@payload$ += "\"town_name\":\"" + $@InvasionTownName$ + "\",";
	.@payload$ += "\"duration\":" + .@duration;
	.@payload$ += "}";

	query_sql "INSERT INTO discord_webhook_queue (webhook_type, payload, created_at) VALUES ('invasion', '" + escape_sql(.@payload$) + "', NOW())";
	return;

// ============================================================
// DISCORD WEBHOOK - INVASION END
// ============================================================
S_QueueDiscordEnd:
	.@status$ = getarg(0);

	// Get top contributor
	query_sql "SELECT char_name, kills FROM event_invasion_contributions WHERE invasion_id = " + $@InvasionID + " ORDER BY kills DESC LIMIT 1", .@top_name$, .@top_kills;

	// Count defenders
	query_sql "SELECT COUNT(*) FROM event_invasion_contributions WHERE invasion_id = " + $@InvasionID, .@defenders;

	// Calculate waves cleared based on outcome
	// Victory: all waves cleared (current wave = total)
	// Failure: current wave was not cleared (wave - 1)
	if (.@status$ == "victory") {
		.@waves_cleared = $@InvasionWave;  // All waves completed
	} else {
		.@waves_cleared = $@InvasionWave - 1;  // Current wave failed
		if (.@waves_cleared < 0) .@waves_cleared = 0;
	}

	// Build JSON payload
	.@payload$ = "{";
	.@payload$ += "\"type\":\"invasion_end\",";
	.@payload$ += "\"status\":\"" + .@status$ + "\",";
	.@payload$ += "\"town_name\":\"" + $@InvasionTownName$ + "\",";
	.@payload$ += "\"theme_name\":\"" + $@InvasionThemeName$ + "\",";
	.@payload$ += "\"waves_cleared\":" + .@waves_cleared + ",";
	.@payload$ += "\"total_waves\":" + $@InvasionTotalWaves + ",";
	.@payload$ += "\"monsters_killed\":" + $@InvasionMonstersKilled + ",";
	.@payload$ += "\"defenders\":" + .@defenders + ",";
	.@payload$ += "\"top_defender\":\"" + escape_sql(.@top_name$) + "\",";
	.@payload$ += "\"top_kills\":" + .@top_kills;
	.@payload$ += "}";

	query_sql "INSERT INTO discord_webhook_queue (webhook_type, payload, created_at) VALUES ('invasion', '" + escape_sql(.@payload$) + "', NOW())";
	return;

// ============================================================
// HELPER FUNCTIONS
// ============================================================
S_GetThreatName:
	switch(getarg(0)) {
		case 0: return "WOLF (Easy)";
		case 1: return "TIGER (Medium)";
		case 2: return "DEMON (Hard)";
		case 3: return "DRAGON (Very Hard)";
		case 4: return "GOD (Extreme)";
	}
	return "UNKNOWN";

S_GetThreatIcon:
	switch(getarg(0)) {
		case 0: return "[WOLF - Easy]";
		case 1: return "[TIGER - Medium]";
		case 2: return "[DEMON - Hard]";
		case 3: return "[DRAGON - V.Hard]";
		case 4: return "[GOD - Extreme]";
	}
	return "[?]";

S_GetTownName:
	query_sql "SELECT display_name FROM event_invasion_towns WHERE map_name = '" + escape_sql(getarg(0)) + "'", .@name$;
	return .@name$;

}
