//===== rAthena Script =======================================
//= MOTD Kill Bonus
//= Grants bonus EXP when killing Monster of the Day
//===== Description: =========================================
//= Uses OnNPCKillEvent to give 2x additional EXP when killing
//= the current MOTD monster, resulting in 3x total EXP.
//=
//= The getexp command automatically applies server rates, so
//= bonus scales correctly during rate events (e.g., Happy Hour).
//=
//= Limitations:
//= - Only triggers for the character dealing the killing blow
//= - Bonus EXP does not share with party members
//= - For full party share support, C++ modification needed
//============================================================

-	script	MOTD_KillBonus	FAKE_NPC,{
	end;

OnInit:
	// Store original config rates before any events modify them
	// getmonsterinfo() uses these config rates, not runtime-modified rates
	set $@MOTD_OrigBaseRate, getbattleflag("base_exp_rate");
	set $@MOTD_OrigJobRate, getbattleflag("job_exp_rate");
	end;

OnNPCKillEvent:
	// Check if MOTD is enabled and killed monster is current MOTD
	// $@LastMOTDMobID is set by EventScheduler when MOTD rotates
	if (callfunc("F_IsMOTDEnabled") && killedrid == $@LastMOTDMobID && $@LastMOTDMobID > 0) {
		// Get monster's EXP values (uses CONFIG rate, not runtime rate!)
		.@base_exp = getmonsterinfo(killedrid, MOB_BASEEXP);
		.@job_exp = getmonsterinfo(killedrid, MOB_JOBEXP);

		// Convert to raw using ORIGINAL config rate (not current boosted rate)
		// This allows getexp to apply the CURRENT rate, making bonus scale with events
		.@base_raw = .@base_exp * 100 / $@MOTD_OrigBaseRate;
		.@job_raw = .@job_exp * 100 / $@MOTD_OrigJobRate;

		// Give 2x additional raw EXP (getexp applies current rate including boosts)
		// Result: normal (1x) + bonus (2x) = 3x total, scales with Happy Hour/EXP Boost
		.@base_bonus = .@base_raw * 2;
		.@job_bonus = .@job_raw * 2;

		getexp .@base_bonus, .@job_bonus;
	}
	end;
}
