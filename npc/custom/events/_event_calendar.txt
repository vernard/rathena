//===== rAthena Script =======================================
//= Events Manager NPC
//= Shows all current and upcoming server events
//= Includes GM management options for staff
//============================================================

prontera,141,172,5	script	Events Manager	4_M_MINSTREL,{
	mes "[Events Manager]";
	mes "Welcome to the Server Events!";
	mes "Here you can see all active and upcoming events.";
	next;

	// ========================================
	// CURRENTLY ACTIVE EVENTS
	// ========================================
	mes "[^FF0000Currently Active^000000]";
	set .@has_active, 0;

	// Check Happy Hour
	if ($@HappyHourActive) {
		set .@remaining, callfunc("F_HappyHourRemaining");
		mes "^FFD700[Active] Happy Hour^000000";
		mes "   2x EXP rates!";
		mes "   Time left: ^00AA00"+.@remaining+" minutes^000000";
		set .@has_active, 1;
	}

	// Check EXP Boost
	if ($@ExpBoostActive) {
		set .@mult$, callfunc("F_FormatExpMultiplier", $@ExpBoostMultiplier);
		set .@duration, callfunc("F_ExpBoostDuration");
		mes "^00BFFF[Active] EXP Boost^000000";
		mes "   "+.@mult$+" EXP rates!";
		mes "   Running for: ^00AA00"+.@duration+" minutes^000000";
		set .@has_active, 1;
	}

	// Check Monster of the Day
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@motd_enabled$;
	if (.@motd_enabled$ == "0") {
		mes "^888888[Disabled] Monster of the Day^000000";
	} else {
		query_sql "SELECT mob_name, TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@motd_name$, .@motd_mins;
		if (.@motd_name$ != "") {
			mes "^00FF00[Active] Monster of the Day^000000";
			mes "   Featured: ^0000FF"+.@motd_name$+"^000000";
			mes "   3x EXP & Hunt Points!";
			mes "   Time left: ^00AA00"+.@motd_mins+" minutes^000000";
			set .@has_active, 1;
		}
	}

	if (!.@has_active) {
		mes "^888888No events currently active.^000000";
	}
	next;

	// ========================================
	// UPCOMING EVENTS
	// ========================================
	mes "[^0055FFUpcoming Events^000000]";

	// Get scheduled Happy Hours (with duration)
	query_sql "SELECT DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), start_hour, start_minute, duration_minutes FROM event_schedule WHERE event_code = 'happy_hour' AND enabled = 1 ORDER BY day_of_week LIMIT 3", .@hh_days$, .@hh_hours, .@hh_mins, .@hh_durations;

	if (getarraysize(.@hh_days$) > 0) {
		mes "^FFD700Happy Hour Schedule:^000000";
		for (set .@i, 0; .@i < getarraysize(.@hh_days$); set .@i, .@i + 1) {
			set .@time$, (.@hh_hours[.@i] < 10 ? "0" : "") + .@hh_hours[.@i] + ":" + (.@hh_mins[.@i] < 10 ? "0" : "") + .@hh_mins[.@i];
			// Convert duration to hours/minutes display
			if (.@hh_durations[.@i] >= 60) {
				set .@dur_hours, .@hh_durations[.@i] / 60;
				set .@dur_mins, .@hh_durations[.@i] % 60;
				if (.@dur_mins > 0) {
					set .@dur$, .@dur_hours+"h "+.@dur_mins+"m";
				} else {
					set .@dur$, .@dur_hours+" hour" + (.@dur_hours > 1 ? "s" : "");
				}
			} else {
				set .@dur$, .@hh_durations[.@i]+" min";
			}
			mes "   "+.@hh_days$[.@i]+" at "+.@time$+" ("+.@dur$+")";
		}
	} else {
		mes "^888888No Happy Hour scheduled.^000000";
	}

	mes " ";
	mes "^00FF00Monster of the Day:^000000";
	mes "   Rotates every 3 hours";
	mes "   Check the Monster Hunter Board!";
	next;

	// ========================================
	// MENU (with GM options if applicable)
	// ========================================
	if (getgmlevel() >= 60) {
		switch(select("View Event Details:View My Event Stats:^FF0000[GM] Manage Events^000000:Close")) {
			case 1: goto L_EventDetails;
			case 2: goto L_MyStats;
			case 3: goto L_GMMenu;
			case 4: close;
		}
	} else {
		switch(select("View Event Details:View My Event Stats:Close")) {
			case 1: goto L_EventDetails;
			case 2: goto L_MyStats;
			case 3: close;
		}
	}
	end;

L_EventDetails:
	mes "[Event Details]";
	mes "Which event would you like to learn about?";
	next;
	switch(select("Happy Hour:Monster of the Day:Back")) {
		case 1:
			mes "[^FFD700Happy Hour^000000]";
			mes " ";
			mes "^0000FFWhat is it?^000000";
			mes "A server-wide EXP boost event!";
			mes "All players receive ^FF00002x Base EXP^000000";
			mes "and ^FF00002x Job EXP^000000 from monsters.";
			next;
			mes "[^FFD700Happy Hour^000000]";
			mes "^0000FFWhat about drops?^000000";
			mes "Drop rates stay the same to keep";
			mes "the economy balanced. This is a";
			mes "leveling event, not a farming one!";
			next;
			mes "[^FFD700Happy Hour^000000]";
			mes "^0000FFWhen does it happen?^000000";
			// Get next scheduled Happy Hour
			query_sql "SELECT DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), start_hour, start_minute, duration_minutes FROM event_schedule WHERE event_code = 'happy_hour' AND enabled = 1 ORDER BY (day_of_week - WEEKDAY(CURDATE()) + 7) % 7 LIMIT 1", .@next_day$, .@next_hour, .@next_min, .@next_dur;
			if (.@next_day$ != "") {
				set .@time$, (.@next_hour < 10 ? "0" : "") + .@next_hour + ":" + (.@next_min < 10 ? "0" : "") + .@next_min;
				if (.@next_dur >= 60) {
					set .@dur$, (.@next_dur / 60) + " hour" + ((.@next_dur / 60) > 1 ? "s" : "");
				} else {
					set .@dur$, .@next_dur + " minutes";
				}
				mes "Next: ^00AA00"+.@next_day$+" at "+.@time$+"^000000";
				mes "Duration: ^00AA00"+.@dur$+"^000000";
			} else {
				mes "No scheduled Happy Hours.";
				mes "Check back later!";
			}
			next;
			mes "[^FFD700Happy Hour^000000]";
			mes "^0000FFTips:^000000";
			mes "- Great for leveling new jobs";
			mes "- Perfect for pushing past plateaus";
			mes "- Stack with Monster of the Day!";
			close;

		case 2:
			mes "[^00FF00Monster of the Day^000000]";
			mes " ";
			mes "^0000FFWhat is it?^000000";
			mes "A rotating featured monster that";
			mes "gives ^FF00003x Base/Job EXP^000000 and bonus";
			mes "^FF0000Hunt Points^000000 when killed!";
			next;
			mes "[^00FF00Monster of the Day^000000]";
			mes "^0000FFHow does it work?^000000";
			mes "The featured monster changes";
			mes "every ^00AA003 hours^000000. Different tiers";
			mes "rotate throughout the day so";
			mes "everyone can participate.";
			next;
			mes "[^00FF00Monster of the Day^000000]";
			mes "^0000FFWhat are Hunt Points?^000000";
			mes "A currency earned by hunting";
			mes "MOTD targets. Spend them on";
			mes "special rewards like costumes,";
			mes "consumables, and rare items!";
			next;
			mes "[^00FF00Monster of the Day^000000]";
			mes "^0000FFWhere do I find it?^000000";
			mes "Check the ^0000FFMonster Hunter Board^000000";
			mes "in Prontera for the current";
			mes "target, location hints, and";
			mes "your progress!";
			close;

		case 3:
			close;
	}
	end;

L_MyStats:
	mes "[Your Event Stats]";
	// Get event coin balance
	query_sql "SELECT coins, total_earned FROM event_coins WHERE account_id = "+getcharid(3), .@coins, .@total;
	mes "Event Coins: ^0000FF"+.@coins+"^000000";
	mes "Total Earned: "+.@total;
	mes " ";
	// Get MOTD completions
	query_sql "SELECT COUNT(*) FROM motd_completions WHERE char_id = "+getcharid(0), .@motd_count;
	mes "MOTD Hunts Completed: "+.@motd_count;
	close;

// ============================================================
// GM MANAGEMENT MENU
// ============================================================
L_GMMenu:
	mes "[^FF0000GM Event Manager^000000]";
	mes "Welcome, "+strcharinfo(0)+".";
	mes "What would you like to manage?";
	next;

	switch(select("Happy Hour:EXP Boost:Monster of the Day:Regenerate MOTD Schedule:Event History:Back")) {
		case 1: goto L_HappyHour;
		case 2: goto L_ExpBoost;
		case 3: goto L_MOTD;
		case 4:
			donpcevent "EventScheduler::OnGenerateMOTDSchedule";
			callsub S_InvalidateCache;
			mes "MOTD schedule regenerated!";
			close;
		case 5: goto L_History;
		case 6: close;
	}
	end;

// ============================================================
// Happy Hour Management
// ============================================================
L_HappyHour:
	mes "[Happy Hour Management]";

	if ($@HappyHourActive) {
		set .@remaining, callfunc("F_HappyHourRemaining");
		mes "^00AA00Status: ACTIVE^000000";
		mes "Time remaining: "+.@remaining+" minutes";
		next;

		if (select("Stop Happy Hour:Back") == 1) {
			donpcevent "HappyHour::OnManualStop";
			mes "[Events Manager]";
			mes "Happy Hour has been stopped.";
			close;
		}
		goto L_HappyHour;
	}

	mes "^FF0000Status: INACTIVE^000000";
	next;

	switch(select("Start Happy Hour:Manage Schedule:Back")) {
		case 1:
			mes "[Start Happy Hour]";
			mes "Select duration:";
			next;
			set .@choice, select("30 minutes:1 hour:2 hours:3 hours:Custom");
			switch(.@choice) {
				case 1: set $@HappyHourDuration, 30; break;
				case 2: set $@HappyHourDuration, 60; break;
				case 3: set $@HappyHourDuration, 120; break;
				case 4: set $@HappyHourDuration, 180; break;
				case 5:
					mes "[Custom Duration]";
					mes "Enter duration in minutes:";
					input .@custom;
					if (.@custom < 10 || .@custom > 480) {
						mes "Invalid duration. Must be 10-480 minutes.";
						close;
					}
					set $@HappyHourDuration, .@custom;
					break;
			}
			donpcevent "HappyHour::OnEventStart";
			mes "[Events Manager]";
			mes "Happy Hour started for "+$@HappyHourDuration+" minutes!";
			close;

		case 2:
			mes "[Happy Hour Schedule]";
			query_sql "SELECT id, DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), start_hour, start_minute, duration_minutes, enabled FROM event_schedule WHERE event_code = 'happy_hour' ORDER BY day_of_week", .@ids, .@days$, .@hours, .@mins, .@durations, .@enabled;

			if (getarraysize(.@ids) == 0) {
				mes "No scheduled Happy Hours.";
			} else {
				for (set .@i, 0; .@i < getarraysize(.@ids); set .@i, .@i + 1) {
					set .@status$, (.@enabled[.@i] ? "^00AA00ON^000000" : "^FF0000OFF^000000");
					mes .@days$[.@i]+" "+.@hours[.@i]+":"+(.@mins[.@i] < 10 ? "0" : "")+.@mins[.@i]+" ("+.@durations[.@i]+"m) ["+.@status$+"]";
				}
			}
			next;

			switch(select("Add Schedule:Toggle Schedule:Remove Schedule:Back")) {
				case 1:
					mes "Select day:";
					next;
					set .@day, select("Monday:Tuesday:Wednesday:Thursday:Friday:Saturday:Sunday") - 1;
					mes "Enter hour (0-23):";
					input .@hour;
					if (.@hour < 0 || .@hour > 23) { mes "Invalid hour."; close; }
					mes "Enter minute (0-59):";
					input .@minute;
					if (.@minute < 0 || .@minute > 59) { mes "Invalid minute."; close; }
					mes "Duration (minutes):";
					input .@duration;
					if (.@duration < 10 || .@duration > 480) { mes "Invalid duration."; close; }

					query_sql "INSERT INTO event_schedule (event_code, schedule_type, day_of_week, start_hour, start_minute, duration_minutes, enabled) VALUES ('happy_hour', 'fixed', "+.@day+", "+.@hour+", "+.@minute+", "+.@duration+", 1)";
					callsub S_InvalidateCache;
					mes "Schedule added!";
					close;

				case 2:
					if (getarraysize(.@ids) == 0) { mes "No schedules to toggle."; close; }
					mes "Enter schedule ID to toggle:";
					input .@toggle_id;
					query_sql "UPDATE event_schedule SET enabled = NOT enabled WHERE id = "+.@toggle_id;
					callsub S_InvalidateCache;
					mes "Schedule toggled!";
					close;

				case 3:
					if (getarraysize(.@ids) == 0) { mes "No schedules to remove."; close; }
					mes "Enter schedule ID to remove:";
					input .@remove_id;
					query_sql "DELETE FROM event_schedule WHERE id = "+.@remove_id;
					callsub S_InvalidateCache;
					mes "Schedule removed!";
					close;

				case 4:
					goto L_GMMenu;
			}
			close;

		case 3:
			goto L_GMMenu;
	}
	end;

// ============================================================
// EXP Boost Management
// ============================================================
L_ExpBoost:
	mes "[EXP Boost Management]";

	if ($@ExpBoostActive) {
		set .@mult$, callfunc("F_FormatExpMultiplier", $@ExpBoostMultiplier);
		set .@duration, callfunc("F_ExpBoostDuration");
		mes "^00AA00Status: ACTIVE^000000";
		mes "Multiplier: "+.@mult$;
		mes "Running for: "+.@duration+" minutes";
		next;

		if (select("Stop EXP Boost:Back") == 1) {
			donpcevent "ExpBoost::OnEventEnd";
			mes "[Events Manager]";
			mes "EXP Boost has been stopped.";
			mes "Rates restored to normal.";
			close;
		}
		goto L_ExpBoost;
	}

	mes "^FF0000Status: INACTIVE^000000";
	mes " ";
	mes "EXP Boost multiplies server rates";
	mes "by a custom amount (1.5x to 5x).";
	mes "Stacks with Happy Hour!";
	next;

	switch(select("Start EXP Boost:Back")) {
		case 1:
			mes "[Start EXP Boost]";
			mes "Select multiplier:";
			next;
			set .@choice, select("1.5x:2x:3x:4x:5x:Custom");
			switch(.@choice) {
				case 1: set $@ExpBoostMultiplier, 150; break;
				case 2: set $@ExpBoostMultiplier, 200; break;
				case 3: set $@ExpBoostMultiplier, 300; break;
				case 4: set $@ExpBoostMultiplier, 400; break;
				case 5: set $@ExpBoostMultiplier, 500; break;
				case 6:
					mes "[Custom Multiplier]";
					mes "Enter multiplier as percentage:";
					mes "(150 = 1.5x, 200 = 2x, 300 = 3x)";
					input .@custom;
					if (.@custom < 150 || .@custom > 500) {
						mes "Invalid. Must be 150-500.";
						close;
					}
					set $@ExpBoostMultiplier, .@custom;
					break;
			}
			donpcevent "ExpBoost::OnEventStart";
			set .@mult$, callfunc("F_FormatExpMultiplier", $@ExpBoostMultiplier);
			mes "[Events Manager]";
			mes "EXP Boost started at "+.@mult$+"!";
			close;

		case 2:
			goto L_GMMenu;
	}
	end;

// ============================================================
// Monster of the Day Management
// ============================================================
L_MOTD:
	mes "[MOTD Settings]";

	// Get current config
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_enabled'", .@motd_enabled$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_selection_mode'", .@mode$;
	query_sql "SELECT config_value FROM event_config WHERE config_key = 'motd_min_level'", .@min_level$;

	// Get current MOTD
	query_sql "SELECT mob_id, mob_name, TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@cur_mob_id, .@cur_mob_name$, .@cur_mins;

	if (.@motd_enabled$ == "0") {
		mes "Status: ^FF0000DISABLED^000000";
	} else {
		mes "Status: ^00AA00ENABLED^000000";
	}
	if (.@cur_mob_name$ != "") {
		mes "Current: ^0000FF"+.@cur_mob_name$+"^000000 ("+.@cur_mins+"m left)";
	}
	mes "Mode: ^0000FF"+.@mode$+"^000000";
	next;

	switch(select("Toggle MOTD On/Off:Change Current MOTD:Edit Today's Schedule:Change Selection Mode:Back")) {
		case 1:
			if (.@motd_enabled$ == "0") {
				query_sql "UPDATE event_config SET config_value = '1' WHERE config_key = 'motd_enabled'";
				callsub S_InvalidateCache;
				mes "[Events Manager]";
				mes "Monster of the Day has been ^00AA00ENABLED^000000.";
				mes "The next rotation will be announced.";
			} else {
				query_sql "UPDATE event_config SET config_value = '0' WHERE config_key = 'motd_enabled'";
				callsub S_InvalidateCache;
				// Clear the active MOTD immediately
				set $@LastMOTDMobID, 0;
				set $@CurrentMOTDName$, "";
				mes "[Events Manager]";
				mes "Monster of the Day has been ^FF0000DISABLED^000000.";
				mes "No bonuses will be given until re-enabled.";
			}
			close;

		case 2:
			goto L_ChangeMOTD;

		case 3:
			goto L_EditMOTDSchedule;

		case 4:
			mes "Select mode:";
			next;
			set .@new_mode, select("Random:Tier Rotation");
			if (.@new_mode == 1) {
				query_sql "UPDATE event_config SET config_value = 'random' WHERE config_key = 'motd_selection_mode'";
			} else {
				query_sql "UPDATE event_config SET config_value = 'tier_rotation' WHERE config_key = 'motd_selection_mode'";
			}
			callsub S_InvalidateCache;
			mes "Mode updated!";
			close;

		case 5:
			goto L_GMMenu;
	}
	end;

// ============================================================
// Change Current MOTD Monster
// ============================================================
L_ChangeMOTD:
	mes "[Change Current MOTD]";

	// Check if there's an active slot
	query_sql "SELECT id FROM motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@slot_id;
	if (.@slot_id == 0) {
		mes "No active MOTD time slot.";
		mes "Generate a schedule first.";
		close;
	}

	mes "Enter Monster ID:";
	mes "(e.g., 1002 = Poring, 1038 = Osiris)";
	input .@mob_id;

	// Validate monster exists
	set .@mob_name$, getmonsterinfo(.@mob_id, MOB_NAME);
	if (.@mob_name$ == "" || .@mob_name$ == "null") {
		mes "Invalid monster ID.";
		close;
	}

	// Check if in MOTD pool, get tier
	query_sql "SELECT tier FROM motd_monsters WHERE mob_id = "+.@mob_id, .@tier$;
	if (.@tier$ == "") {
		set .@tier$, "custom";  // Not in pool, mark as custom
	}

	// Update the current schedule slot
	query_sql "UPDATE motd_schedule SET mob_id = "+.@mob_id+", mob_name = '"+escape_sql(.@mob_name$)+"', tier = '"+.@tier$+"' WHERE id = "+.@slot_id;

	// Update global variables
	set $@LastMOTDMobID, .@mob_id;
	set $@CurrentMOTDName$, .@mob_name$;

	// Invalidate cache
	callsub S_InvalidateCache;

	// Announce the change
	query_sql "SELECT TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM motd_schedule WHERE id = "+.@slot_id, .@mins_left;
	announce "Monster of the Day changed: "+.@mob_name$+"! 3x EXP for "+.@mins_left+" minutes!", bc_all, 0x00FF00;

	mes "[Events Manager]";
	mes "MOTD changed to ^0000FF"+.@mob_name$+"^000000!";
	mes "Announced to all players.";
	close;

// ============================================================
// Edit Today's MOTD Schedule
// ============================================================
L_EditMOTDSchedule:
	mes "[Today's MOTD Schedule]";
	query_sql "SELECT id, mob_name, tier, DATE_FORMAT(start_time, '%H:%i'), DATE_FORMAT(end_time, '%H:%i'), NOW() BETWEEN start_time AND end_time FROM motd_schedule WHERE DATE(start_time) = CURDATE() ORDER BY start_time", .@ids, .@names$, .@tiers$, .@starts$, .@ends$, .@is_current;

	if (getarraysize(.@ids) == 0) {
		mes "No schedule for today.";
		mes "Use 'Regenerate MOTD Schedule'";
		mes "from the main GM menu.";
		close;
	}

	for (set .@i, 0; .@i < getarraysize(.@ids); set .@i, .@i + 1) {
		if (.@is_current[.@i]) {
			mes "^00FF00► "+.@starts$[.@i]+"-"+.@ends$[.@i]+": "+.@names$[.@i]+" ["+.@tiers$[.@i]+"]^000000";
		} else {
			mes .@starts$[.@i]+"-"+.@ends$[.@i]+": "+.@names$[.@i]+" ["+.@tiers$[.@i]+"]";
		}
	}
	next;

	mes "Select a time slot to change:";
	next;

	// Build menu from time slots
	set .@menu$, "";
	for (set .@i, 0; .@i < getarraysize(.@ids); set .@i, .@i + 1) {
		set .@menu$, .@menu$ + .@starts$[.@i] + "-" + .@ends$[.@i] + " (" + .@names$[.@i] + ")";
		if (.@i < getarraysize(.@ids) - 1) set .@menu$, .@menu$ + ":";
	}
	set .@menu$, .@menu$ + ":Back";

	set .@slot_choice, select(.@menu$);
	if (.@slot_choice > getarraysize(.@ids)) {
		goto L_MOTD;
	}

	set .@edit_slot_id, .@ids[.@slot_choice - 1];
	set .@edit_is_current, .@is_current[.@slot_choice - 1];

	// Enter monster ID
	mes "[Change Slot "+.@starts$[.@slot_choice - 1]+"-"+.@ends$[.@slot_choice - 1]+"]";
	mes "Enter Monster ID:";
	input .@mob_id;

	// Validate monster exists
	set .@mob_name$, getmonsterinfo(.@mob_id, MOB_NAME);
	if (.@mob_name$ == "" || .@mob_name$ == "null") {
		mes "Invalid monster ID.";
		close;
	}

	// Check if in MOTD pool, get tier
	query_sql "SELECT tier FROM motd_monsters WHERE mob_id = "+.@mob_id, .@tier$;
	if (.@tier$ == "") {
		set .@tier$, "custom";
	}

	// Update the schedule slot
	query_sql "UPDATE motd_schedule SET mob_id = "+.@mob_id+", mob_name = '"+escape_sql(.@mob_name$)+"', tier = '"+.@tier$+"' WHERE id = "+.@edit_slot_id;

	// If this is the current slot, update globals and announce
	if (.@edit_is_current) {
		set $@LastMOTDMobID, .@mob_id;
		set $@CurrentMOTDName$, .@mob_name$;

		query_sql "SELECT TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM motd_schedule WHERE id = "+.@edit_slot_id, .@mins_left;
		announce "Monster of the Day changed: "+.@mob_name$+"! 3x EXP for "+.@mins_left+" minutes!", bc_all, 0x00FF00;
	}

	callsub S_InvalidateCache;

	mes "[Events Manager]";
	mes "Slot updated to ^0000FF"+.@mob_name$+"^000000!";
	if (.@edit_is_current) {
		mes "Announced to all players.";
	}
	close;

// ============================================================
// Event History
// ============================================================
L_History:
	mes "[Event History]";
	query_sql "SELECT event_code, DATE_FORMAT(actual_start, '%Y-%m-%d %H:%i'), DATE_FORMAT(actual_end, '%H:%i'), status FROM event_instances ORDER BY id DESC LIMIT 10", .@codes$, .@starts$, .@ends$, .@status$;

	if (getarraysize(.@codes$) == 0) {
		mes "No event history found.";
	} else {
		for (set .@i, 0; .@i < getarraysize(.@codes$); set .@i, .@i + 1) {
			mes .@starts$[.@i]+" - "+.@ends$[.@i];
			mes "  "+.@codes$[.@i]+" ["+.@status$[.@i]+"]";
		}
	}
	close;

// ============================================================
// Invalidate Website Cache
// ============================================================
S_InvalidateCache:
	query_sql "UPDATE event_config SET config_value = config_value + 1 WHERE config_key = 'cache_version'";
	return;
}
