//===== rAthena Script =======================================
//= Events Manager NPC
//= Shows all current and upcoming server events
//= Includes GM management options for staff
//============================================================

prontera,141,172,5	script	Events Manager	4_M_MINSTREL,{
	mes "[Events Manager]";
	mes "Welcome to the Server Events!";
	mes "Here you can see all active and upcoming events.";
	next;

	// ========================================
	// CURRENTLY ACTIVE EVENTS
	// ========================================
	mes "[^FF0000Currently Active^000000]";
	set .@has_active, 0;

	// Check Happy Hour
	if ($@HappyHourActive) {
		set .@remaining, callfunc("F_HappyHourRemaining");
		mes "^FFD700[Active] Happy Hour^000000";
		mes "   2x EXP rates!";
		mes "   Time left: ^00AA00"+.@remaining+" minutes^000000";
		set .@has_active, 1;
	}

	// Check EXP Boost (check DB if not loaded yet)
	if ($@ExpBoostActive) {
		set .@mult$, callfunc("F_FormatExpMultiplier", $@ExpBoostMultiplier);
		set .@remaining, callfunc("F_ExpBoostRemaining");
		mes "^00BFFF[Active] EXP Boost^000000";
		mes "   "+.@mult$+" EXP rates!";
		if (.@remaining > 0) {
			set .@remaining$, callfunc("F_FormatDuration", .@remaining);
			mes "   Time left: ^00AA00"+.@remaining$+"^000000";
		}
		set .@has_active, 1;
	} else {
		// Check database for active boost not yet loaded by scheduler
		// One-time events are now stored in event_instances
		query_sql "SELECT CAST(JSON_EXTRACT(ei.event_data, '$.multiplier') AS UNSIGNED), TIMESTAMPDIFF(MINUTE, NOW(), ei.scheduled_end) FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'exp_boost' AND et.enabled = 1 AND ei.trigger_type = 'one_time' AND ei.status IN ('scheduled', 'active') AND NOW() BETWEEN COALESCE(ei.scheduled_start, ei.actual_start) AND ei.scheduled_end LIMIT 1", .@eb_mult, .@eb_remaining;
		if (.@eb_mult > 0 && .@eb_remaining > 0) {
			set .@mult$, callfunc("F_FormatExpMultiplier", .@eb_mult);
			set .@remaining$, callfunc("F_FormatDuration", .@eb_remaining);
			mes "^00BFFF[Active] EXP Boost^000000 ^888888(activating...)^000000";
			mes "   "+.@mult$+" EXP rates!";
			mes "   Time left: ^00AA00"+.@remaining$+"^000000";
			set .@has_active, 1;
		}
	}

	// Check Monster of the Day (enabled status from event_types)
	query_sql "SELECT enabled FROM event_types WHERE code = 'motd'", .@motd_enabled;
	if (!.@motd_enabled) {
		mes "^888888[Disabled] Monster of the Day^000000";
	} else {
		query_sql "SELECT mob_name, TIMESTAMPDIFF(MINUTE, NOW(), end_time) FROM event_motd_schedule WHERE NOW() BETWEEN start_time AND end_time LIMIT 1", .@motd_name$, .@motd_mins;
		if (.@motd_name$ != "") {
			mes "^00FF00[Active] Monster of the Day^000000";
			mes "   Featured: ^0000FF"+.@motd_name$+"^000000";
			mes "   3x EXP, Drops & Hunt Points!";
			mes "   Time left: ^00AA00"+.@motd_mins+" minutes^000000";
			set .@has_active, 1;
		}
	}

	if (!.@has_active) {
		mes "^888888No events currently active.^000000";
	}
	next;

	// ========================================
	// UPCOMING EVENTS
	// ========================================
	mes "[^0055FFUpcoming Events^000000]";

	// Get scheduled Happy Hours (with duration) - using JOIN with event_types
	query_sql "SELECT DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((es.day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), es.start_hour, es.start_minute, es.duration_minutes FROM event_schedule es JOIN event_types et ON es.event_type_id = et.id WHERE et.code = 'happy_hour' AND es.enabled = 1 ORDER BY es.day_of_week LIMIT 3", .@hh_days$, .@hh_hours, .@hh_mins, .@hh_durations;

	if (getarraysize(.@hh_days$) > 0) {
		mes "^FFD700Happy Hour Schedule:^000000";
		for (set .@i, 0; .@i < getarraysize(.@hh_days$); set .@i, .@i + 1) {
			set .@time$, (.@hh_hours[.@i] < 10 ? "0" : "") + .@hh_hours[.@i] + ":" + (.@hh_mins[.@i] < 10 ? "0" : "") + .@hh_mins[.@i];
			// Convert duration to hours/minutes display
			if (.@hh_durations[.@i] >= 60) {
				set .@dur_hours, .@hh_durations[.@i] / 60;
				set .@dur_mins, .@hh_durations[.@i] % 60;
				if (.@dur_mins > 0) {
					set .@dur$, .@dur_hours+"h "+.@dur_mins+"m";
				} else {
					set .@dur$, .@dur_hours+" hour" + (.@dur_hours > 1 ? "s" : "");
				}
			} else {
				set .@dur$, .@hh_durations[.@i]+" min";
			}
			mes "   "+.@hh_days$[.@i]+" at "+.@time$+" ("+.@dur$+")";
		}
	} else {
		mes "^888888No Happy Hour scheduled.^000000";
	}

	mes " ";
	mes "^00FF00Monster of the Day:^000000";
	mes "   Rotates every 3 hours";
	mes "   Check the Monster Hunter Board!";

	// Get scheduled Town Invasions (one-time with details from event_data)
	mes " ";
	query_sql "SELECT DATE_FORMAT(ei.scheduled_start, '%m/%d %H:%i'), CAST(JSON_EXTRACT(ei.event_data, '$.threat') AS UNSIGNED), CAST(JSON_EXTRACT(ei.event_data, '$.town_id') AS UNSIGNED) FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'invasion' AND ei.status = 'scheduled' AND ei.scheduled_start > NOW() ORDER BY ei.scheduled_start LIMIT 3", .@inv_times_fmt$, .@inv_threats, .@inv_town_ids;

	// Get recurring schedules
	query_sql "SELECT DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((es.day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), es.start_hour, es.start_minute FROM event_schedule es JOIN event_types et ON es.event_type_id = et.id WHERE et.code = 'invasion' AND es.enabled = 1 ORDER BY (es.day_of_week - WEEKDAY(CURDATE()) + 7) % 7 LIMIT 3", .@inv_days$, .@inv_hours, .@inv_mins;

	if (getarraysize(.@inv_times_fmt$) > 0 || getarraysize(.@inv_days$) > 0) {
		mes "^FF4444Town Invasions:^000000";
		// Show one-time scheduled with threat and town
		for (.@i = 0; .@i < getarraysize(.@inv_times_fmt$); .@i++) {
			// Get threat name
			switch(.@inv_threats[.@i]) {
				case 0: .@threat$ = "Wolf (Easy)"; break;
				case 1: .@threat$ = "Tiger (Medium)"; break;
				case 2: .@threat$ = "Demon (Hard)"; break;
				case 3: .@threat$ = "Dragon (V.Hard)"; break;
				case 4: .@threat$ = "God (Extreme)"; break;
				default: .@threat$ = "Random"; break;
			}
			// Get town name
			.@town$ = "";
			if (.@inv_town_ids[.@i] > 0) {
				query_sql "SELECT display_name FROM event_invasion_towns WHERE id = "+.@inv_town_ids[.@i], .@town$;
			}
			if (.@town$ == "") .@town$ = "Random";
			mes "   "+.@inv_times_fmt$[.@i]+" ["+.@threat$+"] "+.@town$;
		}
		// Show recurring (only if no one-time, to avoid clutter)
		if (getarraysize(.@inv_times_fmt$) == 0) {
			for (.@i = 0; .@i < getarraysize(.@inv_days$); .@i++) {
				.@time$ = (.@inv_hours[.@i] < 10 ? "0" : "") + .@inv_hours[.@i] + ":" + (.@inv_mins[.@i] < 10 ? "0" : "") + .@inv_mins[.@i];
				mes "   " + .@inv_days$[.@i] + " at " + .@time$;
			}
		}
	}
	next;

	// ========================================
	// MENU (with GM options if applicable)
	// ========================================
	if (getgmlevel() >= 60) {
		switch(select("View Event Details:View My Event Stats:^FF0000[GM] Manage Events^000000:Close")) {
			case 1: goto L_EventDetails;
			case 2: goto L_MyStats;
			case 3: goto L_GMMenu;
			case 4: close;
		}
	} else {
		switch(select("View Event Details:View My Event Stats:Close")) {
			case 1: goto L_EventDetails;
			case 2: goto L_MyStats;
			case 3: close;
		}
	}
	end;

L_EventDetails:
	mes "[Event Details]";
	mes "Which event would you like to learn about?";
	next;
	switch(select("Happy Hour:Monster of the Day:Back")) {
		case 1:
			mes "[^FFD700Happy Hour^000000]";
			mes " ";
			mes "^0000FFWhat is it?^000000";
			mes "A server-wide EXP boost event!";
			mes "All players receive ^FF00002x Base EXP^000000";
			mes "and ^FF00002x Job EXP^000000 from monsters.";
			next;
			mes "[^FFD700Happy Hour^000000]";
			mes "^0000FFWhat about drops?^000000";
			mes "Drop rates stay the same to keep";
			mes "the economy balanced. This is a";
			mes "leveling event, not a farming one!";
			next;
			mes "[^FFD700Happy Hour^000000]";
			mes "^0000FFWhen does it happen?^000000";
			// Get next scheduled Happy Hour - using JOIN with event_types
			query_sql "SELECT DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((es.day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), es.start_hour, es.start_minute, es.duration_minutes FROM event_schedule es JOIN event_types et ON es.event_type_id = et.id WHERE et.code = 'happy_hour' AND es.enabled = 1 ORDER BY (es.day_of_week - WEEKDAY(CURDATE()) + 7) % 7 LIMIT 1", .@next_day$, .@next_hour, .@next_min, .@next_dur;
			if (.@next_day$ != "") {
				set .@time$, (.@next_hour < 10 ? "0" : "") + .@next_hour + ":" + (.@next_min < 10 ? "0" : "") + .@next_min;
				if (.@next_dur >= 60) {
					set .@dur$, (.@next_dur / 60) + " hour" + ((.@next_dur / 60) > 1 ? "s" : "");
				} else {
					set .@dur$, .@next_dur + " minutes";
				}
				mes "Next: ^00AA00"+.@next_day$+" at "+.@time$+"^000000";
				mes "Duration: ^00AA00"+.@dur$+"^000000";
			} else {
				mes "No scheduled Happy Hours.";
				mes "Check back later!";
			}
			next;
			mes "[^FFD700Happy Hour^000000]";
			mes "^0000FFTips:^000000";
			mes "- Great for leveling new jobs";
			mes "- Perfect for pushing past plateaus";
			mes "- Stack with Monster of the Day!";
			close;

		case 2:
			mes "[^00FF00Monster of the Day^000000]";
			mes " ";
			mes "^0000FFWhat is it?^000000";
			mes "A rotating featured monster that";
			mes "gives ^FF00003x EXP, 3x Drops^000000, and";
			mes "bonus ^FF0000Hunt Points^000000 when killed!";
			next;
			mes "[^00FF00Monster of the Day^000000]";
			mes "^0000FFHow does it work?^000000";
			mes "The featured monster changes";
			mes "every ^00AA003 hours^000000. Different tiers";
			mes "rotate throughout the day so";
			mes "everyone can participate.";
			next;
			mes "[^00FF00Monster of the Day^000000]";
			mes "^0000FFWhat are Hunt Points?^000000";
			mes "A currency earned by hunting";
			mes "MOTD targets. Spend them on";
			mes "special rewards like costumes,";
			mes "consumables, and rare items!";
			next;
			mes "[^00FF00Monster of the Day^000000]";
			mes "^0000FFWhere do I find it?^000000";
			mes "Check the ^0000FFMonster Hunter Board^000000";
			mes "in Prontera for the current";
			mes "target, location hints, and";
			mes "your progress!";
			close;

		case 3:
			close;
	}
	end;

L_MyStats:
	mes "[Your Event Stats]";
	// Get event coin balance
	query_sql "SELECT coins, total_earned FROM event_coins WHERE account_id = "+getcharid(3), .@coins, .@total;
	mes "Event Coins: ^0000FF"+.@coins+"^000000";
	mes "Total Earned: "+.@total;
	mes " ";
	// Get MOTD completions
	query_sql "SELECT COUNT(*) FROM event_motd_completions WHERE char_id = "+getcharid(0), .@motd_count;
	mes "MOTD Hunts Completed: "+.@motd_count;
	close;

// ============================================================
// GM MANAGEMENT MENU
// ============================================================
L_GMMenu:
	mes "[^FF0000GM Event Manager^000000]";
	mes "Welcome, "+strcharinfo(0)+".";
	mes "What would you like to manage?";
	next;

	switch(select("Happy Hour:EXP Boost:Monster of the Day:Town Invasion:Event History:Back")) {
		case 1: goto L_HappyHour;
		case 2: goto L_ExpBoost;
		case 3: goto L_MOTD;
		case 4: goto L_Invasion;
		case 5: goto L_History;
		case 6: close;
	}
	end;

// ============================================================
// Happy Hour Schedule Management
// ============================================================
L_HappyHour:
	mes "[Happy Hour Schedule]";
	mes "^888888Weekly recurring 2x EXP events^000000";
	mes " ";

	// Show current status
	if ($@HappyHourActive) {
		set .@remaining, callfunc("F_HappyHourRemaining");
		mes "^00AA00Status: ACTIVE^000000 ("+.@remaining+"m left)";
	} else {
		mes "^FF0000Status: INACTIVE^000000";
	}
	mes " ";

	// Show scheduled Happy Hours
	query_sql "SELECT es.id, DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((es.day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), es.start_hour, es.start_minute, es.duration_minutes, es.enabled FROM event_schedule es JOIN event_types et ON es.event_type_id = et.id WHERE et.code = 'happy_hour' ORDER BY es.day_of_week, es.start_hour", .@ids, .@days$, .@hours, .@mins, .@durations, .@enabled;

	if (getarraysize(.@ids) == 0) {
		mes "^888888No schedules configured.^000000";
	} else {
		mes "^0055FFScheduled:^000000";
		for (set .@i, 0; .@i < getarraysize(.@ids); set .@i, .@i + 1) {
			set .@status$, (.@enabled[.@i] ? "^00AA00ON^000000" : "^FF0000OFF^000000");
			// Format duration as hours (e.g., "3h" instead of "180m")
			if (.@durations[.@i] >= 60 && .@durations[.@i] % 60 == 0) {
				set .@dur$, (.@durations[.@i] / 60) + "h";
			} else if (.@durations[.@i] >= 60) {
				set .@dur$, (.@durations[.@i] / 60) + "h" + (.@durations[.@i] % 60) + "m";
			} else {
				set .@dur$, .@durations[.@i] + "m";
			}
			mes "["+.@ids[.@i]+"] "+.@days$[.@i]+" "+.@hours[.@i]+":"+(.@mins[.@i] < 10 ? "0" : "")+.@mins[.@i]+" ("+.@dur$+") "+.@status$;
		}
	}
	next;

	// Menu options
	if ($@HappyHourActive) {
		switch(select("Add Schedule:Toggle Schedule:Remove Schedule:Stop Current:Manage Triggers:Back")) {
			case 1: goto L_HH_Add;
			case 2: goto L_HH_Toggle;
			case 3: goto L_HH_Remove;
			case 4:
				donpcevent "HappyHour::OnManualStop";
				mes "[Events Manager]";
				mes "Happy Hour stopped.";
				close;
			case 5: goto L_HH_Triggers;
			case 6: goto L_GMMenu;
		}
	} else {
		switch(select("Add Schedule:Toggle Schedule:Remove Schedule:Manage Triggers:Back")) {
			case 1: goto L_HH_Add;
			case 2: goto L_HH_Toggle;
			case 3: goto L_HH_Remove;
			case 4: goto L_HH_Triggers;
			case 5: goto L_GMMenu;
		}
	}
	end;

L_HH_Triggers:
	callfunc "F_TriggerMenu", "happy_hour";
	goto L_HappyHour;

L_HH_Add:
	mes "[Add Happy Hour]";
	mes "Select day:";
	next;
	set .@day, select("Monday:Tuesday:Wednesday:Thursday:Friday:Saturday:Sunday") - 1;

	mes "Select start time:";
	next;
	set .@time_choice, select("9 AM:12 PM:3 PM:6 PM:9 PM:Custom");
	switch(.@time_choice) {
		case 1: set .@hour, 9; set .@minute, 0; break;
		case 2: set .@hour, 12; set .@minute, 0; break;
		case 3: set .@hour, 15; set .@minute, 0; break;
		case 4: set .@hour, 18; set .@minute, 0; break;
		case 5: set .@hour, 21; set .@minute, 0; break;
		case 6:
			mes "Enter hour (0-23):";
			input .@hour;
			if (.@hour < 0 || .@hour > 23) { mes "Invalid hour."; close; }
			mes "Enter minute (0-59):";
			input .@minute;
			if (.@minute < 0 || .@minute > 59) { mes "Invalid minute."; close; }
			break;
	}

	mes "Select duration:";
	next;
	set .@dur_choice, select("30 minutes:1 hour:2 hours:3 hours");
	switch(.@dur_choice) {
		case 1: set .@duration, 30; set .@dur$, "30 min"; break;
		case 2: set .@duration, 60; set .@dur$, "1 hour"; break;
		case 3: set .@duration, 120; set .@dur$, "2 hours"; break;
		case 4: set .@duration, 180; set .@dur$, "3 hours"; break;
	}

	// Format day name and time for confirmation
	setarray .@daynames$[0], "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday";

	// Format start time in 12-hour AM/PM
	set .@ampm$, (.@hour >= 12) ? "PM" : "AM";
	set .@hour12, (.@hour > 12) ? .@hour - 12 : ((.@hour == 0) ? 12 : .@hour);
	set .@time$, .@hour12 + ":" + ((.@minute < 10) ? "0" : "") + .@minute + " " + .@ampm$;

	// Calculate end time
	.@end_total_mins = .@hour * 60 + .@minute + .@duration;
	.@end_hour = .@end_total_mins / 60;
	if (.@end_hour >= 24) set .@end_hour, .@end_hour - 24;  // Wrap around midnight
	.@end_min = .@end_total_mins % 60;
	set .@end_ampm$, (.@end_hour >= 12) ? "PM" : "AM";
	set .@end_hour12, (.@end_hour > 12) ? .@end_hour - 12 : ((.@end_hour == 0) ? 12 : .@end_hour);
	set .@end_time$, .@end_hour12 + ":" + ((.@end_min < 10) ? "0" : "") + .@end_min + " " + .@end_ampm$;

	// Get the actual date for next occurrence
	query_sql "SELECT DATE_FORMAT(DATE_ADD(CURDATE(), INTERVAL (("+.@day+" - WEEKDAY(CURDATE()) + 7) % 7) DAY), '%b %d')", .@next_date$;

	mes "[Confirm Schedule]";
	mes "^0055FF" + .@daynames$[.@day] + ", " + .@next_date$ + "^000000";
	mes "Time: ^0055FF" + .@time$ + " - " + .@end_time$ + "^000000";
	mes "Duration: ^0055FF" + .@dur$ + "^000000";
	mes " ";
	mes "^888888(Repeats weekly)^000000";
	next;
	if (select("Confirm:Cancel") == 2) {
		mes "Schedule cancelled.";
		next;
		goto L_HappyHour;
	}

	// Get event_type_id for happy_hour
	query_sql "SELECT id FROM event_types WHERE code = 'happy_hour'", .@type_id;
	query_sql "INSERT INTO event_schedule (event_type_id, event_code, schedule_type, day_of_week, start_hour, start_minute, duration_minutes, enabled) VALUES ("+.@type_id+", 'happy_hour', 'weekly', "+.@day+", "+.@hour+", "+.@minute+", "+.@duration+", 1)";
	query_sql "SELECT LAST_INSERT_ID()", .@new_id;

	// Calculate next_run for the new schedule
	query_sql "UPDATE event_schedule SET next_run = DATE_ADD(DATE_ADD(CURDATE(), INTERVAL (("+.@day+" - WEEKDAY(CURDATE()) + 7) % 7) DAY), INTERVAL "+.@hour+" * 60 + "+.@minute+" MINUTE) WHERE id = "+.@new_id;

	// Check if we're currently within the active window and should trigger now
	query_sql "SELECT WEEKDAY(NOW()), HOUR(NOW()) * 60 + MINUTE(NOW())", .@today_dow, .@cur_mins;
	.@start_mins = .@hour * 60 + .@minute;
	.@end_mins = .@start_mins + .@duration;

	if (.@today_dow == .@day && .@cur_mins >= .@start_mins && .@cur_mins < .@end_mins && !$@HappyHourActive) {
		// We're in the window! Trigger Happy Hour
		.@remaining = .@end_mins - .@cur_mins;
		set $@HappyHourDuration, .@remaining;
		set $@HappyHourScheduleId, .@new_id;
		donpcevent "HappyHour::OnEventStart";
		callsub S_InvalidateCache;
		mes "[Events Manager]";
		mes "Schedule added and ^00AA00Happy Hour started^000000!";
		mes "Time remaining: "+.@remaining+" minutes";
		next;
		goto L_HappyHour;
	}

	callsub S_InvalidateCache;
	mes "[Events Manager]";
	mes "Schedule added!";
	next;
	goto L_HappyHour;

L_HH_Toggle:
	query_sql "SELECT es.id, DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((es.day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), es.start_hour, es.enabled FROM event_schedule es JOIN event_types et ON es.event_type_id = et.id WHERE et.code = 'happy_hour' ORDER BY es.day_of_week", .@ids, .@days$, .@hours, .@enabled;
	if (getarraysize(.@ids) == 0) {
		mes "No schedules to toggle.";
		close;
	}
	mes "[Toggle Schedule]";
	for (set .@i, 0; .@i < getarraysize(.@ids); set .@i, .@i + 1) {
		set .@status$, (.@enabled[.@i] ? "ON" : "OFF");
		mes "["+.@ids[.@i]+"] "+.@days$[.@i]+" "+.@hours[.@i]+":00 - "+.@status$;
	}
	mes " ";
	mes "Enter ID to toggle:";
	input .@toggle_id;
	query_sql "UPDATE event_schedule SET enabled = NOT enabled WHERE id = "+.@toggle_id+" AND event_code = 'happy_hour'";
	callsub S_InvalidateCache;

	// Check if toggled ON and we're currently in the active window
	query_sql "SELECT es.enabled, es.day_of_week, es.start_hour, es.start_minute, es.duration_minutes FROM event_schedule es WHERE es.id = "+.@toggle_id, .@now_enabled, .@day, .@hour, .@minute, .@duration;

	if (.@now_enabled && !$@HappyHourActive) {
		// Check if we're in the window right now
		query_sql "SELECT WEEKDAY(NOW()), HOUR(NOW()) * 60 + MINUTE(NOW())", .@today_dow, .@cur_mins;
		.@start_mins = .@hour * 60 + .@minute;
		.@end_mins = .@start_mins + .@duration;

		if (.@today_dow == .@day && .@cur_mins >= .@start_mins && .@cur_mins < .@end_mins) {
			// We're in the window! Trigger Happy Hour
			.@remaining = .@end_mins - .@cur_mins;
			set $@HappyHourDuration, .@remaining;
			set $@HappyHourScheduleId, .@toggle_id;
			donpcevent "HappyHour::OnEventStart";
			mes "[Events Manager]";
			mes "Schedule enabled and ^00AA00Happy Hour started^000000!";
			mes "Time remaining: "+.@remaining+" minutes";
			next;
			goto L_HappyHour;
		}
	}

	mes "Toggled!";
	next;
	goto L_HappyHour;

L_HH_Remove:
	query_sql "SELECT es.id, DAYNAME(DATE_ADD(CURDATE(), INTERVAL ((es.day_of_week - WEEKDAY(CURDATE()) + 7) % 7) DAY)), es.start_hour FROM event_schedule es JOIN event_types et ON es.event_type_id = et.id WHERE et.code = 'happy_hour' ORDER BY es.day_of_week", .@ids, .@days$, .@hours;

				close;

	// One-time events are now stored in event_instances
	query_sql "SELECT ei.id, CAST(JSON_EXTRACT(ei.event_data, '$.multiplier') AS UNSIGNED), TIMESTAMPDIFF(MINUTE, NOW(), ei.scheduled_end) FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'exp_boost' AND et.enabled = 1 AND ei.trigger_type = 'one_time' AND ei.status IN ('scheduled', 'active') AND NOW() BETWEEN COALESCE(ei.scheduled_start, ei.actual_start) AND ei.scheduled_end LIMIT 1", .@active_id, .@active_mult, .@active_remaining;

		set .@mult$, callfunc("F_FormatExpMultiplier", .@active_mult);
		set .@remaining$, callfunc("F_FormatDuration", .@active_remaining);

		mes "Multiplier: "+.@mult$;
		mes "Time left: "+.@remaining$;
		next;

		switch(select("Stop EXP Boost:Back")) {
			case 1:
				// Mark as completed in event_instances
				query_sql "UPDATE event_instances SET status = 'completed', actual_end = NOW() WHERE id = "+.@active_id;

				goto L_GMMenu;

	query_sql "SELECT ei.id, DATE_FORMAT(ei.scheduled_start, '%m/%d %H:%i'), TIMESTAMPDIFF(MINUTE, ei.scheduled_start, ei.scheduled_end), CAST(JSON_EXTRACT(ei.event_data, '$.multiplier') AS UNSIGNED) FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id WHERE et.code = 'exp_boost' AND ei.trigger_type = 'one_time' AND ei.status = 'scheduled' AND ei.scheduled_start > NOW() ORDER BY ei.scheduled_start LIMIT 3", .@pend_ids, .@pend_starts$, .@pend_durs, .@pend_mults;

L_ScheduleExpBoost:
	mes "[Schedule EXP Boost]";
	mes "When should it start?";
	next;

	switch(select("Start NOW:Schedule for later:Back")) {
		case 1:
			set .@start_now, 1;
			break;
		case 2:
			set .@start_now, 0;
			mes "Enter start date/time:";
			mes "^888888Format: YYYY-MM-DD HH:MM^000000";
			mes "^888888(e.g., 2025-01-15 18:00)^000000";
			input .@start_dt$;
			next;
			break;
		case 3:
			goto L_ExpBoost;
	}

	mes "How long should it last?";
	next;
	switch(select("1 hour:2 hours:6 hours:12 hours:1 day:3 days:7 days:Custom")) {
		case 1: set .@duration, 60; break;
		case 2: set .@duration, 120; break;
		case 3: set .@duration, 360; break;
		case 4: set .@duration, 720; break;
		case 5: set .@duration, 1440; break;
		case 6: set .@duration, 4320; break;
		case 7: set .@duration, 10080; break;
		case 8:
			mes "Enter duration in hours:";
			mes "^888888(e.g., 48 = 2 days)^000000";
			input .@hours;
			if (.@hours < 1) {
				mes "Minimum 1 hour.";
				close;
			}
			set .@duration, .@hours * 60;
			break;
	}

	mes "Select multiplier:";
	next;
	switch(select("1.5x:2x:3x:4x:5x:Custom")) {
		case 1: set .@mult, 150; break;
		case 2: set .@mult, 200; break;
		case 3: set .@mult, 300; break;
		case 4: set .@mult, 400; break;
		case 5: set .@mult, 500; break;
		case 6:
			mes "Enter multiplier as percentage:";
			mes "^888888(150 = 1.5x, 200 = 2x, 300 = 3x)^000000";
			input .@mult;
			if (.@mult < 150 || .@mult > 500) {
				mes "Invalid. Must be 150-500.";
				close;
			}
			break;
	}

	// Insert one-time events directly into event_instances (not event_schedule)
	query_sql "SELECT id FROM event_types WHERE code = 'exp_boost'", .@type_id;

	if (.@start_now) {
		// Create instance with actual_start = NOW (already active)
		query_sql "INSERT INTO event_instances (event_type_id, event_code, scheduled_start, scheduled_end, actual_start, status, trigger_type, triggered_by, event_data) VALUES ("+.@type_id+", 'exp_boost', NOW(), DATE_ADD(NOW(), INTERVAL "+.@duration+" MINUTE), NOW(), 'active', 'one_time', "+getcharid(3)+", JSON_OBJECT('multiplier', "+.@mult+"))";

		// Start immediately
		set $@ExpBoostMultiplier, .@mult;
		set $@ExpBoostDuration, .@duration;
		set $@ExpBoostInstanceId, .@new_id;
		donpcevent "ExpBoost::OnEventStart";

		set .@mult$, callfunc("F_FormatExpMultiplier", .@mult);
		set .@dur$, callfunc("F_FormatDuration", .@duration);
		mes "[Events Manager]";
		mes "EXP Boost started!";
		mes "Multiplier: "+.@mult$;
		mes "Duration: "+.@dur$;
	} else {
		// Create instance with scheduled_start (pending)
		query_sql "INSERT INTO event_instances (event_type_id, event_code, scheduled_start, scheduled_end, status, trigger_type, triggered_by, event_data) VALUES ("+.@type_id+", 'exp_boost', '"+.@start_dt$+"', DATE_ADD('"+.@start_dt$+"', INTERVAL "+.@duration+" MINUTE), 'scheduled', 'one_time', "+getcharid(3)+", JSON_OBJECT('multiplier', "+.@mult+"))";

		set .@mult$, callfunc("F_FormatExpMultiplier", .@mult);
		set .@dur$, callfunc("F_FormatDuration", .@duration);
		mes "[Events Manager]";
		mes "EXP Boost scheduled!";
		mes "Start: "+.@start_dt$;
		mes "Multiplier: "+.@mult$;
		mes "Duration: "+.@dur$;
	}
	close;

	if ($@InvasionActive) {

			case 5: goto L_Inv_Triggers;
			case 6: goto L_GMMenu;
		}
	} else {
		switch(select("Start Now:Schedule Invasion:View Schedules:Configure:Manage Triggers:Invasion History:Back")) {

L_History:
	// Initialize page if not set
	if (.@history_page < 1) .@history_page = 1;

L_HistoryDisplay:
	mes "[Event History - Page "+.@history_page+"]";

	// Pagination: 10 events per page, max 30 total (3 pages)
	.@per_page = 10;
	.@max_events = 30;
	.@offset = (.@history_page - 1) * .@per_page;

	// Query with pagination - extract JSON fields directly via SQL
	query_sql "SELECT ei.id, et.code, DATE_FORMAT(COALESCE(ei.actual_start, ei.scheduled_start), '%m/%d %H:%i'), DATE_FORMAT(ei.actual_end, '%H:%i'), ei.status, COALESCE(JSON_UNQUOTE(JSON_EXTRACT(ei.event_data, '$.mob_name')), ''), COALESCE(JSON_UNQUOTE(JSON_EXTRACT(ei.event_data, '$.threat')), '0') FROM event_instances ei JOIN event_types et ON ei.event_type_id = et.id ORDER BY (ei.status = 'active') DESC, ei.id DESC LIMIT "+.@per_page+" OFFSET "+.@offset, .@ids, .@codes$, .@starts$, .@ends$, .@status$, .@mob_names$, .@threats$;

	if (getarraysize(.@ids) == 0) {
		if (.@history_page > 1) {
			mes "No more events.";
		} else {
			mes "No event history found.";
		}
	} else {
		for (set .@i, 0; .@i < getarraysize(.@ids); set .@i, .@i + 1) {
			// Format event name based on type
			if (.@codes$[.@i] == "exp_boost") {
				set .@name$, "EXP Boost";
			} else if (.@codes$[.@i] == "happy_hour") {
				set .@name$, "Happy Hour";
			} else if (.@codes$[.@i] == "motd") {
				// mob_name extracted via SQL
				if (.@mob_names$[.@i] != "" && .@mob_names$[.@i] != "null") {
					set .@name$, "MOTD: " + .@mob_names$[.@i];
				} else {
					set .@name$, "Monster of the Day";
				}
			} else if (.@codes$[.@i] == "invasion") {
				// threat extracted via SQL
				.@threat = atoi(.@threats$[.@i]);
				.@town$ = "";
				.@outcome$ = "";

				// Get threat name
				switch(.@threat) {
					case 0: .@threat$ = "Wolf (Easy)"; break;
					case 1: .@threat$ = "Tiger (Medium)"; break;
					case 2: .@threat$ = "Demon (Hard)"; break;
					case 3: .@threat$ = "Dragon (V.Hard)"; break;
					case 4: .@threat$ = "God (Extreme)"; break;
					default: .@threat$ = "?"; break;
				}

				// Try to get town from event_invasion_instances
				query_sql "SELECT t.display_name FROM event_invasion_instances ii JOIN event_invasion_towns t ON ii.town_id = t.id JOIN event_instances ei ON ei.actual_start BETWEEN DATE_SUB(ii.started_at, INTERVAL 1 MINUTE) AND DATE_ADD(ii.started_at, INTERVAL 1 MINUTE) WHERE ei.id = "+.@ids[.@i]+" LIMIT 1", .@town$;
				if (.@town$ == "") .@town$ = "Town";

				// Set outcome based on status
				if (.@status$[.@i] == "completed") {
					.@outcome$ = "^00AA00Victory^000000";
				} else if (.@status$[.@i] == "failed") {
					.@outcome$ = "^FF4444Defeat^000000";
				} else if (.@status$[.@i] == "cancelled") {
					.@outcome$ = "^888888Cancelled^000000";
				} else if (.@status$[.@i] == "active") {
					.@outcome$ = "^FFAA00Active^000000";
				}

				set .@name$, "["+.@threat$+"] "+.@town$+" "+.@outcome$;
			} else {
				set .@name$, .@codes$[.@i];
			}

			// Status color (for non-invasion events)
			if (.@codes$[.@i] != "invasion") {
				if (.@status$[.@i] == "active") {
					set .@stat$, "^00AA00ACTIVE^000000";
				} else if (.@status$[.@i] == "scheduled") {
					set .@stat$, "^0055FFpending^000000";
				} else if (.@status$[.@i] == "completed") {
					set .@stat$, "^888888done^000000";
				} else {
					set .@stat$, "^FF0000"+.@status$[.@i]+"^000000";
				}
				.@end_display$ = (.@ends$[.@i] != "" && .@ends$[.@i] != "NULL") ? .@ends$[.@i] : "...";
				mes .@starts$[.@i]+" - "+.@end_display$+" ["+.@stat$+"]";
				mes "  "+.@name$;
			} else {
				// Invasion: show time and name (which includes outcome)
				.@end_display$ = (.@ends$[.@i] != "" && .@ends$[.@i] != "NULL") ? .@ends$[.@i] : "...";
				mes .@starts$[.@i]+" - "+.@end_display$;
				mes "  "+.@name$;
			}
		}
	}
	next;

	// Pagination menu
	.@has_prev = (.@history_page > 1);
	.@has_next = (getarraysize(.@ids) == .@per_page && .@offset + .@per_page < .@max_events);

	if (.@has_prev && .@has_next) {
		switch(select("Previous Page:Next Page:Back")) {
			case 1: .@history_page--; goto L_HistoryDisplay;
			case 2: .@history_page++; goto L_HistoryDisplay;
			case 3: close;
		}
	} else if (.@has_prev) {
		switch(select("Previous Page:Back")) {
			case 1: .@history_page--; goto L_HistoryDisplay;
			case 2: close;
		}
	} else if (.@has_next) {
		switch(select("Next Page:Back")) {
			case 1: .@history_page++; goto L_HistoryDisplay;
			case 2: close;
		}
	} else {
		close;
	}
	end;

// ============================================================
// Invalidate Website Cache
// ============================================================
S_InvalidateCache:
	query_sql "UPDATE event_config SET config_value = config_value + 1 WHERE config_key = 'cache_version'";
	return;
}
